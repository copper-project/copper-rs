22const HEADER: u64 = SECTION_HEADER_COMPACT_SIZE as u64;
23const LEASES: usize = 64;
24const IDLE: usize = 1;
25const BUSY: usize = 2;
26const SEALED: usize = 3;
27const LINK_PENDING: usize = 4;

30struct Lease {
31    state: AtomicUsize,
32    next: portable_atomic::AtomicU64,
33    allocated: UnsafeCell<u64>,
34    offset: UnsafeCell<u64>,
35}
36// SAFETY: Only the serialized allocator accesses `offset` and `allocated`.
37unsafe impl Sync for Lease {}
38impl Lease {
39    fn load(&self, ordering: Ordering) -> usize {
40        let state = self.state.load(ordering);
41        #[cfg(all(test, feature = "std"))]
42        tests::lease_event(self, tests::LeaseEvent::Loaded(state));
43        state
44    }
45    fn store(&self, state: usize, ordering: Ordering) {
46        self.state.store(state, ordering);
47    }
48    fn compare_exchange(
49        &self,
50        current: usize,
51        new: usize,
52        success: Ordering,
53        failure: Ordering,
54    ) -> Result<usize, usize> {
55        #[cfg(all(test, feature = "std"))]
56        tests::lease_event(self, tests::LeaseEvent::BeforeCompareExchange);
57        let result = self.state.compare_exchange(current, new, success, failure);
58        #[cfg(all(test, feature = "std"))]
59        tests::lease_event(self, tests::LeaseEvent::AfterCompareExchange);
60        result
61    }
62    fn offset(&self) -> u64 {
63        // SAFETY: Only the owning allocator accesses this field; it serializes allocation.
64        unsafe { *self.offset.get() }
65    }
66    fn set_offset(&self, offset: u64) {
67        // SAFETY: Only the owning allocator accesses this field, through exclusive allocation.
68        unsafe {
69            *self.offset.get() = offset;
70        }
71    }
72    fn next(&self) -> u64 {
73        self.next.load(Ordering::Acquire)
74    }
75    fn set_next(&self, offset: u64) {
76        self.next.store(offset, Ordering::Release);
77    }
78    fn allocated(&self) -> u64 {
79        // SAFETY: Only the serialized allocator accesses this field.
80        unsafe { *self.allocated.get() }
81    }
82    fn set_allocated(&self, allocated: u64) {
83        // SAFETY: Only the serialized allocator accesses this field.
84        unsafe {
85            *self.allocated.get() = allocated;
86        }
87    }
88}

101struct Claims {
102    leases: Arc<LeaseTable>,
103    held: [Option<(usize, usize)>; LEASES],
104}
105impl Claims {
106    fn new(leases: Arc<LeaseTable>) -> Self {
107        Self {
108            leases,
109            held: [None; LEASES],
110        }
111    }
112    fn claim(&mut self, slot: usize) -> CuResult<()> {
113        if self.held.iter().flatten().any(|&(s, _)| s == slot) {
114            return Ok(());
115        }
116        let state = self.leases[slot].load(Ordering::Acquire);
117        if state % 4 == SEALED || state == 0 {
118            return Ok(());
119        }
120        if state % 4 != IDLE {
121            return Err(CuError::from("Space is blocked by a write in progress"));
122        }
123        self.leases[slot]
124            .compare_exchange(
125                state,
126                state / 4 * 4 + BUSY,
127                Ordering::AcqRel,
128                Ordering::Acquire,
129            )
130            .map_err(|_| CuError::from("Space is blocked by a write in progress"))?;
131        *self
132            .held
133            .iter_mut()
134            .find(|s| s.is_none())
135            .ok_or_else(|| CuError::from("Lease claim table is full"))? = Some((slot, state));
136        Ok(())
137    }
138    fn seal(&mut self, slot: usize) {
139        if let Some(held) = self
140            .held
141            .iter_mut()
142            .find(|s| s.is_some_and(|(s, _)| s == slot))
143        {
144            let (_, old) = held.take().unwrap();
145            self.leases[slot].store(old / 4 * 4 + SEALED, Ordering::Release);
146        }
147    }
148}
149impl Drop for Claims {
150    fn drop(&mut self) {
151        for &(slot, old) in self.held.iter().flatten() {
152            self.leases[slot].store(old, Ordering::Release);
153        }
154    }
155}

157/// A bounded byte range owned by a section. Adapters write native encoder output directly.
158pub trait ByteRegion: Send + Sync {
159    fn len(&self) -> u64;
160    fn is_empty(&self) -> bool {
161        self.len() == 0
162    }
163    fn read(&self, offset: u64, bytes: &mut [u8]) -> CuResult<()>;
164    fn write(&mut self, offset: u64, bytes: &[u8]) -> CuResult<()>;
165    fn flush(&mut self) -> CuResult<()>;
166    fn commit(&mut self) -> CuResult<()> {
167        self.flush()
168    }
169    /// Discard a failed write's pending block state. Committed bytes remain readable.
170    fn abort(&mut self) {}
171}

188struct RegionWriter<'a, R> {
189    region: &'a mut R,
190    position: u64,
191    limit: u64,
192}
193impl<R: ByteRegion> Writer for RegionWriter<'_, R> {
194    fn write(&mut self, bytes: &[u8]) -> Result<(), EncodeError> {
195        let end = self
196            .position
197            .checked_add(bytes.len() as u64)
198            .ok_or(EncodeError::UnexpectedEnd)?;
199        if end > self.limit {
200            return Err(EncodeError::UnexpectedEnd);
201        }
202        self.region
203            .write(self.position, bytes)
204            .map_err(encode_error)?;
205        self.position = end;
206        Ok(())
207    }
208}

210struct CompareWriter<'a, R> {
211    region: &'a R,
212    position: u64,
213    limit: u64,
214}
215impl<R: ByteRegion> Writer for CompareWriter<'_, R> {
216    fn write(&mut self, mut bytes: &[u8]) -> Result<(), EncodeError> {
217        if self
218            .position
219            .checked_add(bytes.len() as u64)
220            .is_none_or(|end| end > self.limit)
221        {
222            return Err(EncodeError::Other("Static metadata differs"));
223        }
224        let mut scratch = [0u8; 512];
225        while !bytes.is_empty() {
226            let count = bytes.len().min(scratch.len());
227            self.region
228                .read(self.position, &mut scratch[..count])
229                .map_err(encode_error)?;
230            if scratch[..count] != bytes[..count] {
231                return Err(EncodeError::Other("Static metadata differs"));
232            }
233            self.position += count as u64;
234            bytes = &bytes[count..];
235        }
236        Ok(())
237    }
238}

248fn write_header<R: ByteRegion, E: Encode>(region: &mut R, value: &E) -> CuResult<()> {
249    let mut bytes = [0u8; 512];
250    let count = encode_into_slice(value, &mut bytes, standard())
251        .map_err(|e| CuError::from("Cannot encode log header").add_cause(&e.to_string()))?;
252    region.write(0, &bytes[..count])?;
253    region.commit()
254}

256pub fn read_header<B: ByteStorage>(storage: &B, offset: u64) -> CuResult<SectionHeader> {
257    let mut bytes = [0u8; 512];
258    storage.region(offset, HEADER)?.read(0, &mut bytes)?;
259    let (header, _) = decode_from_slice::<SectionHeader, _>(&bytes, standard())
260        .map_err(|e| CuError::from("Invalid section header").add_cause(&e.to_string()))?;
261    if header.magic != SECTION_MAGIC
262        || header.block_size != SECTION_HEADER_COMPACT_SIZE
263        || header.allocated < HEADER
264        || header.used as u64 > header.allocated - HEADER
265        || !header.allocated.is_multiple_of(HEADER)
266    {
267        return Err(CuError::from("Invalid section size or committed length"));
268    }
269    Ok(header)
270}

272pub fn read_main<B: ByteStorage>(storage: &B) -> CuResult<MainHeader> {
273    let mut bytes = [0u8; 512];
274    storage.region(0, HEADER)?.read(0, &mut bytes)?;
275    let (header, _) = decode_from_slice::<MainHeader, _>(&bytes, standard())
276        .map_err(|e| CuError::from("Invalid main header").add_cause(&e.to_string()))?;
277    if header.magic != MAIN_MAGIC
278        || header.format_version != UNIFIED_LOG_FORMAT_VERSION
279        || header.page_size == 0
280        || !(header.page_size as u64).is_multiple_of(HEADER)
281        || header.sections_begin < header.page_size as u64
282        || header.sections_begin > header.sections_end
283        || header.sections_end > storage.len()
284        || !header.sections_begin.is_multiple_of(HEADER)
285        || !header.sections_end.is_multiple_of(HEADER)
286        || (header.head_section == 0) != (header.tail_section == 0)
287    {
288        return Err(CuError::from(
289            "Invalid log version, alignment or byte bounds",
290        ));
291    }
292    Ok(header)
293}

295/// A section has a generation lease in a fixed startup-allocated table.
296/// Reclamation seals idle leases; concurrent encoding holds BUSY until commit.
297pub struct ByteSection<R: ByteRegion> {
298    region: R,
299    position: u64,
300    header: SectionHeader,
301    leases: Arc<LeaseTable>,
302    slot: usize,
303    generation: usize,
304}
305impl<R: ByteRegion> ByteSection<R> {
306    fn token(&self, state: usize) -> usize {
307        self.generation * GENERATION_STRIDE + state
308    }
309    fn release_write(&mut self, state: usize) -> CuResult<()> {
310        match self.leases[self.slot].compare_exchange(
311            self.token(BUSY),
312            self.token(state),
313            Ordering::AcqRel,
314            Ordering::Acquire,
315        ) {
316            Ok(_) => return Ok(()),
317            Err(actual) if actual == self.token(BUSY) | LINK_PENDING => {}
318            Err(_) => {
319                self.leases.poisoned.store(true, Ordering::Release);
320                return Err(CuError::from("Section lease changed during write"));
321            }
322        }
323        // The allocator handed this encoder the previous tail's link update.
324        // It never touches a header while an encoder owns BUSY.
325        self.header.next_section = self.leases[self.slot].next();
326        let result = write_header(&mut self.region, &self.header);
327        if result.is_err() {
328            self.leases.poisoned.store(true, Ordering::Release);
329        }
330        self.leases[self.slot].store(self.token(state), Ordering::Release);
331        result
332    }
333}
334struct SectionWrite<'a, R: ByteRegion> {
335    section: &'a mut ByteSection<R>,
336    released: bool,
337    observing: bool,
338}
339impl<R: ByteRegion> SectionWrite<'_, R> {
340    fn finish(mut self, state: usize) -> CuResult<()> {
341        self.released = true;
342        self.section.release_write(state)
343    }
344}
345impl<R: ByteRegion> Drop for SectionWrite<'_, R> {
346    fn drop(&mut self) {
347        if !self.released {
348            if self.observing {
349                abort_observed_encode();
350            }
351            self.section.region.abort();
352            let _ = self.section.release_write(IDLE);
353        }
354    }
355}

357impl<R: ByteRegion> SectionStorage for ByteSection<R> {
358    fn initialize<E: Encode>(&mut self, _: &E) -> Result<usize, EncodeError> {
359        Ok(HEADER as usize)
360    }
361    fn post_update_header<E: Encode>(&mut self, _: &E) -> Result<usize, EncodeError> {
362        // The authoritative header is committed with the payload while holding the lease.
363        Ok(HEADER as usize)
364    }
365    fn is_sealed(&self) -> bool {
366        self.leases[self.slot].load(Ordering::Acquire) != self.token(IDLE)
367    }
368    fn append<E: Encode>(&mut self, entry: &E) -> Result<usize, EncodeError> {
369        match self.append_or_rebind(entry)? {
370            SectionAppendOutcome::Appended(count) => Ok(count),
371            SectionAppendOutcome::Rebind => {
372                Err(EncodeError::Other("Section lease is sealed or busy"))
373            }
374        }
375    }
376    fn append_or_rebind<E: Encode>(
377        &mut self,
378        entry: &E,
379    ) -> Result<SectionAppendOutcome, EncodeError> {
380        if self.leases.poisoned.load(Ordering::Acquire) {
381            return Err(EncodeError::Other("Logger faulted after a storage failure"));
382        }
383        if self.leases[self.slot]
384            .compare_exchange(
385                self.token(IDLE),
386                self.token(BUSY),
387                Ordering::AcqRel,
388                Ordering::Acquire,
389            )
390            .is_err()
391        {
392            return Ok(SectionAppendOutcome::Rebind);
393        }
394        let guard = SectionWrite {
395            section: self,
396            released: false,
397            observing: true,
398        };
399        let section = &mut *guard.section;
400        begin_observed_encode();
401        let old_position = section.position;
402        let result = (|| {
403            let writer = RegionWriter {
404                region: &mut section.region,
405                position: old_position,
406                limit: section.header.allocated,
407            };
408            let mut encoder = EncoderImpl::new(ObservedWriter::new(writer), standard());
409            entry.encode(&mut encoder)?;
410            let end = encoder.into_writer().into_inner().position;
411            section.region.commit().map_err(encode_error)?;
412            let mut committed = section.header.clone();
413            committed.next_section = section.leases[section.slot].next();
414            committed.used = u32::try_from(end - HEADER).map_err(|_| EncodeError::UnexpectedEnd)?;
415            if let Err(error) = write_header(&mut section.region, &committed) {
416                section.leases.poisoned.store(true, Ordering::Release);
417                return Err(encode_error(error));
418            }
419            section.header = committed;
420            section.position = end;
421            Ok((end - old_position) as usize)
422        })();
423        match &result {
424            Ok(count) => {
425                let observed = finish_observed_encode();
426                debug_assert_eq!(*count, observed);
427            }
428            Err(_) => {
429                abort_observed_encode();
430                section.region.abort();
431            }
432        }
433        let release = guard.finish(IDLE).map_err(encode_error);
434        result.and_then(|count| release.map(|()| SectionAppendOutcome::Appended(count)))
435    }
436    fn flush(&mut self) -> CuResult<usize> {
437        if self.leases[self.slot]
438            .compare_exchange(
439                self.token(IDLE),
440                self.token(BUSY),
441                Ordering::AcqRel,
442                Ordering::Acquire,
443            )
444            .is_ok()
445        {
446            let guard = SectionWrite {
447                section: self,
448                released: false,
449                observing: false,
450            };
451            let section = &mut *guard.section;
452            let mut header = section.header.clone();
453            header.next_section = section.leases[section.slot].next();
454            header.is_open = false;
455            let result =
456                write_header(&mut section.region, &header).and_then(|()| section.region.flush());
457            let state = if result.is_ok() {
458                section.header = header;
459                SEALED
460            } else {
461                section.leases.poisoned.store(true, Ordering::Release);
462                IDLE
463            };
464            let release = guard.finish(state);
465            result.and(release)?;
466        }
467        Ok(self.position as usize)
468    }
469}

471/// One allocator implements both backing adapters. Section bookkeeping is fixed-capacity.
472pub struct ByteLogger<B: ByteStorage> {
473    storage: B,
474    header: MainHeader,
475    policy: CapacityPolicy,
476    capacity: u64,
477    leases: Arc<LeaseTable>,
478    generation: usize,
479    next_run_id: u64,
480    retained_space: u64,
481    append_pending: bool,
482    faulted: bool,
483}
484impl<B: ByteStorage> ByteLogger<B> {
485    pub fn new(
486        mut storage: B,
487        alignment: u16,
488        policy: CapacityPolicy,
489        capacity: u64,
490    ) -> CuResult<Self> {
491        if alignment == 0
492            || !(alignment as u64).is_multiple_of(HEADER)
493            || capacity <= alignment as u64
494            || !capacity.is_multiple_of(HEADER)
495        {
496            return Err(CuError::from(
497                "Invalid log capacity or allocation alignment",
498            ));
499        }
500        storage.grow(capacity)?;
501        let header = MainHeader {
502            magic: MAIN_MAGIC,
503            format_version: UNIFIED_LOG_FORMAT_VERSION,
504            page_size: alignment,
505            metadata_offset: 0,
506            sections_begin: alignment as u64,
507            sections_end: capacity,
508            head_section: 0,
509            tail_section: 0,
510            clean_close: false,
511        };
512        let mut logger = Self::from_parts(storage, header, policy, capacity, false);
513        logger.publish()?;
514        Ok(logger)
515    }
516    fn from_parts(
517        storage: B,
518        header: MainHeader,
519        policy: CapacityPolicy,
520        capacity: u64,
521        append_pending: bool,
522    ) -> Self {
523        Self {
524            storage,
525            header,
526            policy,
527            capacity,
528            leases: Arc::new(LeaseTable {
529                slots: core::array::from_fn(|_| Lease {
530                    state: AtomicUsize::new(0),
531                    next: portable_atomic::AtomicU64::new(0),
532                    allocated: UnsafeCell::new(0),
533                    offset: UnsafeCell::new(0),
534                }),
535                poisoned: AtomicBool::new(false),
536            }),
537            generation: 0,
538            next_run_id: 1,
539            retained_space: 0,
540            append_pending,
541            faulted: false,
542        }
543    }
544    /// Read-only validation. No storage extension or header write precedes metadata comparison.
545    pub fn append(storage: B, policy: CapacityPolicy, capacity: u64) -> CuResult<Self> {
546        let header = read_main(&storage)?;
547        if !header.clean_close {
548            return Err(CuError::from("Cannot append: log was not cleanly closed"));
549        }
550        if !capacity.is_multiple_of(HEADER) || capacity <= header.sections_begin {
551            return Err(CuError::from("Invalid append capacity"));
552        }
553        let mut logger = Self::from_parts(storage, header, policy, capacity, true);
554        let mut cursor = logger.header.head_section;
555        let mut remaining = (logger.header.sections_end - logger.header.sections_begin) / HEADER;
556        while cursor != 0 {
557            if remaining == 0 {
558                return Err(CuError::from("Cyclic section links"));
559            }
560            remaining -= 1;
561            let section = logger.data_header(cursor)?;
562            if cursor + section.allocated > capacity {
563                return Err(CuError::from(
564                    "Cannot shrink storage beneath retained sections",
565                ));
566            }
567            if section.is_open {
568                return Err(CuError::from("Cannot append: retained section is open"));
569            }
570            logger.retained_space = logger
571                .retained_space
572                .checked_add(section.allocated)
573                .filter(|used| *used <= logger.header.sections_end - logger.header.sections_begin)
574                .ok_or_else(|| CuError::from("Retained sections exceed data capacity"))?;
575            logger.next_run_id = logger.next_run_id.max(
576                section
577                    .context
578                    .run_id
579                    .checked_add(1)
580                    .ok_or_else(|| CuError::from("Run ID exhausted"))?,
581            );
582            if section.next_section == 0 && cursor != logger.header.tail_section {
583                return Err(CuError::from("Invalid tail section"));
584            }
585            if cursor == logger.header.tail_section && section.next_section != 0 {
586                return Err(CuError::from("Invalid tail link"));
587            }
588            cursor = section.next_section;
589        }
590        logger.validate_metadata()?;
591        Ok(logger)
592    }
593    fn validate_metadata(&self) -> CuResult<()> {
594        if self.header.metadata_offset == 0 {
595            return Ok(());
596        }
597        let offset = self.header.metadata_offset;
598        if offset != self.header.page_size as u64 {
599            return Err(CuError::from("Invalid metadata offset"));
600        }
601        let h = read_header(&self.storage, offset)?;
602        if h.is_open
603            || h.entry_type != UnifiedLogType::ApplicationMetadata
604            || offset
605                .checked_add(h.allocated)
606                .is_none_or(|end| end > self.header.sections_begin)
607        {
608            return Err(CuError::from("Invalid application metadata section"));
609        }
610        if h.next_section != 0 {
611            let c = read_header(&self.storage, h.next_section)?;
612            if c.is_open
613                || c.entry_type != UnifiedLogType::ValueDecodeCatalog
614                || c.next_section != 0
615                || h.next_section != offset + h.allocated
616                || h.next_section.checked_add(c.allocated) != Some(self.header.sections_begin)
617            {
618                return Err(CuError::from("Invalid catalog metadata section"));
619            }
620        } else if offset + h.allocated != self.header.sections_begin {
621            return Err(CuError::from("Invalid metadata bounds"));
622        }
623        Ok(())
624    }
625    fn section_extent(&self, offset: u64) -> CuResult<(u64, u64)> {
626        if let Some(lease) = self.leases.iter().find(|lease| lease.offset() == offset) {
627            return Ok((lease.allocated(), lease.next()));
628        }
629        let header = self.data_header_unlocked(offset)?;
630        Ok((header.allocated, header.next_section))
631    }
632    fn data_header(&self, offset: u64) -> CuResult<SectionHeader> {
633        let mut claims = Claims::new(self.leases.clone());
634        if let Some(slot) = self
635            .leases
636            .iter()
637            .position(|lease| lease.offset() == offset)
638        {
639            claims.claim(slot)?;
640        }
641        self.data_header_unlocked(offset)
642    }
643    fn data_header_unlocked(&self, offset: u64) -> CuResult<SectionHeader> {
644        if offset < self.header.sections_begin
645            || offset >= self.header.sections_end
646            || !offset.is_multiple_of(HEADER)
647        {
648            return Err(CuError::from("Section offset is outside the data region"));
649        }
650        let h = read_header(&self.storage, offset)?;
651        if offset
652            .checked_add(h.allocated)
653            .is_none_or(|end| end > self.header.sections_end)
654            || (h.next_section != 0
655                && (h.next_section < self.header.sections_begin
656                    || h.next_section >= self.header.sections_end
657                    || !h.next_section.is_multiple_of(HEADER)))
658        {
659            return Err(CuError::from("Invalid section bounds or next link"));
660        }
661        Ok(h)
662    }
663    fn publish(&mut self) -> CuResult<()> {
664        let result = (|| {
665            let mut region = self.storage.region(0, HEADER)?;
666            write_header(&mut region, &self.header)?;
667            region.flush()
668        })();
669        if result.is_err() {
670            self.faulted = true;
671        }
672        result
673    }
674    fn activate(&mut self) -> CuResult<()> {
675        if self.faulted || self.leases.poisoned.load(Ordering::Acquire) {
676            return Err(CuError::from("Logger faulted after a storage failure"));
677        }
678        if self.append_pending {
679            self.storage.grow(self.capacity)?;
680            self.header.sections_end = self.capacity;
681            self.header.clean_close = false;
682            self.publish()?;
683            self.append_pending = false;
684        }
685        Ok(())
686    }
687    fn close_idle(&mut self, offset: u64) -> CuResult<()> {
688        if let Some(slot) = self
689            .leases
690            .iter()
691            .position(|lease| lease.offset() == offset)
692        {
693            let state = self.leases[slot].load(Ordering::Acquire);
694            if state % 4 == BUSY {
695                return Err(CuError::from("Space is blocked by a write in progress"));
696            }
697            if state % 4 == IDLE {
698                self.leases[slot]
699                    .compare_exchange(
700                        state,
701                        state / 4 * 4 + SEALED,
702                        Ordering::AcqRel,
703                        Ordering::Acquire,
704                    )
705                    .map_err(|_| CuError::from("Space is blocked by a write in progress"))?;
706                let mut h = self.data_header(offset)?;
707                h.is_open = false;
708                write_header(&mut self.storage.region(offset, HEADER)?, &h)?;
709                self.storage.region(offset, h.allocated)?.flush()?;
710            }
711        }
712        if self.data_header(offset)?.is_open {
713            return Err(CuError::from("Space is blocked by an open section"));
714        }
715        Ok(())
716    }
717    fn compare<E: Encode>(&self, offset: u64, kind: UnifiedLogType, value: &E) -> CuResult<()> {
718        let h = read_header(&self.storage, offset)?;
719        if h.entry_type != kind {
720            return Err(CuError::from("Static metadata section type differs"));
721        }
722        let region = self.storage.region(offset + HEADER, h.used as u64)?;
723        let writer = CompareWriter {
724            region: &region,
725            position: 0,
726            limit: h.used as u64,
727        };
728        let mut encoder = EncoderImpl::new(writer, standard());
729        value.encode(&mut encoder).map_err(|_| CuError::from("Cannot append: static metadata does not match; appending different applications or versions to the same log is not supported."))?;
730        if encoder.into_writer().position != h.used as u64 {
731            return Err(CuError::from(
732                "Cannot append: static metadata does not match; appending different applications or versions to the same log is not supported.",
733            ));
734        }
735        Ok(())
736    }
737    fn write_static<E: Encode>(
738        &mut self,
739        offset: u64,
740        allocated: u64,
741        next: u64,
742        kind: UnifiedLogType,
743        value: &E,
744    ) -> CuResult<()> {
745        let mut region = self.storage.region(offset, allocated)?;
746        let writer = RegionWriter {
747            region: &mut region,
748            position: HEADER,
749            limit: allocated,
750        };
751        let mut encoder = EncoderImpl::new(writer, standard());
752        value.encode(&mut encoder).map_err(|e| {
753            CuError::from("Cannot encode static metadata").add_cause(&e.to_string())
754        })?;
755        let used = encoder.into_writer().position - HEADER;
756        region.flush()?;
757        write_header(
758            &mut region,
759            &SectionHeader {
760                entry_type: kind,
761                allocated,
762                next_section: next,
763                used: u32::try_from(used)
764                    .map_err(|_| CuError::from("Metadata exceeds section length limit"))?,
765                is_open: false,
766                ..SectionHeader::default()
767            },
768        )
769    }
770    pub fn close(&mut self) -> CuResult<()> {
771        if self.append_pending {
772            return Ok(());
773        }
774        if self.faulted || self.leases.poisoned.load(Ordering::Acquire) {
775            return Err(CuError::from("Cannot cleanly close a faulted logger"));
776        }
777        self.faulted = true;
778        let mut cursor = self.header.head_section;
779        let mut remaining = (self.header.sections_end - self.header.sections_begin) / HEADER;
780        while cursor != 0 {
781            if remaining == 0 {
782                return Err(CuError::from("Cyclic section links prevent clean close"));
783            }
784            remaining -= 1;
785            self.close_idle(cursor)?;
786            cursor = self.data_header(cursor)?.next_section;
787        }
788        self.header.clean_close = true;
789        self.publish()?;
790        self.faulted = false;
791        Ok(())
792    }
793}
794fn aligned(size: u64) -> CuResult<u64> {
795    size.checked_add(HEADER - 1)
796        .map(|s| s / HEADER * HEADER)
797        .ok_or_else(|| CuError::from("Section size overflow"))
798}

800    fn seal_metadata<C: Encode>(
801        &mut self,
802        metadata: &ApplicationMetadata,
803        catalog: Option<&C>,
804    ) -> CuResult<()> {
805        #[derive(Encode)]
806        struct MetadataBody<'a> {
807            app_type: &'a str,
808            app_name: &'a str,
809            app_version: &'a str,
810            git_commit: Option<&'a str>,
811            git_dirty: Option<bool>,
812            subsystem_id: Option<&'a str>,
813            subsystem_code: u16,
814            effective_config_ron: &'a str,
815            missions: &'a [alloc::string::String],
816            catalog_offset: u64,
817        }
818        let mut metadata = MetadataBody {
819            app_type: &metadata.app_type,
820            app_name: &metadata.app_name,
821            app_version: &metadata.app_version,
822            git_commit: metadata.git_commit.as_deref(),
823            git_dirty: metadata.git_dirty,
824            subsystem_id: metadata.subsystem_id.as_deref(),
825            subsystem_code: metadata.subsystem_code,
826            effective_config_ron: &metadata.effective_config_ron,
827            missions: &metadata.missions,
828            catalog_offset: 0,
829        };
830        let base = self.header.page_size as u64;
831        // The offset changes bincode's varint width; sizing converges monotonically.
832        let mut app_size = 0;
833        for _ in 0..8 {
834            let size = aligned(HEADER + encoded_size(&metadata)?)?;
835            let offset = if catalog.is_some() { base + size } else { 0 };
836            app_size = size;
837            if metadata.catalog_offset == offset {
838                break;
839            }
840            metadata.catalog_offset = offset;
841        }
842        let catalog_size = catalog
843            .map(|c| encoded_size(c).and_then(|s| aligned(HEADER + s)))
844            .transpose()?
845            .unwrap_or(0);
846        let end = base
847            .checked_add(app_size)
848            .and_then(|s| s.checked_add(catalog_size))
849            .ok_or_else(|| CuError::from("Metadata size overflow"))?;
850        if self.header.metadata_offset != 0 {
851            self.compare(base, UnifiedLogType::ApplicationMetadata, &metadata)?;
852            if let Some(catalog) = catalog {
853                self.compare(
854                    metadata.catalog_offset,
855                    UnifiedLogType::ValueDecodeCatalog,
856                    catalog,
857                )?;
858            }
859            self.activate()?;
860            return Ok(());
861        }
862        if self.header.head_section != 0 || self.header.sections_begin != base {
863            return Err(CuError::from("Static metadata region is already sealed"));
864        }
865        if end + HEADER > self.header.sections_end && self.policy != CapacityPolicy::Grow {
866            return Err(CuError::from("Static metadata exceeds log capacity"));
867        }
868        self.storage.grow(end + HEADER)?;
869        self.activate()?;
870        self.faulted = true;
871        self.write_static(
872            base,
873            app_size,
874            metadata.catalog_offset,
875            UnifiedLogType::ApplicationMetadata,
876            &metadata,
877        )?;
878        if let Some(catalog) = catalog {
879            self.write_static(
880                metadata.catalog_offset,
881                catalog_size,
882                0,
883                UnifiedLogType::ValueDecodeCatalog,
884                catalog,
885            )?;
886        }
887        self.header.metadata_offset = base;
888        self.header.sections_begin = end;
889        if end + HEADER > self.header.sections_end {
890            self.header.sections_end = self.storage.len();
891        }
892        self.publish()?;
893        self.faulted = false;
894        Ok(())
895    }

914    fn add_section(
915        &mut self,
916        kind: UnifiedLogType,
917        size: usize,
918    ) -> CuResult<SectionHandle<ByteSection<B::Region>>> {

921    fn add_section_with_context(
922        &mut self,
923        kind: UnifiedLogType,
924        size: usize,
925        context: SectionContext,
926    ) -> CuResult<SectionHandle<ByteSection<B::Region>>> {
927        if self.append_pending && self.header.metadata_offset != 0 {
928            return Err(CuError::from(
929                "Compare static metadata before appending data",
930            ));
931        }
932        let size = aligned(size as u64)?;
933        if size <= HEADER || size - HEADER > u32::MAX as u64 {
934            return Err(CuError::from("Invalid section allocation size"));
935        }
936        let capacity = self.header.sections_end - self.header.sections_begin;
937        if self.policy != CapacityPolicy::Grow && size > capacity {
938            return Err(CuError::from("Section exceeds log capacity"));
939        }
940        let tail = self.header.tail_section;
941        let mut position = if tail == 0 {
942            self.header.sections_begin
943        } else {
944            tail + self.section_extent(tail)?.0
945        };
946        if self.policy == CapacityPolicy::OverwriteOldest
947            && position
948                .checked_add(size)
949                .is_none_or(|end| end > self.header.sections_end)
950        {
951            position = self.header.sections_begin;
952        }
953        let overlaps =
954            |offset: u64, allocated: u64| position < offset + allocated && offset < position + size;
955        let mut last_overlap = 0;
956        let mut cursor = self.header.head_section;
957        let mut remaining = capacity / HEADER;
958        while cursor != 0 {
959            if remaining == 0 {
960                return Err(CuError::from("Cyclic section links"));
961            }
962            remaining -= 1;
963            let (allocated, next) = self.section_extent(cursor)?;
964            if overlaps(cursor, allocated) {
965                last_overlap = cursor;
966            }
967            cursor = next;
968        }
969        if self.policy == CapacityPolicy::Grow
970            && (last_overlap != 0 || position + size > self.header.sections_end)
971        {
972            position = self.header.sections_end;
973            last_overlap = 0;
974        }
975        if self.policy == CapacityPolicy::StopWhenFull
976            && (last_overlap != 0 || position + size > self.header.sections_end)
977        {
978            return Err(CuError::from("Log is full"));
979        }
980        // Claim every affected encoder before any write. A failed preflight restores all leases.
981        let mut claims = Claims::new(self.leases.clone());
982        let mut cursor = self.header.head_section;
983        if last_overlap != 0 {
984            loop {
985                if let Some(slot) = self
986                    .leases
987                    .iter()
988                    .position(|lease| lease.offset() == cursor)
989                {
990                    claims.claim(slot)?;
991                } else if self.data_header_unlocked(cursor)?.is_open {
992                    return Err(CuError::from("Space is blocked by an open section"));
993                }
994                if cursor == last_overlap {
995                    break;
996                }
997                cursor = self.data_header_unlocked(cursor)?.next_section;
998            }
999        }
1000        let free_slot = self
1001            .leases
1002            .iter()
1003            .position(|s| matches!(s.load(Ordering::Acquire) % 4, 0 | SEALED));
1004        if free_slot.is_none() && last_overlap == 0 {
1005            return Err(CuError::from(
1006                "Too many concurrent section streams (maximum 64)",
1007            ));
1008        }
1009        // Validate the mapping geometry before reclamation or header changes.
1010        if position + size <= self.storage.len() {
1011            let _ = self.storage.region(position, size)?;
1012        }
1013        self.activate()?;
1014        self.faulted = true;
1015        if position + size > self.header.sections_end {
1016            self.storage.grow(position + size)?;
1017            self.header.sections_end = self.storage.len();
1018        }
1019        if last_overlap != 0 {
1020            let mut cursor = self.header.head_section;
1021            loop {
1022                let mut h = self.data_header_unlocked(cursor)?;
1023                h.is_open = false;
1024                write_header(&mut self.storage.region(cursor, HEADER)?, &h)?;
1025                self.storage.region(cursor, h.allocated)?.flush()?;
1026                if let Some(slot) = self
1027                    .leases
1028                    .iter()
1029                    .position(|lease| lease.offset() == cursor)
1030                {
1031                    claims.seal(slot);
1032                }
1033                self.retained_space -= h.allocated;
1034                let next = h.next_section;
1035                if cursor == last_overlap {
1036                    self.header.head_section = next;
1037                    if next == 0 {
1038                        self.header.tail_section = 0;
1039                    }
1040                    break;
1041                }
1042                cursor = next;
1043            }
1044            self.publish()?;
1045        }
1046        let slot = self
1047            .leases
1048            .iter()
1049            // A writer may finish between loads. Classify one snapshot and reuse
1050            // only unassigned or sealed slots; idle streams retain their generation.
1051            .position(|s| matches!(s.load(Ordering::Acquire) % 4, 0 | SEALED))
1052            .ok_or_else(|| CuError::from("Too many concurrent section writes"))?;
1053        self.generation = self
1054            .generation
1055            .checked_add(1)
1056            .filter(|g| *g < usize::MAX / GENERATION_STRIDE)
1057            .ok_or_else(|| CuError::from("Section generation exhausted"))?;
1058        let header = SectionHeader {
1059            entry_type: kind,
1060            allocated: size,
1061            context,
1062            ..SectionHeader::default()
1063        };
1064        let mut region = self.storage.region(position, size)?;
1065        write_header(&mut region, &header)?;
1066        if self.header.tail_section != 0 {
1067            let tail = self.header.tail_section;
1068            if let Some(slot) = self.leases.iter().position(|lease| lease.offset() == tail) {
1069                let lease = &self.leases[slot];
1070                lease.set_next(position);
1071                let state = lease.state.fetch_or(LINK_PENDING, Ordering::AcqRel);
1072                if state % 4 != BUSY {
1073                    // An idle/sealed encoder cannot start while LINK_PENDING is set.
1074                    let mut previous = self.data_header_unlocked(tail)?;
1075                    previous.next_section = position;
1076                    write_header(&mut self.storage.region(tail, HEADER)?, &previous)?;
1077                    lease.store(state, Ordering::Release);
1078                }
1079                // A busy encoder publishes the link before releasing its lease.
1080            } else {
1081                let mut previous = self.data_header_unlocked(tail)?;
1082                previous.next_section = position;
1083                write_header(&mut self.storage.region(tail, HEADER)?, &previous)?;
1084            }
1085        }
1086        if self.header.head_section == 0 {
1087            self.header.head_section = position;
1088        }
1089        self.header.tail_section = position;
1090        self.header.clean_close = false;
1091        self.publish()?;
1092        self.retained_space += size;
1093        for lease in self.leases.iter() {
1094            if lease.offset() == position {
1095                lease.set_offset(0);
1096            }
1097        }
1098        self.leases[slot].set_offset(position);
1099        self.leases[slot].set_allocated(size);
1100        self.leases[slot].set_next(0);
1101        self.leases[slot].store(
1102            self.generation * GENERATION_STRIDE + IDLE,
1103            Ordering::Release,
1104        );
1105        let storage = ByteSection {
1106            region,
1107            position: HEADER,
1108            header: header.clone(),
1109            leases: self.leases.clone(),
1110            slot,
1111            generation: self.generation,
1112        };
1113        let handle = SectionHandle::create(header, storage)?;
1114        self.faulted = false;
1115        Ok(handle)
1116    }

1278                    LogStream::new(UnifiedLogType::CopperList, logger.clone(), 1024).unwrap();
1279                if !empty {
1280                    stream.log(&41u32).unwrap();
1281                }
1282                let section = &stream.current_section.storage;
1283                let leases = section.leases.clone();
1284                let slot = section.slot;
1285                let idle = section.token(IDLE);
1286                let injected = Arc::new(AtomicBool::new(false));
1287                let observed = injected.clone();
1288                let mut pending = false;
1289                LEASE_HOOK.with(|cell| {
1290                    *cell.borrow_mut() = Some(Box::new(move |lease, event| {
1291                        if !core::ptr::eq(lease, &leases[slot]) {
1292                            return;
1293                        }
1294                        match event {
1295                            LeaseEvent::BeforeCompareExchange
1296                                if !observed.load(Ordering::Relaxed) =>
1297                            {
1298                                // Allocation temporarily owns the idle tail's header.
1299                                lease.store(idle | LINK_PENDING, Ordering::Release);
1300                                pending = true;
1301                                observed.store(true, Ordering::Relaxed);
1302                            }
1303                            LeaseEvent::AfterCompareExchange if pending => {
1304                                // Publication completes before the stream handles the failed CAS.
1305                                lease.store(idle, Ordering::Release);
1306                                pending = false;
1307                            }
1308                            _ => {}
1309                        }
1310                    }));
1311                });
1312                let encodes = Cell::new(0);
1313                let result = stream.log(&Counted(&encodes));
1314                LEASE_HOOK.with(|cell| cell.borrow_mut().take());
1315                assert!(injected.load(Ordering::Relaxed));
1316                result.unwrap();
1317                assert_eq!(encodes.get(), 1, "rebinding must precede encoding");
1318                drop(stream);
1319                drop(logger);
1320                assert_eq!(
1321                    saved_values(&path),
1322                    if empty { vec![42] } else { vec![41, 42] }
1323                );
1324            }
1325        }
1326    }

1329    fn test_reclamation_before_acquisition_rebinds_a_stale_stream() {
1330        let dir = tempfile::tempdir().unwrap();
1331        let path = dir.path().join("revocation.copper");
1332        let logger = Arc::new(Mutex::new(logger(&path, CapacityPolicy::OverwriteOldest)));
1333        let mut stream = LogStream::new(UnifiedLogType::CopperList, logger.clone(), 1024).unwrap();
1334        stream.log(&41u32).unwrap();
1335        // Leave the next allocation to wrap and reclaim the idle first stream.
1336        for value in 0..6u32 {
1337            let mut logger = logger.lock().unwrap();
1338            let mut section = logger
1339                .add_section(UnifiedLogType::CopperList, 1024)
1340                .unwrap();
1341            section.append(value).unwrap();
1342            logger.try_flush_section(&mut section).unwrap();
1343        }
1344        let first = &stream.current_section.storage;
1345        let leases = first.leases.clone();
1346        let slot = first.slot;
1347        let parent = logger.clone();
1348        let mut injected = false;
1349        LEASE_HOOK.with(|cell| {
1350            *cell.borrow_mut() = Some(Box::new(move |lease, event| {
1351                if core::ptr::eq(lease, &leases[slot])
1352                    && matches!(event, LeaseEvent::BeforeCompareExchange)
1353                    && !injected
1354                {
1355                    injected = true;
1356                    let mut logger = parent.lock().unwrap();
1357                    let mut section = logger
1358                        .add_section(UnifiedLogType::CopperList, 1024)
