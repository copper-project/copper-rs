22const HEADER: u64 = SECTION_HEADER_COMPACT_SIZE as u64;
23const LEASES: usize = 64;
24const IDLE: usize = 1;
25const BUSY: usize = 2;
26const SEALED: usize = 3;

28struct Lease {
29    state: AtomicUsize,
30    next: UnsafeCell<u64>,
31    offset: UnsafeCell<u64>,
32}
33// SAFETY: `next` is accessed only while the corresponding state is exclusively BUSY.
34unsafe impl Sync for Lease {}
35impl Lease {
36    fn load(&self, ordering: Ordering) -> usize {
37        self.state.load(ordering)
38    }
39    fn store(&self, state: usize, ordering: Ordering) {
40        self.state.store(state, ordering);
41    }
42    fn compare_exchange(
43        &self,
44        current: usize,
45        new: usize,
46        success: Ordering,
47        failure: Ordering,
48    ) -> Result<usize, usize> {
49        self.state.compare_exchange(current, new, success, failure)
50    }
51    fn offset(&self) -> u64 {
52        // SAFETY: Only the owning allocator accesses this field; it serializes allocation.
53        unsafe { *self.offset.get() }
54    }
55    fn set_offset(&self, offset: u64) {
56        // SAFETY: Only the owning allocator accesses this field, through exclusive allocation.
57        unsafe {
58            *self.offset.get() = offset;
59        }
60    }
61    fn next(&self) -> u64 {
62        // SAFETY: The caller owns this lease in BUSY state.
63        unsafe { *self.next.get() }
64    }
65    fn set_next(&self, offset: u64) {
66        // SAFETY: The caller owns BUSY, or initializes a generation before publishing IDLE.
67        unsafe {
68            *self.next.get() = offset;
69        }
70    }
71}

84struct Claims {
85    leases: Arc<LeaseTable>,
86    held: [Option<(usize, usize)>; LEASES],
87}
88impl Claims {
89    fn new(leases: Arc<LeaseTable>) -> Self {
90        Self {
91            leases,
92            held: [None; LEASES],
93        }
94    }
95    fn claim(&mut self, slot: usize) -> CuResult<()> {
96        if self.held.iter().flatten().any(|&(s, _)| s == slot) {
97            return Ok(());
98        }
99        let state = self.leases[slot].load(Ordering::Acquire);
100        if state % 4 == SEALED || state == 0 {
101            return Ok(());
102        }
103        if state % 4 != IDLE {
104            return Err(CuError::from("Space is blocked by a write in progress"));
105        }
106        self.leases[slot]
107            .compare_exchange(
108                state,
109                state / 4 * 4 + BUSY,
110                Ordering::AcqRel,
111                Ordering::Acquire,
112            )
113            .map_err(|_| CuError::from("Space is blocked by a write in progress"))?;
114        *self
115            .held
116            .iter_mut()
117            .find(|s| s.is_none())
118            .ok_or_else(|| CuError::from("Lease claim table is full"))? = Some((slot, state));
119        Ok(())
120    }
121    fn seal(&mut self, slot: usize) {
122        if let Some(held) = self
123            .held
124            .iter_mut()
125            .find(|s| s.is_some_and(|(s, _)| s == slot))
126        {
127            let (_, old) = held.take().unwrap();
128            self.leases[slot].store(old / 4 * 4 + SEALED, Ordering::Release);
129        }
130    }
131}
132impl Drop for Claims {
133    fn drop(&mut self) {
134        for &(slot, old) in self.held.iter().flatten() {
135            self.leases[slot].store(old, Ordering::Release);
136        }
137    }
138}

140/// A bounded byte range owned by a section. Adapters write native encoder output directly.
141pub trait ByteRegion: Send + Sync {
142    fn len(&self) -> u64;
143    fn is_empty(&self) -> bool {
144        self.len() == 0
145    }
146    fn read(&self, offset: u64, bytes: &mut [u8]) -> CuResult<()>;
147    fn write(&mut self, offset: u64, bytes: &[u8]) -> CuResult<()>;
148    fn flush(&mut self) -> CuResult<()>;
149    fn commit(&mut self) -> CuResult<()> {
150        self.flush()
151    }
152    /// Discard a failed write's pending block state. Committed bytes remain readable.
153    fn abort(&mut self) {}
154}

164    fn region(&self, offset: u64, len: u64) -> CuResult<Self::Region>;

167fn encode_error(_: CuError) -> EncodeError {
168    EncodeError::Other("Storage write failed")

171struct RegionWriter<'a, R> {
172    region: &'a mut R,
173    position: u64,
174    limit: u64,
175}
176impl<R: ByteRegion> Writer for RegionWriter<'_, R> {
177    fn write(&mut self, bytes: &[u8]) -> Result<(), EncodeError> {
178        let end = self
179            .position
180            .checked_add(bytes.len() as u64)
181            .ok_or(EncodeError::UnexpectedEnd)?;
182        if end > self.limit {
183            return Err(EncodeError::UnexpectedEnd);
184        }
185        self.region
186            .write(self.position, bytes)
187            .map_err(encode_error)?;
188        self.position = end;
189        Ok(())
190    }
191}

193struct CompareWriter<'a, R> {
194    region: &'a R,
195    position: u64,
196    limit: u64,
197}
198impl<R: ByteRegion> Writer for CompareWriter<'_, R> {
199    fn write(&mut self, mut bytes: &[u8]) -> Result<(), EncodeError> {
200        if self
201            .position
202            .checked_add(bytes.len() as u64)
203            .is_none_or(|end| end > self.limit)
204        {
205            return Err(EncodeError::Other("Static metadata differs"));
206        }
207        let mut scratch = [0u8; 512];
208        while !bytes.is_empty() {
209            let count = bytes.len().min(scratch.len());
210            self.region
211                .read(self.position, &mut scratch[..count])
212                .map_err(encode_error)?;
213            if scratch[..count] != bytes[..count] {
214                return Err(EncodeError::Other("Static metadata differs"));
215            }
216            self.position += count as u64;
217            bytes = &bytes[count..];
218        }
219        Ok(())
220    }
221}

223pub fn encoded_size<E: Encode>(value: &E) -> CuResult<u64> {

225    value

231fn write_header<R: ByteRegion, E: Encode>(region: &mut R, value: &E) -> CuResult<()> {
232    let mut bytes = [0u8; 512];
233    let count = encode_into_slice(value, &mut bytes, standard())
234        .map_err(|e| CuError::from("Cannot encode log header").add_cause(&e.to_string()))?;
235    region.write(0, &bytes[..count])?;
236    region.commit()
237}

239pub fn read_header<B: ByteStorage>(storage: &B, offset: u64) -> CuResult<SectionHeader> {
240    let mut bytes = [0u8; 512];
241    storage.region(offset, HEADER)?.read(0, &mut bytes)?;
242    let (header, _) = decode_from_slice::<SectionHeader, _>(&bytes, standard())
243        .map_err(|e| CuError::from("Invalid section header").add_cause(&e.to_string()))?;
244    if header.magic != SECTION_MAGIC
245        || header.block_size != SECTION_HEADER_COMPACT_SIZE
246        || header.allocated < HEADER
247        || header.used as u64 > header.allocated - HEADER
248        || !header.allocated.is_multiple_of(HEADER)
249    {
250        return Err(CuError::from("Invalid section size or committed length"));
251    }
252    Ok(header)
253}
254
255pub fn read_main<B: ByteStorage>(storage: &B) -> CuResult<MainHeader> {
256    let mut bytes = [0u8; 512];
257    storage.region(0, HEADER)?.read(0, &mut bytes)?;
258    let (header, _) = decode_from_slice::<MainHeader, _>(&bytes, standard())
259        .map_err(|e| CuError::from("Invalid main header").add_cause(&e.to_string()))?;
260    if header.magic != MAIN_MAGIC
261        || header.format_version != UNIFIED_LOG_FORMAT_VERSION
262        || header.page_size == 0
263        || !(header.page_size as u64).is_multiple_of(HEADER)
264        || header.sections_begin < header.page_size as u64
265        || header.sections_begin > header.sections_end
266        || header.sections_end > storage.len()
267        || !header.sections_begin.is_multiple_of(HEADER)
268        || !header.sections_end.is_multiple_of(HEADER)
269        || (header.head_section == 0) != (header.tail_section == 0)
270    {
271        return Err(CuError::from(
272            "Invalid log version, alignment or byte bounds",
273        ));
274    }
275    Ok(header)
276}

278/// A section has a generation lease in a fixed startup-allocated table.
279/// Reclamation seals idle leases; concurrent encoding holds BUSY until commit.
280pub struct ByteSection<R: ByteRegion> {
281    region: R,
282    position: u64,
283    header: SectionHeader,
284    leases: Arc<LeaseTable>,
285    slot: usize,
286    generation: usize,
287}
288impl<R: ByteRegion> ByteSection<R> {
289    fn token(&self, state: usize) -> usize {
290        self.generation * 4 + state
291    }
292}
293impl<R: ByteRegion> SectionStorage for ByteSection<R> {
294    fn initialize<E: Encode>(&mut self, _: &E) -> Result<usize, EncodeError> {
295        Ok(HEADER as usize)
296    }
297    fn post_update_header<E: Encode>(&mut self, _: &E) -> Result<usize, EncodeError> {
298        // The authoritative header is committed with the payload while holding the lease.
299        Ok(HEADER as usize)
300    }
301    fn is_sealed(&self) -> bool {
302        self.leases[self.slot].load(Ordering::Acquire) != self.token(IDLE)
303    }
304    fn append<E: Encode>(&mut self, entry: &E) -> Result<usize, EncodeError> {
305        if self.leases.poisoned.load(Ordering::Acquire) {
306            return Err(EncodeError::Other("Logger faulted after a storage failure"));
307        }
308        self.leases[self.slot]
309            .compare_exchange(
310                self.token(IDLE),
311                self.token(BUSY),
312                Ordering::AcqRel,
313                Ordering::Acquire,
314            )
315            .map_err(|_| EncodeError::Other("Section lease is sealed or busy"))?;
316        begin_observed_encode();
317        let old_position = self.position;
318        let result = (|| {
319            let writer = RegionWriter {
320                region: &mut self.region,
321                position: old_position,
322                limit: self.header.allocated,
323            };
324            let mut encoder = EncoderImpl::new(ObservedWriter::new(writer), standard());
325            entry.encode(&mut encoder)?;
326            let end = encoder.into_writer().into_inner().position;
327            self.region.commit().map_err(encode_error)?;
328            let mut committed = self.header.clone();
329            committed.next_section = self.leases[self.slot].next();
330            committed.used = u32::try_from(end - HEADER).map_err(|_| EncodeError::UnexpectedEnd)?;
331            if let Err(error) = write_header(&mut self.region, &committed) {
332                self.leases.poisoned.store(true, Ordering::Release);
333                return Err(encode_error(error));
334            }
335            self.header = committed;
336            self.position = end;
337            Ok((end - old_position) as usize)
338        })();
339        match &result {
340            Ok(count) => {
341                debug_assert_eq!(*count, finish_observed_encode());
342            }
343            Err(_) => {
344                abort_observed_encode();
345                self.region.abort();
346            }
347        }
348        self.leases[self.slot].store(self.token(IDLE), Ordering::Release);
349        result
350    }
351    fn flush(&mut self) -> CuResult<usize> {
352        if self.leases[self.slot]
353            .compare_exchange(
354                self.token(IDLE),
355                self.token(BUSY),
356                Ordering::AcqRel,
357                Ordering::Acquire,
358            )
359            .is_ok()
360        {
361            let mut header = self.header.clone();
362            header.next_section = self.leases[self.slot].next();
363            header.is_open = false;
364            let result = write_header(&mut self.region, &header).and_then(|()| self.region.flush());
365            if result.is_ok() {
366                self.header = header;
367                self.leases[self.slot].store(self.token(SEALED), Ordering::Release);
368            } else {
369                self.leases.poisoned.store(true, Ordering::Release);
370                self.leases[self.slot].store(self.token(IDLE), Ordering::Release);
371                result?;
372            }
373        }
374        Ok(self.position as usize)
375    }
376}

378/// One allocator implements both backing adapters. Section bookkeeping is fixed-capacity.
379pub struct ByteLogger<B: ByteStorage> {
380    storage: B,
381    header: MainHeader,
382    policy: CapacityPolicy,
383    capacity: u64,
384    leases: Arc<LeaseTable>,
385    generation: usize,
386    next_run_id: u64,
387    retained_space: u64,
388    append_pending: bool,
389    faulted: bool,
390}
391impl<B: ByteStorage> ByteLogger<B> {
392    pub fn new(
393        mut storage: B,
394        alignment: u16,
395        policy: CapacityPolicy,
396        capacity: u64,
397    ) -> CuResult<Self> {
398        if alignment == 0
399            || !(alignment as u64).is_multiple_of(HEADER)
400            || capacity <= alignment as u64
401            || !capacity.is_multiple_of(HEADER)
402        {
403            return Err(CuError::from(
404                "Invalid log capacity or allocation alignment",
405            ));
406        }
407        storage.grow(capacity)?;
408        let header = MainHeader {
409            magic: MAIN_MAGIC,
410            format_version: UNIFIED_LOG_FORMAT_VERSION,
411            page_size: alignment,
412            metadata_offset: 0,
413            sections_begin: alignment as u64,
414            sections_end: capacity,
415            head_section: 0,
416            tail_section: 0,
417            clean_close: false,
418        };
419        let mut logger = Self::from_parts(storage, header, policy, capacity, false);
420        logger.publish()?;
421        Ok(logger)
422    }
423    fn from_parts(
424        storage: B,
425        header: MainHeader,
426        policy: CapacityPolicy,
427        capacity: u64,
428        append_pending: bool,
429    ) -> Self {
430        Self {
431            storage,
432            header,
433            policy,
434            capacity,
435            leases: Arc::new(LeaseTable {
436                slots: core::array::from_fn(|_| Lease {
437                    state: AtomicUsize::new(0),
438                    next: UnsafeCell::new(0),
439                    offset: UnsafeCell::new(0),
440                }),
441                poisoned: AtomicBool::new(false),
442            }),
443            generation: 0,
444            next_run_id: 1,
445            retained_space: 0,
446            append_pending,
447            faulted: false,
448        }
449    }
450    /// Read-only validation. No storage extension or header write precedes metadata comparison.
451    pub fn append(storage: B, policy: CapacityPolicy, capacity: u64) -> CuResult<Self> {
452        let header = read_main(&storage)?;
453        if !header.clean_close {
454            return Err(CuError::from("Cannot append: log was not cleanly closed"));
455        }
456        if !capacity.is_multiple_of(HEADER) || capacity <= header.sections_begin {
457            return Err(CuError::from("Invalid append capacity"));
458        }
459        let mut logger = Self::from_parts(storage, header, policy, capacity, true);
460        let mut cursor = logger.header.head_section;
461        let mut remaining = (logger.header.sections_end - logger.header.sections_begin) / HEADER;
462        while cursor != 0 {
463            if remaining == 0 {
464                return Err(CuError::from("Cyclic section links"));
465            }
466            remaining -= 1;
467            let section = logger.data_header(cursor)?;
468            if cursor + section.allocated > capacity {
469                return Err(CuError::from(
470                    "Cannot shrink storage beneath retained sections",
471                ));
472            }
473            if section.is_open {
474                return Err(CuError::from("Cannot append: retained section is open"));
475            }
476            logger.retained_space = logger
477                .retained_space
478                .checked_add(section.allocated)
479                .filter(|used| *used <= logger.header.sections_end - logger.header.sections_begin)
480                .ok_or_else(|| CuError::from("Retained sections exceed data capacity"))?;
481            logger.next_run_id = logger.next_run_id.max(
482                section
483                    .context
484                    .run_id
485                    .checked_add(1)
486                    .ok_or_else(|| CuError::from("Run ID exhausted"))?,
487            );
488            if section.next_section == 0 && cursor != logger.header.tail_section {
489                return Err(CuError::from("Invalid tail section"));
490            }
491            if cursor == logger.header.tail_section && section.next_section != 0 {
492                return Err(CuError::from("Invalid tail link"));
493            }
494            cursor = section.next_section;
495        }
496        logger.validate_metadata()?;
497        Ok(logger)
498    }
499    fn validate_metadata(&self) -> CuResult<()> {
500        if self.header.metadata_offset == 0 {
501            return Ok(());
502        }
503        let offset = self.header.metadata_offset;
504        if offset != self.header.page_size as u64 {
505            return Err(CuError::from("Invalid metadata offset"));
506        }
507        let h = read_header(&self.storage, offset)?;
508        if h.is_open
509            || h.entry_type != UnifiedLogType::ApplicationMetadata
510            || offset
511                .checked_add(h.allocated)
512                .is_none_or(|end| end > self.header.sections_begin)
513        {
514            return Err(CuError::from("Invalid application metadata section"));
515        }
516        if h.next_section != 0 {
517            let c = read_header(&self.storage, h.next_section)?;
518            if c.is_open
519                || c.entry_type != UnifiedLogType::ValueDecodeCatalog
520                || c.next_section != 0
521                || h.next_section != offset + h.allocated
522                || h.next_section.checked_add(c.allocated) != Some(self.header.sections_begin)
523            {
524                return Err(CuError::from("Invalid catalog metadata section"));
525            }
526        } else if offset + h.allocated != self.header.sections_begin {
527            return Err(CuError::from("Invalid metadata bounds"));
528        }
529        Ok(())
530    }
531    fn data_header(&self, offset: u64) -> CuResult<SectionHeader> {
532        let mut claims = Claims::new(self.leases.clone());
533        if let Some(slot) = self
534            .leases
535            .iter()
536            .position(|lease| lease.offset() == offset)
537        {
538            claims.claim(slot)?;
539        }
540        self.data_header_unlocked(offset)
541    }
542    fn data_header_unlocked(&self, offset: u64) -> CuResult<SectionHeader> {
543        if offset < self.header.sections_begin
544            || offset >= self.header.sections_end
545            || !offset.is_multiple_of(HEADER)
546        {
547            return Err(CuError::from("Section offset is outside the data region"));
548        }
549        let h = read_header(&self.storage, offset)?;
550        if offset
551            .checked_add(h.allocated)
552            .is_none_or(|end| end > self.header.sections_end)
553            || (h.next_section != 0
554                && (h.next_section < self.header.sections_begin
555                    || h.next_section >= self.header.sections_end
556                    || !h.next_section.is_multiple_of(HEADER)))
557        {
558            return Err(CuError::from("Invalid section bounds or next link"));
559        }
560        Ok(h)
561    }
562    fn publish(&mut self) -> CuResult<()> {
563        let result = (|| {
564            let mut region = self.storage.region(0, HEADER)?;
565            write_header(&mut region, &self.header)?;
566            region.flush()
567        })();
568        if result.is_err() {
569            self.faulted = true;
570        }
571        result
572    }
573    fn activate(&mut self) -> CuResult<()> {
574        if self.faulted || self.leases.poisoned.load(Ordering::Acquire) {
575            return Err(CuError::from("Logger faulted after a storage failure"));
576        }
577        if self.append_pending {
578            self.storage.grow(self.capacity)?;
579            self.header.sections_end = self.capacity;
580            self.header.clean_close = false;
581            self.publish()?;
582            self.append_pending = false;
583        }
584        Ok(())
585    }
586    fn close_idle(&mut self, offset: u64) -> CuResult<()> {
587        if let Some(slot) = self
588            .leases
589            .iter()
590            .position(|lease| lease.offset() == offset)
591        {
592            let state = self.leases[slot].load(Ordering::Acquire);
593            if state % 4 == BUSY {
594                return Err(CuError::from("Space is blocked by a write in progress"));
595            }
596            if state % 4 == IDLE {
597                self.leases[slot]
598                    .compare_exchange(
599                        state,
600                        state / 4 * 4 + SEALED,
601                        Ordering::AcqRel,
602                        Ordering::Acquire,
603                    )
604                    .map_err(|_| CuError::from("Space is blocked by a write in progress"))?;
605                let mut h = self.data_header(offset)?;
606                h.is_open = false;
607                write_header(&mut self.storage.region(offset, HEADER)?, &h)?;
608                self.storage.region(offset, h.allocated)?.flush()?;
609            }
610        }
611        if self.data_header(offset)?.is_open {
612            return Err(CuError::from("Space is blocked by an open section"));
613        }
614        Ok(())
615    }
616    fn compare<E: Encode>(&self, offset: u64, kind: UnifiedLogType, value: &E) -> CuResult<()> {
617        let h = read_header(&self.storage, offset)?;
618        if h.entry_type != kind {
619            return Err(CuError::from("Static metadata section type differs"));
620        }
621        let region = self.storage.region(offset + HEADER, h.used as u64)?;
622        let writer = CompareWriter {
623            region: &region,
624            position: 0,
625            limit: h.used as u64,
626        };
627        let mut encoder = EncoderImpl::new(writer, standard());
628        value.encode(&mut encoder).map_err(|_| CuError::from("Cannot append: static metadata does not match; appending different applications or versions to the same log is not supported."))?;
629        if encoder.into_writer().position != h.used as u64 {
630            return Err(CuError::from(
631                "Cannot append: static metadata does not match; appending different applications or versions to the same log is not supported.",
632            ));
633        }
634        Ok(())
635    }
636    fn write_static<E: Encode>(
637        &mut self,
638        offset: u64,
639        allocated: u64,
640        next: u64,
641        kind: UnifiedLogType,
642        value: &E,
643    ) -> CuResult<()> {
644        let mut region = self.storage.region(offset, allocated)?;
645        let writer = RegionWriter {
646            region: &mut region,
647            position: HEADER,
648            limit: allocated,
649        };
650        let mut encoder = EncoderImpl::new(writer, standard());
651        value.encode(&mut encoder).map_err(|e| {
652            CuError::from("Cannot encode static metadata").add_cause(&e.to_string())
653        })?;
654        let used = encoder.into_writer().position - HEADER;
655        region.flush()?;
656        write_header(
657            &mut region,
658            &SectionHeader {
659                entry_type: kind,
660                allocated,
661                next_section: next,
662                used: u32::try_from(used)
663                    .map_err(|_| CuError::from("Metadata exceeds section length limit"))?,
664                is_open: false,
665                ..SectionHeader::default()
666            },
667        )
668    }
669    pub fn close(&mut self) -> CuResult<()> {
670        if self.append_pending {
671            return Ok(());
672        }
673        if self.faulted || self.leases.poisoned.load(Ordering::Acquire) {
674            return Err(CuError::from("Cannot cleanly close a faulted logger"));
675        }
676        self.faulted = true;
677        let mut cursor = self.header.head_section;
678        let mut remaining = (self.header.sections_end - self.header.sections_begin) / HEADER;
679        while cursor != 0 {
680            if remaining == 0 {
681                return Err(CuError::from("Cyclic section links prevent clean close"));
682            }
683            remaining -= 1;
684            self.close_idle(cursor)?;
685            cursor = self.data_header(cursor)?.next_section;
686        }
687        self.header.clean_close = true;
688        self.publish()?;
689        self.faulted = false;
690        Ok(())
691    }
692}
693fn aligned(size: u64) -> CuResult<u64> {
694    size.checked_add(HEADER - 1)
695        .map(|s| s / HEADER * HEADER)
696        .ok_or_else(|| CuError::from("Section size overflow"))
697}
698impl<B: ByteStorage> UnifiedLogWrite<ByteSection<B::Region>> for ByteLogger<B> {
699    fn seal_metadata<C: Encode>(
700        &mut self,
701        metadata: &ApplicationMetadata,
702        catalog: Option<&C>,
703    ) -> CuResult<()> {
704        #[derive(Encode)]
705        struct MetadataBody<'a> {
706            app_type: &'a str,
707            app_name: &'a str,
708            app_version: &'a str,
709            git_commit: Option<&'a str>,
710            git_dirty: Option<bool>,
711            subsystem_id: Option<&'a str>,
712            subsystem_code: u16,
713            effective_config_ron: &'a str,
714            missions: &'a [alloc::string::String],
715            catalog_offset: u64,
716        }
717        let mut metadata = MetadataBody {
718            app_type: &metadata.app_type,
719            app_name: &metadata.app_name,
720            app_version: &metadata.app_version,
721            git_commit: metadata.git_commit.as_deref(),
722            git_dirty: metadata.git_dirty,
723            subsystem_id: metadata.subsystem_id.as_deref(),
724            subsystem_code: metadata.subsystem_code,
725            effective_config_ron: &metadata.effective_config_ron,
726            missions: &metadata.missions,
727            catalog_offset: 0,
728        };
729        let base = self.header.page_size as u64;
730        // The offset changes bincode's varint width; sizing converges monotonically.
731        let mut app_size = 0;
732        for _ in 0..8 {
733            let size = aligned(HEADER + encoded_size(&metadata)?)?;
734            let offset = if catalog.is_some() { base + size } else { 0 };
735            app_size = size;
736            if metadata.catalog_offset == offset {
737                break;
738            }
739            metadata.catalog_offset = offset;
740        }
741        let catalog_size = catalog
742            .map(|c| encoded_size(c).and_then(|s| aligned(HEADER + s)))
743            .transpose()?
744            .unwrap_or(0);
745        let end = base
746            .checked_add(app_size)
747            .and_then(|s| s.checked_add(catalog_size))
748            .ok_or_else(|| CuError::from("Metadata size overflow"))?;
749        if self.header.metadata_offset != 0 {
750            self.compare(base, UnifiedLogType::ApplicationMetadata, &metadata)?;
751            if let Some(catalog) = catalog {
752                self.compare(
753                    metadata.catalog_offset,
754                    UnifiedLogType::ValueDecodeCatalog,
755                    catalog,
756                )?;
757            }
758            self.activate()?;
759            return Ok(());
760        }
761        if self.header.head_section != 0 || self.header.sections_begin != base {
762            return Err(CuError::from("Static metadata region is already sealed"));
763        }
764        if end + HEADER > self.header.sections_end && self.policy != CapacityPolicy::Grow {
765            return Err(CuError::from("Static metadata exceeds log capacity"));
766        }
767        self.storage.grow(end + HEADER)?;
768        self.activate()?;
769        self.faulted = true;
770        self.write_static(
771            base,
772            app_size,
773            metadata.catalog_offset,
774            UnifiedLogType::ApplicationMetadata,
775            &metadata,
776        )?;
777        if let Some(catalog) = catalog {
778            self.write_static(
779                metadata.catalog_offset,
780                catalog_size,
781                0,
782                UnifiedLogType::ValueDecodeCatalog,
783                catalog,
784            )?;
785        }
786        self.header.metadata_offset = base;
787        self.header.sections_begin = end;
788        if end + HEADER > self.header.sections_end {
789            self.header.sections_end = self.storage.len();
790        }
791        self.publish()?;
792        self.faulted = false;
793        Ok(())
794    }
795    fn construction_context(
796        &mut self,
797        instance_id: u32,
798        mission_index: u32,
799    ) -> CuResult<SectionContext> {
800        if self.header.metadata_offset == 0 {
801            return Err(CuError::from("Static metadata must precede construction"));
802        }
803        let run_id = self.next_run_id;
804        self.next_run_id = run_id
805            .checked_add(1)
806            .ok_or_else(|| CuError::from("Run ID exhausted"))?;
807        Ok(SectionContext {
808            run_id,
809            instance_id,
810            mission_index,
811        })
812    }
813    fn add_section(
814        &mut self,
815        kind: UnifiedLogType,
816        size: usize,
817    ) -> CuResult<SectionHandle<ByteSection<B::Region>>> {
818        self.add_section_with_context(kind, size, SectionContext::default())
819    }
820    fn add_section_with_context(
821        &mut self,
822        kind: UnifiedLogType,
823        size: usize,
824        context: SectionContext,
825    ) -> CuResult<SectionHandle<ByteSection<B::Region>>> {
826        if self.append_pending && self.header.metadata_offset != 0 {
827            return Err(CuError::from(
828                "Compare static metadata before appending data",
829            ));
830        }
831        let size = aligned(size as u64)?;
832        if size <= HEADER || size - HEADER > u32::MAX as u64 {
833            return Err(CuError::from("Invalid section allocation size"));
834        }
835        let capacity = self.header.sections_end - self.header.sections_begin;
836        if self.policy != CapacityPolicy::Grow && size > capacity {
837            return Err(CuError::from("Section exceeds log capacity"));
838        }
839        let tail = self.header.tail_section;
840        let mut position = if tail == 0 {
841            self.header.sections_begin
842        } else {
843            tail + self.data_header(tail)?.allocated
844        };
845        if self.policy == CapacityPolicy::OverwriteOldest
846            && position
847                .checked_add(size)
848                .is_none_or(|end| end > self.header.sections_end)
849        {
850            position = self.header.sections_begin;
851        }
852        let overlaps = |offset: u64, header: &SectionHeader| {
853            position < offset + header.allocated && offset < position + size
854        };
855        let mut last_overlap = 0;
856        let mut cursor = self.header.head_section;
857        let mut remaining = capacity / HEADER;
858        while cursor != 0 {
859            if remaining == 0 {
860                return Err(CuError::from("Cyclic section links"));
861            }
862            remaining -= 1;
863            let h = self.data_header(cursor)?;
864            if overlaps(cursor, &h) {
865                last_overlap = cursor;
866            }
867            cursor = h.next_section;
868        }
869        if self.policy == CapacityPolicy::Grow
870            && (last_overlap != 0 || position + size > self.header.sections_end)
871        {
872            position = self.header.sections_end;
873            last_overlap = 0;
874        }
875        if self.policy == CapacityPolicy::StopWhenFull
876            && (last_overlap != 0 || position + size > self.header.sections_end)
877        {
878            return Err(CuError::from("Log is full"));
879        }
880        // Claim every affected encoder before any write. A failed preflight restores all leases.
881        let mut claims = Claims::new(self.leases.clone());
882        if let Some(slot) = self.leases.iter().position(|lease| lease.offset() == tail) {
883            claims.claim(slot)?;
884        }
885        let mut cursor = self.header.head_section;
886        if last_overlap != 0 {
887            loop {
888                if let Some(slot) = self
889                    .leases
890                    .iter()
891                    .position(|lease| lease.offset() == cursor)
892                {
893                    claims.claim(slot)?;
894                } else if self.data_header_unlocked(cursor)?.is_open {
895                    return Err(CuError::from("Space is blocked by an open section"));
896                }
897                if cursor == last_overlap {
898                    break;
899                }
900                cursor = self.data_header_unlocked(cursor)?.next_section;
901            }
902        }
903        let free_slot = self
904            .leases
905            .iter()
906            .position(|s| matches!(s.load(Ordering::Acquire) % 4, 0 | SEALED));
907        if free_slot.is_none() && last_overlap == 0 {
908            return Err(CuError::from(
909                "Too many concurrent section streams (maximum 64)",
910            ));
911        }
912        // Validate the mapping geometry before reclamation or header changes.
913        if position + size <= self.storage.len() {
914            let _ = self.storage.region(position, size)?;
915        }
916        self.activate()?;
917        self.faulted = true;
918        if position + size > self.header.sections_end {
919            self.storage.grow(position + size)?;
920            self.header.sections_end = self.storage.len();
921        }
922        if last_overlap != 0 {
923            let mut cursor = self.header.head_section;
924            loop {
925                let mut h = self.data_header_unlocked(cursor)?;
926                h.is_open = false;
927                write_header(&mut self.storage.region(cursor, HEADER)?, &h)?;
928                self.storage.region(cursor, h.allocated)?.flush()?;
929                if let Some(slot) = self
930                    .leases
931                    .iter()
932                    .position(|lease| lease.offset() == cursor)
933                {
934                    claims.seal(slot);
935                }
936                self.retained_space -= h.allocated;
937                let next = h.next_section;
938                if cursor == last_overlap {
939                    self.header.head_section = next;
940                    if next == 0 {
941                        self.header.tail_section = 0;
942                    }
943                    break;
944                }
945                cursor = next;
946            }
947            self.publish()?;
948        }
949        let slot = self
950            .leases
951            .iter()
952            .position(|s| {
953                s.load(Ordering::Acquire) % 4 != IDLE && s.load(Ordering::Acquire) % 4 != BUSY
954            })
955            .ok_or_else(|| CuError::from("Too many concurrent section writes"))?;
956        self.generation = self
957            .generation
958            .checked_add(1)
959            .filter(|g| *g < usize::MAX / 4)
960            .ok_or_else(|| CuError::from("Section generation exhausted"))?;
961        let header = SectionHeader {
962            entry_type: kind,
963            allocated: size,
964            context,
965            ..SectionHeader::default()
966        };
967        let mut region = self.storage.region(position, size)?;
968        write_header(&mut region, &header)?;
969        if self.header.tail_section != 0 {
970            let mut previous = self.data_header_unlocked(self.header.tail_section)?;
971            previous.next_section = position;
972            if let Some(slot) = self
973                .leases
974                .iter()
975                .position(|lease| lease.offset() == self.header.tail_section)
976            {
977                // This lease was claimed before allocation, so its link can be updated.
978                if self.leases[slot].load(Ordering::Acquire) % 4 == BUSY {
979                    self.leases[slot].set_next(position);
980                }
981            }
982            write_header(
983                &mut self.storage.region(self.header.tail_section, HEADER)?,
984                &previous,
985            )?;
986        }
987        if self.header.head_section == 0 {
988            self.header.head_section = position;
989        }
990        self.header.tail_section = position;
991        self.header.clean_close = false;
992        self.publish()?;
993        self.retained_space += size;
994        for lease in self.leases.iter() {
995            if lease.offset() == position {
996                lease.set_offset(0);
997            }
998        }
999        self.leases[slot].set_offset(position);
1000        self.leases[slot].set_next(0);
1001        self.leases[slot].store(self.generation * 4 + IDLE, Ordering::Release);
1002        let storage = ByteSection {
1003            region,
1004            position: HEADER,
1005            header: header.clone(),
1006            leases: self.leases.clone(),
1007            slot,
1008            generation: self.generation,
1009        };
1010        let handle = SectionHandle::create(header, storage)?;
1011        self.faulted = false;
1012        Ok(handle)
1013    }
1014    fn flush_section(&mut self, section: &mut SectionHandle<ByteSection<B::Region>>) {
1015        let _ = self.try_flush_section(section);
1016    }
1017    fn try_flush_section(
1018        &mut self,
1019        section: &mut SectionHandle<ByteSection<B::Region>>,
1020    ) -> CuResult<()> {
1021        if self.faulted || self.leases.poisoned.load(Ordering::Acquire) {
1022            return Err(CuError::from("Logger faulted after a storage failure"));
1023        }
1024        if let Err(error) = section.storage.flush() {
1025            self.faulted = true;
1026            return Err(error);
1027        }
1028        section.mark_closed();
1029        Ok(())
1030    }
1031    fn status(&self) -> UnifiedLogStatus {
1032        UnifiedLogStatus {
1033            total_used_space: usize::try_from(self.header.sections_begin + self.retained_space)
1034                .unwrap_or(usize::MAX),
1035            total_allocated_space: usize::try_from(self.header.sections_end).unwrap_or(usize::MAX),
1036        }
1037    }
