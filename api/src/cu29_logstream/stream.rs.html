1//! Transport-independent, packet-oriented stream resource contracts.
2
3use core::fmt::Debug;
4
5/// Failure to submit one complete logstream packet.
6#[derive(Clone, Copy, Debug, PartialEq, Eq)]
7pub enum CuStreamTxError {
8    /// The bounded transport cannot accept this packet immediately.
9    WouldBlock,
10    /// The transport has failed and cannot accept the packet.
11    Failed(&'static str),
12}
13
14/// Failure to receive one complete logstream packet.
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub enum CuStreamRxError {
17    /// The next packet does not fit in the caller-provided buffer.
18    BufferTooSmall { needed: usize },
19    /// The transport has failed and cannot receive packets.
20    Failed(&'static str),
21}
22
23/// One-way, fire-and-forget transmission of complete logstream packets.
24///
25/// Implementations must return immediately without waiting, locking, allocating,
26/// retrying, acknowledging, or accepting a partial packet. `Ok(())` means only
27/// that the resource accepted this packet once; it does not guarantee delivery.
28pub trait CuStreamTx: Debug + Send + Sync {
29    fn try_send(&mut self, packet: &[u8]) -> core::result::Result<(), CuStreamTxError>;
30}
31
32impl<T: CuStreamTx + ?Sized> CuStreamTx for alloc::boxed::Box<T> {
33    #[inline]
34    fn try_send(&mut self, packet: &[u8]) -> core::result::Result<(), CuStreamTxError> {
35        (**self).try_send(packet)
36    }
37}
38
39/// Nonblocking reception of complete logstream packets.
40///
41/// `Ok(None)` means no complete packet is available. `Ok(Some(len))` places one
42/// complete packet in `packet[..len]`. This is the receive half of the one-way
43/// data plane; it does not acknowledge traffic or imply a reverse channel.
44pub trait CuStreamRx: Debug + Send + Sync {
45    fn try_recv(
46        &mut self,
47        packet: &mut [u8],
48    ) -> core::result::Result<Option<usize>, CuStreamRxError>;
49}
50
51impl<T: CuStreamRx + ?Sized> CuStreamRx for alloc::boxed::Box<T> {
52    #[inline]
53    fn try_recv(
54        &mut self,
55        packet: &mut [u8],
56    ) -> core::result::Result<Option<usize>, CuStreamRxError> {
57        (**self).try_recv(packet)
58    }
59}
