14/// Sender policy for one finite-object lane.
15#[derive(Clone, Copy, Debug, PartialEq, Eq)]
16pub struct FiniteObjectSenderConfig {
17    pub identity: StreamIdentity,
18    pub first_packet_sequence: u64,
19    pub lane: Lane,
20    /// RaptorQ symbol bytes, excluding the Copper wire header.
21    pub symbol_size: u16,
22    /// Hard limit checked before constructing the RaptorQ encoder.
23    pub max_object_bytes: u64,
24    /// Repair symbols emitted for every RaptorQ source block.

85    /// Encodes one already-framed semantic record and appends its datagrams.
86    pub fn push_record(&mut self, record: &[u8], output: &mut Vec<Vec<u8>>) -> Result<usize> {
87        self.push_record_with(record, |datagram| {
88            output.push(datagram.to_vec());
89            Ok(())
90        })
91    }
92
93    /// Encodes one record and emits each datagram immediately.
94    pub fn push_record_with(
95        &mut self,
96        record: &[u8],
97        mut emit: impl FnMut(&[u8]) -> Result<()>,
98    ) -> Result<usize> {
99        let decoded = decode_record(record)?;
100        let record_len = u64::try_from(record.len())
101            .map_err(|_| Error::InvalidConfig("finite object length exceeds u64"))?;
102        if record_len > self.config.max_object_bytes {
103            return Err(Error::ObjectTooLarge {
104                actual: record_len,
105                maximum: self.config.max_object_bytes,
106            });
107        }
108
109        let encoder = Encoder::with_defaults(record, self.config.symbol_size);
110        let oti = encoder.get_config().serialize();
111        let datagram_len = PACKET_HEADER_LEN
112            .checked_add(self.config.symbol_size as usize)
113            .ok_or(Error::InvalidConfig(
114                "finite-object datagram length overflow",
115            ))?;
116        let mut datagram = vec![0; datagram_len];
117        let mut emitted = 0_usize;
118
119        for block in encoder.get_block_encoders() {
120            for packet in block.source_packets() {
121                self.emit_packet(
122                    decoded,
123                    FecSymbolKind::Source,
124                    oti,
125                    packet,
126                    &mut datagram,
127                    &mut emit,
128                )?;
129                emitted = emitted.saturating_add(1);
130            }
131            for packet in block.repair_packets(0, self.config.repair_symbols_per_block) {
132                self.emit_packet(
133                    decoded,
134                    FecSymbolKind::Repair,
135                    oti,
136                    packet,
137                    &mut datagram,
138                    &mut emit,
139                )?;
140                emitted = emitted.saturating_add(1);
141            }
142        }
143        Ok(emitted)
144    }
145
146    fn emit_packet(
147        &mut self,
148        record: crate::DecodedRecord<'_>,
149        symbol_kind: FecSymbolKind,
150        oti: [u8; 12],
151        packet: EncodingPacket,
152        datagram: &mut [u8],
153        emit: &mut impl FnMut(&[u8]) -> Result<()>,
154    ) -> Result<()> {
155        let (payload_id, payload) = packet.split();
156        let header = WireHeader {
157            lane: self.config.lane,
158            record_kind: record.kind,
159            fec_scheme: FecScheme::RaptorQ,
160            symbol_kind,
161            session_id: self.config.identity.session_id,
162            sender_id: self.config.identity.sender_id,
163            packet_sequence: self.packet_sequence,
164            object_id: record.object_id,
165            fec_metadata: oti,
166            fragment_count: u32::from_be_bytes(payload_id.serialize()),
167        };
168        let encoded = encode_packet_into(header, &payload, datagram)?;
169        emit(&datagram[..encoded])?;
170        self.packet_sequence = self.packet_sequence.wrapping_add(1);
171        Ok(())
172    }
173}

175/// Bounded assembler for independently reorderable finite RaptorQ objects.
176pub struct FiniteObjectDecoder {
177    identity: StreamIdentity,
178    lane: Lane,
179    limits: FiniteObjectLimits,
180    objects: Vec<ObjectDecoder>,
181    ready: Vec<ReadyObject>,
182    completed: Vec<CompletedObject>,
183    stats: FiniteObjectRecoveryStats,
184}
185
186impl FiniteObjectDecoder {
187    pub fn new(identity: StreamIdentity, lane: Lane, limits: FiniteObjectLimits) -> Result<Self> {
188        validate_limits(limits)?;
189        Ok(Self {
190            identity,
191            lane,
192            limits,
193            objects: Vec::with_capacity(limits.max_concurrent_objects),
194            ready: Vec::with_capacity(limits.max_concurrent_objects),
195            completed: Vec::with_capacity(limits.max_concurrent_objects),
196            stats: FiniteObjectRecoveryStats::default(),
197        })
198    }
199
200    pub const fn stats(&self) -> FiniteObjectRecoveryStats {
201        self.stats
202    }
203
204    pub fn occupancy(&self) -> usize {
205        self.objects.len().saturating_add(self.ready.len())
206    }
207
208    /// Accepts one datagram and emits a digest-verified record when its object completes.
209    pub fn receive_datagram<E>(
210        &mut self,
211        datagram: &[u8],
212        mut emit: impl FnMut(&RecoveredRecord) -> core::result::Result<(), E>,
213    ) -> core::result::Result<(), ReceiveError<E>> {
214        self.drain_records(&mut emit)?;
215        self.stats.datagrams_seen = self.stats.datagrams_seen.saturating_add(1);
216        let packet = match WirePacket::decode(datagram) {
217            Ok(packet) => packet,
218            Err(_) => {
219                self.stats.invalid_datagrams = self.stats.invalid_datagrams.saturating_add(1);
220                return Ok(());
221            }
222        };
223        if packet.header.session_id != self.identity.session_id
224            || packet.header.sender_id != self.identity.sender_id
225            || packet.header.lane != self.lane
226            || packet.header.fec_scheme != FecScheme::RaptorQ
227            || packet.header.record_kind == RecordKind::CopperList
228        {
229            self.stats.inconsistent_datagrams = self.stats.inconsistent_datagrams.saturating_add(1);
230            return Ok(());
231        }
232
233        let oti = match validate_oti(packet.header.fec_metadata, self.limits) {
234            Ok(oti) => oti,
235            Err(_) => {
236                self.stats.invalid_datagrams = self.stats.invalid_datagrams.saturating_add(1);
237                return Ok(());
238            }
239        };
240        if packet.payload.len() != oti.symbol_size() as usize {
241            self.stats.invalid_datagrams = self.stats.invalid_datagrams.saturating_add(1);
242            return Ok(());
243        }
244
245        let payload_id = PayloadId::deserialize(&packet.header.fragment_count.to_be_bytes());
246        if payload_id.source_block_number() >= oti.source_blocks() {
247            self.stats.invalid_datagrams = self.stats.invalid_datagrams.saturating_add(1);
248            return Ok(());
249        }
250        if let Some(object) = self.ready.iter().find(|object| {
251            object.kind == packet.header.record_kind && object.object_id == packet.header.object_id
252        }) {
253            if object.oti != oti {
254                self.stats.inconsistent_datagrams =
255                    self.stats.inconsistent_datagrams.saturating_add(1);
256            } else {
257                self.stats.duplicate_symbols = self.stats.duplicate_symbols.saturating_add(1);
258            }
259            return Ok(());
260        }
261        if let Some(object) = self.completed.iter().find(|object| {
262            object.kind == packet.header.record_kind && object.object_id == packet.header.object_id
263        }) {
264            if object.oti != oti {
265                self.stats.inconsistent_datagrams =
266                    self.stats.inconsistent_datagrams.saturating_add(1);
267            } else {
268                self.stats.duplicate_symbols = self.stats.duplicate_symbols.saturating_add(1);
269            }
270            return Ok(());
271        }
272        let object_index = match self.objects.iter().position(|object| {
273            object.kind == packet.header.record_kind && object.object_id == packet.header.object_id
274        }) {
275            Some(index) => {
276                let object = &self.objects[index];
277                if object.oti != oti {
278                    self.stats.inconsistent_datagrams =
279                        self.stats.inconsistent_datagrams.saturating_add(1);
280                    return Ok(());
281                }
282                index
283            }
284            None => {
285                let actual = self
286                    .objects
287                    .len()
288                    .saturating_add(self.ready.len())
289                    .saturating_add(1);
290                if actual > self.limits.max_concurrent_objects {
291                    return Err(Error::TooManyRecords {
292                        actual,
293                        maximum: self.limits.max_concurrent_objects,
294                    }
295                    .into());
296                }
297                self.objects.push(ObjectDecoder {
298                    kind: packet.header.record_kind,
299                    object_id: packet.header.object_id,
300                    oti,
301                    decoder: Decoder::new(oti),
302                });
303                self.stats.peak_concurrent_objects =
304                    self.stats.peak_concurrent_objects.max(self.objects.len());
305                self.objects.len() - 1
306            }
307        };
308
309        match packet.header.symbol_kind {
310            FecSymbolKind::Source => {
311                self.stats.source_symbols_received =
312                    self.stats.source_symbols_received.saturating_add(1);
313            }
314            FecSymbolKind::Repair => {
315                self.stats.repair_symbols_received =
316                    self.stats.repair_symbols_received.saturating_add(1);
317            }
318        }
319        self.stats.valid_datagrams = self.stats.valid_datagrams.saturating_add(1);
320        let completed = self.objects[object_index]
321            .decoder
322            .decode(EncodingPacket::new(payload_id, packet.payload));
323        let Some(bytes) = completed else {
324            return Ok(());
325        };
326
327        let object = self.objects.swap_remove(object_index);
328        let decoded = decode_record(&bytes)?;
329        if decoded.kind != object.kind || decoded.object_id != object.object_id {
330            return Err(Error::InconsistentObject.into());
331        }
332        self.ready.push(ReadyObject {
333            kind: object.kind,
334            object_id: object.object_id,
335            oti: object.oti,
336            record: RecoveredRecord::from_bytes(bytes),
337        });
338        self.stats.objects_recovered = self.stats.objects_recovered.saturating_add(1);
339        self.drain_records(&mut emit)
340    }
341
342    /// Retries delivery of records retained after a consumer failure.
343    pub fn drain_records<E>(
344        &mut self,
345        mut emit: impl FnMut(&RecoveredRecord) -> core::result::Result<(), E>,
346    ) -> core::result::Result<(), ReceiveError<E>> {
347        while !self.ready.is_empty() {
348            emit(&self.ready[0].record).map_err(ReceiveError::Consumer)?;
349            let delivered = self.ready.remove(0);
350            if self.completed.len() == self.limits.max_concurrent_objects {
351                self.completed.remove(0);
352            }
353            self.completed.push(CompletedObject {
354                kind: delivered.kind,
355                object_id: delivered.object_id,
356                oti: delivered.oti,
357            });
358        }
359        Ok(())
360    }
361}

383fn validate_sender_config(config: FiniteObjectSenderConfig) -> Result<()> {
384    if config.symbol_size == 0 {
385        return Err(Error::InvalidConfig("RaptorQ symbol size must be nonzero"));
386    }
387    if config.max_object_bytes == 0 || config.max_object_bytes > RFC6330_MAX_TRANSFER_LENGTH {
388        return Err(Error::InvalidConfig("invalid RaptorQ maximum object size"));
389    }
390    Ok(())
391}
392
393fn validate_limits(limits: FiniteObjectLimits) -> Result<()> {
394    if limits.max_object_bytes == 0 || limits.max_object_bytes > RFC6330_MAX_TRANSFER_LENGTH {
395        return Err(Error::InvalidConfig("invalid RaptorQ maximum object size"));
396    }
397    if limits.max_symbol_size == 0 {
398        return Err(Error::InvalidConfig(
399            "RaptorQ maximum symbol size must be nonzero",
400        ));
401    }
402    if limits.max_concurrent_objects == 0 {
403        return Err(Error::InvalidConfig(
404            "finite-object receiver capacity must be nonzero",
405        ));
406    }
407    Ok(())
408}
409
410fn validate_oti(
411    encoded: [u8; 12],
412    limits: FiniteObjectLimits,
413) -> Result<ObjectTransmissionInformation> {
414    if encoded[5] != 0 {
415        return Err(Error::InvalidFecMetadata(
416            "RaptorQ OTI reserved byte is nonzero",
417        ));
418    }
419    let oti = ObjectTransmissionInformation::deserialize(&encoded);
420    let transfer_length = oti.transfer_length();
421    let symbol_size = oti.symbol_size();
422    let source_blocks = oti.source_blocks();
423    let sub_blocks = oti.sub_blocks();
424    let alignment = oti.symbol_alignment();
425    if transfer_length == 0 || transfer_length > limits.max_object_bytes {
426        return Err(Error::ObjectTooLarge {
427            actual: transfer_length,
428            maximum: limits.max_object_bytes,
429        });
430    }
431    if symbol_size == 0 || symbol_size > limits.max_symbol_size {
432        return Err(Error::InvalidFecMetadata(
433            "RaptorQ symbol size exceeds receiver limit",
434        ));
435    }
436    if source_blocks == 0 || sub_blocks == 0 || alignment == 0 {
437        return Err(Error::InvalidFecMetadata(
438            "RaptorQ OTI contains a zero parameter",
439        ));
440    }
441    if !symbol_size.is_multiple_of(alignment as u16) {
442        return Err(Error::InvalidFecMetadata(
443            "RaptorQ symbol alignment is invalid",
444        ));
445    }
446    let total_source_symbols = transfer_length.div_ceil(symbol_size as u64);
447    if source_blocks as u64 > total_source_symbols {
448        return Err(Error::InvalidFecMetadata(
449            "RaptorQ has more source blocks than source symbols",
450        ));
451    }
452    if sub_blocks as u32 > u32::from(symbol_size / alignment as u16) {
453        return Err(Error::InvalidFecMetadata(
454            "RaptorQ sub-block count exceeds symbol geometry",
455        ));
456    }
457    let largest_block = total_source_symbols.div_ceil(source_blocks as u64);
458    if largest_block > RFC6330_MAX_SOURCE_SYMBOLS_PER_BLOCK {
459        return Err(Error::InvalidFecMetadata(
460            "RaptorQ source block is too large",
461        ));
462    }
463    Ok(oti)
