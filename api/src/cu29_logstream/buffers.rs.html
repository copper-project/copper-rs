1use crate::pacing::STRUCTURED_RECORD_BYTES;
2use cu29_traits::{CuError, CuResult};
3/// Preallocated native records shared by all destinations on one telemetry worker.
4/// A full pool drops the incoming record of that kind. Only the keyframe count may be zero.
5#[derive(Clone, Copy, Debug, PartialEq, Eq)]
6pub struct TelemetryBufferConfig {
7    /// Maximum encoded payload per CopperList capture.
8    pub max_capture_bytes: usize,
9    /// Maximum encoded payload per keyframe. Structured logs use the smaller of this and 4096 bytes.
10    pub max_keyframe_bytes: usize,
11    /// Number of capture buffers.
12    pub capture_count: u16,
13    /// Number of keyframe buffers. Zero disables native telemetry keyframes.
14    pub keyframe_count: u16,
15    /// Number of structured-log buffers.
16    pub structured_log_count: u16,
17}
18
19impl TelemetryBufferConfig {
20    /// Reserve four captures, two keyframes and four structured logs.
21    pub const fn new(max_capture_bytes: usize, max_keyframe_bytes: usize) -> Self {
22        Self {
23            max_capture_bytes,
24            max_keyframe_bytes,
25            capture_count: 4,
26            keyframe_count: 2,
27            structured_log_count: 4,
28        }
29    }
30
31    pub(crate) fn structured_bytes(self) -> usize {
32        self.max_keyframe_bytes.min(STRUCTURED_RECORD_BYTES)
33    }
34
35    pub(crate) fn payload_bytes(self) -> CuResult<usize> {
36        if self.max_capture_bytes == 0 || self.max_keyframe_bytes == 0 {
37            return Err("telemetry record limits must be positive".into());
38        }
39        if self.capture_count == 0 || self.structured_log_count == 0 {
40            return Err("telemetry capture and structured-log counts must be positive".into());
41        }
42        let pools = [
43            (self.max_capture_bytes, self.capture_count),
44            (self.max_keyframe_bytes, self.keyframe_count),
45            (self.structured_bytes(), self.structured_log_count),
46        ];
47        pools
48            .into_iter()
49            .try_fold(0usize, |total, (bytes, count)| {
50                total.checked_add(bytes.checked_mul(usize::from(count))?)
51            })
52            .filter(|&bytes| bytes <= isize::MAX as usize)
53            .ok_or_else(|| CuError::from("telemetry buffer size overflow"))
54    }
55}
