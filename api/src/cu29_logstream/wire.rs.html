1//! Packed transport framing. No version or schema is sent in packet headers;
2//! sender and receiver must be built for the matching application.
3
4use crate::{Error, RecordKind, Result};
5use alloc::vec::Vec;
6
7const PACKET_MAGIC: [u8; 4] = *b"CULS";
8/// Maximum packet header length, used to bound storage and the shared FEC symbol size.
9/// RaptorQ uses 51 bytes; RLC uses 32 for source packets and 36 for repair packets.
10pub const PACKET_HEADER_LEN: usize = 51;
11const COMMON_HEADER_LEN: usize = 28;
12const RLC_SOURCE_HEADER_LEN: usize = 32;
13const RLC_REPAIR_HEADER_LEN: usize = 36;
14
15const fn packet_header_len(scheme: FecScheme, kind: FecSymbolKind) -> usize {
16    match (scheme, kind) {
17        (FecScheme::RaptorQ, _) => PACKET_HEADER_LEN,
18        (_, FecSymbolKind::Source) => RLC_SOURCE_HEADER_LEN,
19        (_, FecSymbolKind::Repair) => RLC_REPAIR_HEADER_LEN,
20    }
21}
22
23/// Independently scheduled transport lanes.
24#[derive(Clone, Copy, Debug, PartialEq, Eq)]
25#[repr(u8)]
26pub enum Lane {
27    Control = 0,
28    ReplayCritical = 1,
29    LargeObject = 2,
30    StructuredLog = 3,
31    Visualization = 4,
32}
33
34impl TryFrom<u8> for Lane {
35    type Error = Error;
36
37    fn try_from(value: u8) -> Result<Self> {
38        match value {
39            0 => Ok(Self::Control),
40            1 => Ok(Self::ReplayCritical),
41            2 => Ok(Self::LargeObject),
42            3 => Ok(Self::StructuredLog),
43            4 => Ok(Self::Visualization),
44            _ => Err(Error::UnknownLane(value)),
45        }
46    }
47}

49#[derive(Clone, Copy, Debug, PartialEq, Eq)]
50#[repr(u8)]
51pub enum FecScheme {
52    RlcGf2 = 0,
53    RlcGf256 = 1,
54    RaptorQ = 2,
55}
56
57impl TryFrom<u8> for FecScheme {
58    type Error = Error;
59
60    fn try_from(value: u8) -> Result<Self> {
61        match value {
62            0 => Ok(Self::RlcGf2),
63            1 => Ok(Self::RlcGf256),
64            2 => Ok(Self::RaptorQ),
65            _ => Err(Error::UnknownFecScheme(value)),
66        }
67    }
68}

71#[repr(u8)]
72pub enum FecSymbolKind {
73    Source = 0,
74    Repair = 1,
75}

77impl TryFrom<u8> for FecSymbolKind {

89/// Fixed-endian metadata repeated on every independently reorderable symbol.
90#[derive(Clone, Copy, Debug, PartialEq, Eq)]
91pub struct WireHeader {
92    pub lane: Lane,
93    pub record_kind: RecordKind,
94    pub fec_scheme: FecScheme,
95    pub symbol_kind: FecSymbolKind,
96    pub session_id: [u8; 16],
97    pub sender_id: u32,
98    /// RaptorQ object identity. Not transmitted for RLC; decoded as zero.
99    pub object_id: u64,
100    /// RaptorQ OTI (12 bytes), RLC source ID (first 4), or RLC repair ID (first 8).
101    /// RaptorQ's reserved byte at index 5 must be zero; it is omitted on the wire
102    /// and restored locally when decoding.
103    /// Unused RLC bytes are not transmitted and decode as zero.
104    pub fec_metadata: [u8; 12],
105    /// RaptorQ payload ID. Not transmitted for RLC; decoded as zero.
106    pub fragment_count: u32,
107}

109/// Owned packet for callers that retain or edit payloads.
110/// Receiver parsing uses [`WirePacketRef`] to borrow the datagram instead.
111#[derive(Clone, Debug, PartialEq, Eq)]
112pub struct WirePacket {
113    pub header: WireHeader,
114    pub payload: Vec<u8>,
115}
116
117impl WirePacket {
118    pub fn encode(&self) -> Result<Vec<u8>> {
119        let mut bytes = alloc::vec![0; packet_header_len(self.header.fec_scheme, self.header.symbol_kind) + self.payload.len()];
120        let encoded = encode_packet_into(self.header, &self.payload, &mut bytes)?;
121        debug_assert_eq!(encoded, bytes.len());
122        Ok(bytes)
123    }
124
125    pub fn decode(bytes: &[u8]) -> Result<Self> {
126        let packet = WirePacketRef::decode(bytes)?;
127        Ok(Self {
128            header: packet.header,
129            payload: packet.payload.to_vec(),
130        })
131    }
132}

134/// Structurally validated packet view borrowing the transport's receive buffer.
135#[derive(Clone, Copy, Debug, PartialEq, Eq)]
136pub struct WirePacketRef<'a> {
137    pub header: WireHeader,
138    pub payload: &'a [u8],
139}
140
141impl<'a> WirePacketRef<'a> {
142    /// Decodes exactly one complete packet supplied by the carrier.
143    /// The carrier must verify packet integrity before calling this parser.
144    /// This validates header structure only; FEC cannot correct corrupted symbols.
145    /// The payload occupies the remaining packet extent after the header.
146    pub fn decode(bytes: &'a [u8]) -> Result<Self> {
147        if bytes.len() < COMMON_HEADER_LEN {
148            return Err(Error::TruncatedPacket);
149        }
150        if bytes[..4] != PACKET_MAGIC {
151            return Err(Error::InvalidMagic);
152        }
153        let fec_scheme = FecScheme::try_from(bytes[6])?;
154        let symbol_kind = FecSymbolKind::try_from(bytes[7])?;
155        let header_len = packet_header_len(fec_scheme, symbol_kind);
156        if bytes.len() < header_len {
157            return Err(Error::TruncatedPacket);
158        }
159        let mut session_id = [0_u8; 16];
160        session_id.copy_from_slice(&bytes[8..24]);
161        let mut fec_metadata = [0_u8; 12];
162        let (object_id, fragment_count) = match fec_scheme {
163            FecScheme::RaptorQ => {
164                fec_metadata[..5].copy_from_slice(&bytes[36..41]);
165                fec_metadata[6..].copy_from_slice(&bytes[41..47]);
166                (
167                    u64::from_be_bytes(bytes[28..36].try_into().unwrap()),
168                    u32::from_be_bytes(bytes[47..51].try_into().unwrap()),
169                )
170            }
171            FecScheme::RlcGf2 | FecScheme::RlcGf256 => {
172                let metadata_len = header_len - COMMON_HEADER_LEN;
173                fec_metadata[..metadata_len].copy_from_slice(&bytes[COMMON_HEADER_LEN..header_len]);
174                (0, 0)
175            }
176        };
177
178        Ok(Self {
179            header: WireHeader {
180                lane: Lane::try_from(bytes[4])?,
181                record_kind: RecordKind::try_from(bytes[5])?,
182                fec_scheme,
183                symbol_kind,
184                session_id,
185                sender_id: u32::from_be_bytes(bytes[24..28].try_into().unwrap()),
186                object_id,
187                fec_metadata,
188                fragment_count,
189            },
190            payload: &bytes[header_len..],
191        })
192    }
193}

195/// Encodes one packet into caller-owned storage and returns its exact length.
196pub fn encode_packet_into(header: WireHeader, payload: &[u8], output: &mut [u8]) -> Result<usize> {
197    if header.fec_scheme == FecScheme::RaptorQ && header.fec_metadata[5] != 0 {
198        return Err(Error::InvalidFecMetadata(
199            "RaptorQ OTI reserved byte is nonzero",
200        ));
201    }
202    let header_len = packet_header_len(header.fec_scheme, header.symbol_kind);
203    let needed = header_len
204        .checked_add(payload.len())
205        .ok_or(Error::InvalidConfig("packet length overflow"))?;
206    if output.len() < needed {
207        return Err(Error::BufferTooSmall {
208            needed,
209            available: output.len(),
210        });
211    }
212
213    let bytes = &mut output[..needed];
214    bytes.fill(0);
215    bytes[..4].copy_from_slice(&PACKET_MAGIC);
216    bytes[4] = header.lane as u8;
217    bytes[5] = header.record_kind as u8;
218    bytes[6] = header.fec_scheme as u8;
219    bytes[7] = header.symbol_kind as u8;
220    bytes[8..24].copy_from_slice(&header.session_id);
221    bytes[24..28].copy_from_slice(&header.sender_id.to_be_bytes());
222    match header.fec_scheme {
223        FecScheme::RaptorQ => {
224            bytes[28..36].copy_from_slice(&header.object_id.to_be_bytes());
225            bytes[36..41].copy_from_slice(&header.fec_metadata[..5]);
226            bytes[41..47].copy_from_slice(&header.fec_metadata[6..]);
227            bytes[47..51].copy_from_slice(&header.fragment_count.to_be_bytes());
228        }
229        FecScheme::RlcGf2 | FecScheme::RlcGf256 => {
230            bytes[COMMON_HEADER_LEN..header_len]
231                .copy_from_slice(&header.fec_metadata[..header_len - COMMON_HEADER_LEN]);
232        }
233    }
234    bytes[header_len..].copy_from_slice(payload);
235    Ok(needed)
236}

416    #[test]
417    fn packet_extent_defines_payload_length_for_every_layout() {
418        for scheme in [FecScheme::RlcGf2, FecScheme::RlcGf256, FecScheme::RaptorQ] {
419            for kind in [FecSymbolKind::Source, FecSymbolKind::Repair] {
420                let mut packet = if scheme == FecScheme::RaptorQ {
421                    fixture()
422                } else {
423                    rlc_fixture(scheme, kind)
424                };
425                packet.header.symbol_kind = kind;
426                let header_len = packet_header_len(scheme, kind);
427                for payload_len in [0, 1, crate::DEFAULT_MAX_SYMBOL_SIZE, u16::MAX as usize + 1] {
428                    packet.payload = vec![0x5a; payload_len];
429                    let needed = header_len + payload_len;
430                    let mut storage = vec![0xaa; needed + 1];
431                    assert_eq!(
432                        encode_packet_into(packet.header, &packet.payload, &mut storage).unwrap(),
433                        needed
434                    );
435                    assert_eq!(storage[needed], 0xaa);
436                    let decoded = WirePacketRef::decode(&storage[..needed]).unwrap();
437                    assert_eq!(decoded.header, packet.header);
438                    assert_eq!(decoded.payload, packet.payload);
439                    assert_eq!(decoded.payload.as_ptr(), storage[header_len..].as_ptr());
440                    // Complete packet boundaries and integrity belong to the carrier.
441                    let extended = WirePacketRef::decode(&storage).unwrap();
442                    assert_eq!(extended.payload.len(), payload_len + 1);
443                    assert_eq!(extended.payload[payload_len], 0xaa);
444                    for end in [header_len, header_len + payload_len / 2, needed] {
445                        assert_eq!(
446                            WirePacketRef::decode(&storage[..end]).unwrap().payload,
447                            &packet.payload[..end - header_len]
448                        );
449                    }
450                    assert_eq!(
451                        encode_packet_into(
452                            packet.header,
453                            &packet.payload,
454                            &mut storage[..needed - 1]
455                        ),
456                        Err(Error::BufferTooSmall {
457                            needed,
458                            available: needed - 1,
