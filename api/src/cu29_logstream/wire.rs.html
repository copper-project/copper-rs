5const PACKET_MAGIC: [u8; 4] = *b"CULS";

11/// Independently scheduled transport lanes.

77/// Fixed-endian metadata repeated on every independently reorderable symbol.
78#[derive(Clone, Copy, Debug, PartialEq, Eq)]
79pub struct WireHeader {
80    pub lane: Lane,
81    pub record_kind: RecordKind,
82    pub fec_scheme: FecScheme,
83    pub symbol_kind: FecSymbolKind,
84    pub session_id: [u8; 16],
85    pub sender_id: u32,
86    pub packet_sequence: u64,
87    pub object_id: u64,
88    /// Scheme-specific fixed-width FEC metadata.
89    pub fec_metadata: [u8; 12],
90    /// Scheme-specific 32-bit identifier: RLC fragment count or RaptorQ payload ID.
91    pub fragment_count: u32,
92}
93
94/// One validated wire symbol and its FEC payload.
95#[derive(Clone, Debug, PartialEq, Eq)]
96pub struct WirePacket {
97    pub header: WireHeader,
98    pub payload: Vec<u8>,
99}
100
101impl WirePacket {
102    pub fn encode(&self) -> Result<Vec<u8>> {
103        let mut bytes = alloc::vec![0; PACKET_HEADER_LEN + self.payload.len()];
104        let encoded = encode_packet_into(self.header, &self.payload, &mut bytes)?;
105        debug_assert_eq!(encoded, bytes.len());
106        Ok(bytes)
107    }
108
109    pub fn decode(bytes: &[u8]) -> Result<Self> {
110        if bytes.len() < PACKET_HEADER_LEN {
111            return Err(Error::TruncatedPacket);
112        }
113        if bytes[..4] != PACKET_MAGIC {
114            return Err(Error::InvalidMagic);
115        }
116        let version = bytes[4];
117        if version != WIRE_VERSION {
118            return Err(Error::UnsupportedVersion(version));
119        }
120        if bytes[5] as usize != PACKET_HEADER_LEN {
121            return Err(Error::InvalidHeaderLength(bytes[5]));
122        }
123        let payload_len = u16::from_be_bytes(bytes[64..66].try_into().unwrap()) as usize;
124        if bytes.len() != PACKET_HEADER_LEN + payload_len {
125            return Err(Error::PayloadLengthMismatch);
126        }
127
128        let expected_checksum =
129            u32::from_be_bytes(bytes[CRC_OFFSET..PACKET_HEADER_LEN].try_into().unwrap());
130        let mut digest = CRC32C.digest();
131        digest.update(&bytes[..CRC_OFFSET]);
132        digest.update(&[0_u8; 4]);
133        digest.update(&bytes[PACKET_HEADER_LEN..]);
134        if digest.finalize() != expected_checksum {
135            return Err(Error::CrcMismatch);
136        }
137
138        let mut session_id = [0_u8; 16];
139        session_id.copy_from_slice(&bytes[12..28]);
140        let mut fec_metadata = [0_u8; 12];
141        fec_metadata.copy_from_slice(&bytes[48..60]);
142
143        Ok(Self {
144            header: WireHeader {
145                lane: Lane::try_from(bytes[6])?,
146                record_kind: RecordKind::try_from(bytes[7])?,
147                fec_scheme: FecScheme::try_from(bytes[8])?,
148                symbol_kind: FecSymbolKind::try_from(bytes[9])?,
149                session_id,
150                sender_id: u32::from_be_bytes(bytes[28..32].try_into().unwrap()),
151                packet_sequence: u64::from_be_bytes(bytes[32..40].try_into().unwrap()),
152                object_id: u64::from_be_bytes(bytes[40..48].try_into().unwrap()),
153                fec_metadata,
154                fragment_count: u32::from_be_bytes(bytes[60..64].try_into().unwrap()),
155            },
156            payload: bytes[PACKET_HEADER_LEN..].to_vec(),
157        })
158    }
159}

161/// Encodes one packet into caller-owned storage and returns its exact length.
162pub fn encode_packet_into(header: WireHeader, payload: &[u8], output: &mut [u8]) -> Result<usize> {
163    let payload_len = u16::try_from(payload.len())
164        .map_err(|_| Error::InvalidConfig("wire payload exceeds u16"))?;
165    let needed = PACKET_HEADER_LEN
166        .checked_add(payload.len())
167        .ok_or(Error::InvalidConfig("packet length overflow"))?;
168    if output.len() < needed {
169        return Err(Error::BufferTooSmall {
170            needed,
171            available: output.len(),
172        });
173    }
174
175    let bytes = &mut output[..needed];
176    bytes.fill(0);
177    bytes[..4].copy_from_slice(&PACKET_MAGIC);
178    bytes[4] = WIRE_VERSION;
179    bytes[5] = PACKET_HEADER_LEN as u8;
180    bytes[6] = header.lane as u8;
181    bytes[7] = header.record_kind as u8;
182    bytes[8] = header.fec_scheme as u8;
183    bytes[9] = header.symbol_kind as u8;
184    bytes[12..28].copy_from_slice(&header.session_id);
185    bytes[28..32].copy_from_slice(&header.sender_id.to_be_bytes());
186    bytes[32..40].copy_from_slice(&header.packet_sequence.to_be_bytes());
187    bytes[40..48].copy_from_slice(&header.object_id.to_be_bytes());
188    bytes[48..60].copy_from_slice(&header.fec_metadata);
189    bytes[60..64].copy_from_slice(&header.fragment_count.to_be_bytes());
190    bytes[64..66].copy_from_slice(&payload_len.to_be_bytes());
191    bytes[PACKET_HEADER_LEN..].copy_from_slice(payload);
192    let checksum = CRC32C.checksum(bytes);
193    bytes[CRC_OFFSET..PACKET_HEADER_LEN].copy_from_slice(&checksum.to_be_bytes());
194    Ok(needed)
195}

202    fn fixture() -> WirePacket {
203        WirePacket {
204            header: WireHeader {
205                lane: Lane::LargeObject,

210                sender_id: 0x2233_4455,
211                packet_sequence: 0x6677_8899_aabb_ccdd,
212                object_id: 0x0102_0304_0506_0708,
213                fec_metadata: [0, 0, 0, 0, 3, 0, 0, 8, 1, 0, 1, 1],
214                fragment_count: 1,
215            },
216            payload: vec![0, 0, 0, 0, 1, 2, 3, 0, 0, 0, 0, 0],

220    #[test]
221    fn fixed_header_has_a_stable_golden_prefix() {
222        let encoded = fixture().encode().unwrap();
223        assert_eq!(
224            &encoded[..12],
225            &[b'C', b'U', b'L', b'S', 1, 72, 2, 2, 2, 0, 0, 0]
226        );
227        assert_eq!(WirePacket::decode(&encoded).unwrap(), fixture());
228    }
229
230    #[test]
231    fn corruption_is_rejected_before_fec() {
232        let mut encoded = fixture().encode().unwrap();
233        *encoded.last_mut().unwrap() ^= 0x40;
234        assert_eq!(WirePacket::decode(&encoded), Err(Error::CrcMismatch));
235    }
236
237    #[test]
238    fn caller_owned_encoding_matches_allocating_encoding() {
239        let packet = fixture();
240        let expected = packet.encode().unwrap();
241        let mut storage = [0xaa; 128];
242        let encoded = encode_packet_into(packet.header, &packet.payload, &mut storage).unwrap();
243
244        assert_eq!(&storage[..encoded], expected);
245        assert!(storage[encoded..].iter().all(|byte| *byte == 0xaa));
246    }
