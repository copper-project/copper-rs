30/// One application-typed archive per `(session id, sender id)`. Pass all ordered
31/// router events for that sender to `accept`. The expected schema comes from
32/// `P`'s generated output specs, independently of the remote manifest.
33/// `CaptureArchive` applies the generated selective codec for a live twin.
34/// A write failure poisons the writer: do not retry a partially committed event.
35pub struct NativeArchive<P: CopperListTuple> {
36    copperlists: NativeStream,
37    keyframes: NativeStream,
38    continuity: NativeStream,
39    identity: StreamIdentity,
40    manifest: SessionManifest,
41    next_id: u64,
42    poisoned: bool,
43    payload: PhantomData<P>,
44    decode: ArchiveDecoder<P>,
45}
46
47struct CanonicalEntry<'a>(&'a [u8]);
48impl Encode for CanonicalEntry<'_> {
49    fn encode<E: Encoder>(&self, encoder: &mut E) -> core::result::Result<(), EncodeError> {
50        encoder.writer().write(self.0)
51    }
52}

54impl<P: CopperListTuple> NativeArchive<P> {
55    /// Creates native CopperList, FrozenTasks and StreamContinuity sections.
56    /// The section size must fit the largest accepted canonical entry.
57    pub fn new(
58        path: &Path,
59        manifest: SessionManifest,
60        slab_bytes: usize,
61        section_bytes: usize,
62    ) -> Result<Self> {
63        let expected = ApplicationSchema::from_output_specs(P::get_output_specs());
64        Self::new_checked(
65            path,
66            manifest,
67            slab_bytes,
68            section_bytes,
69            expected,
70            |bytes| Ok((CapturedList::new(crate::decode_copperlist(bytes)?), bytes)),
71        )
72    }
73
74    fn new_checked(
75        path: &Path,
76        manifest: SessionManifest,
77        slab_bytes: usize,
78        section_bytes: usize,
79        expected_schema: ApplicationSchema,
80        decode: ArchiveDecoder<P>,
81    ) -> Result<Self> {
82        manifest.plan.validate()?;
83        if manifest.version != crate::SESSION_MANIFEST_VERSION {
84            return Err(Error::UnsupportedManifestVersion(manifest.version));
85        }
86        if manifest.application_schema != expected_schema || !manifest.plan.content.archive {
87            return Err(Error::InvalidConfig(
88                "archive requires the matching application schema and archive content policy",
89            ));
90        }
91        if section_bytes == 0 || section_bytes > slab_bytes {
92            return Err(Error::InvalidConfig("archive section must fit in its slab"));
93        }
94        let logger = UnifiedLoggerBuilder::new()
95            .file_base_name(path)
96            .preallocated_size(slab_bytes)
97            .write(true)
98            .create(true)
99            .build()
100            .map_err(io_error)?;
101        let UnifiedLogger::Write(logger) = logger else {
102            unreachable!()
103        };
104        let logger = Arc::new(Mutex::new(logger));
105        let mut archive = Self {
106            copperlists: NativeStream::new(
107                UnifiedLogType::CopperList,
108                logger.clone(),
109                section_bytes,
110            )
111            .map_err(io_error)?,
112            keyframes: NativeStream::new(
113                UnifiedLogType::FrozenTasks,
114                logger.clone(),
115                section_bytes,
116            )
117            .map_err(io_error)?,
118            continuity: NativeStream::new(UnifiedLogType::StreamContinuity, logger, section_bytes)
119                .map_err(io_error)?,
120            identity: manifest.identity,
121            manifest,
122            next_id: 0,
123            poisoned: false,
124            payload: PhantomData,
125            decode,
126        };
127        archive
128            .continuity
129            .log(&StreamContinuityRecord::Manifest {
130                record: archive.manifest.encode_record()?,
131            })
132            .map_err(io_error)?;
133        Ok(archive)
134    }
135
136    /// Validates a typed CopperList once, appends its original canonical bytes,
137    /// and returns the typed value for an in-process consumer. Keyframes are
138    /// archived only after the router has verified their anchor references.
139    pub fn accept(&mut self, event: &SessionEvent) -> Result<Option<CopperList<P>>> {
140        self.accept_capture(event)
141            .map(|capture| capture.map(|c| c.copperlist))
142    }
143
144    fn accept_capture(&mut self, event: &SessionEvent) -> Result<Option<CapturedList<P>>> {
145        if self.poisoned {
146            return Err(Error::InvalidConfig(
147                "archive failed; close it before continuing",
148            ));
149        }
150        let result = self.accept_inner(event);
151        if result.is_err() {
152            self.poisoned = true;
153        }
154        result
155    }
156
157    fn accept_inner(&mut self, event: &SessionEvent) -> Result<Option<CapturedList<P>>> {
158        let identity = match event {
159            SessionEvent::Manifest(manifest) => manifest.identity,
160            SessionEvent::ContinuousRecord { identity, .. }
161            | SessionEvent::Gap { identity, .. }
162            | SessionEvent::Object { identity, .. }
163            | SessionEvent::VerifiedAnchor { identity, .. } => *identity,
164        };
165        if identity != self.identity {
166            return Err(Error::InconsistentObject);
167        }
168        match event {
169            SessionEvent::Manifest(manifest) => {
170                if manifest != &self.manifest {
171                    return Err(Error::InconsistentObject);
172                }
173            }
174            SessionEvent::ContinuousRecord { record, .. } => {
175                let decoded = record.decoded()?;
176                if decoded.kind != RecordKind::CopperList || decoded.object_id != self.next_id {
177                    return Err(Error::InconsistentObject);
178                }
179                let (capture, native) = (self.decode)(decoded.payload)?;
180                let copperlist = &capture.copperlist;
181                if copperlist.id != decoded.object_id {
182                    return Err(Error::InconsistentObject);
183                }
184                let next = self
185                    .next_id
186                    .checked_add(1)
187                    .ok_or(Error::InconsistentObject)?;
188                self.copperlists
189                    .log(&CanonicalEntry(native))
190                    .map_err(io_error)?;
191                self.next_id = next;
192                return Ok(Some(capture));
193            }
194            SessionEvent::Gap { gap, .. } => {
195                if gap.first_id != self.next_id || gap.last_id < gap.first_id {
196                    return Err(Error::InconsistentObject);
197                }
198                let next = gap
199                    .last_id
200                    .checked_add(1)
201                    .ok_or(Error::InconsistentObject)?;
202                let reason = match gap.reason {
203                    crate::GapReason::RlcWindowExpired => SourceGapReason::RlcWindowExpired,
204                    crate::GapReason::SessionEnded => SourceGapReason::SessionEnded,
205                    crate::GapReason::LateJoin => SourceGapReason::LateJoin,
206                    crate::GapReason::AnchorRecovery => SourceGapReason::AnchorRecovery,
207                };
208                self.continuity
209                    .log(&StreamContinuityRecord::Gap {
210                        first_id: gap.first_id,
211                        last_id: gap.last_id,
212                        reason,
213                    })
214                    .map_err(io_error)?;
215                self.next_id = next;
216            }
217            SessionEvent::VerifiedAnchor {
218                anchor, keyframe, ..
219            } => {
220                let decoded = keyframe.decoded()?;
221                let manifest_record = self.manifest.encode_record()?;
222                if !anchor.references_manifest(crate::decode_record(&manifest_record)?)
223                    || !anchor.references_keyframe(decoded)
224                    || crate::decode_keyframe(decoded.payload)?.culistid != anchor.copperlist_id
225                {
226                    return Err(Error::InconsistentObject);
227                }
228                self.keyframes
229                    .log(&CanonicalEntry(decoded.payload))
230                    .map_err(io_error)?;
231                let record = crate::encode_record(
232                    RecordKind::Anchor,
233                    anchor.copperlist_id,
234                    &crate::encode_anchor(anchor)?,
235                )?;
236                self.continuity
237                    .log(&StreamContinuityRecord::Anchor {
238                        copperlist_id: anchor.copperlist_id,
239                        record,
240                    })
241                    .map_err(io_error)?;
242            }
243            SessionEvent::Object { .. } => {}
244        }
245        Ok(None)
246    }
247
248    /// Records the receiver's final known boundary and closes native sections.
249    /// The router must first be finalized through any caller-known sender tail.
250    pub fn finish(mut self) -> Result<()> {
251        if self.poisoned {
252            return Err(Error::InvalidConfig("archive failed before finalization"));
253        }
254        self.continuity
255            .log(&StreamContinuityRecord::Finished {
256                next_copperlist_id: self.next_id,
257            })
258            .map_err(io_error)
259    }
260}
261
262fn io_error(error: impl core::fmt::Display) -> Error {
263    Error::Codec(error.to_string())
264}

266/// Capture-aware native archive. Synthesized payloads never enter this writer.
267/// Native CopperList metadata retains both original and captured presence planes;
268/// StreamContinuity retains the reconstruction contract and source gaps.
269pub struct CaptureArchive<P: crate::capture::CaptureDataSet>(NativeArchive<P>);
270impl<P: crate::capture::CaptureDataSet> CaptureArchive<P> {
271    pub fn new(
272        path: &Path,
273        manifest: SessionManifest,
274        slab_bytes: usize,
275        section_bytes: usize,
276    ) -> Result<Self> {
277        Ok(Self(NativeArchive::new_checked(
278            path,
279            manifest,
280            slab_bytes,
281            section_bytes,
282            P::stream_schema(),
283            crate::capture::decode_capture,
284        )?))
285    }
286    pub fn accept(
287        &mut self,
288        event: &SessionEvent,
289    ) -> Result<Option<crate::capture::CapturedList<P>>> {
290        self.0.accept_capture(event)
291    }
292
293    pub fn finish(self) -> Result<()> {
294        self.0.finish()
295    }
