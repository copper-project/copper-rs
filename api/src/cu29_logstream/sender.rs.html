54impl LogStreamSenderConfig {
55    pub fn validate(&self) -> Result<()> {
56        if self.continuous.identity != self.recovery.finite.identity {
57            return Err(Error::InvalidConfig(
58                "continuous and recovery sender identities differ",
59            ));
60        }
61        if self.continuous.lane == self.recovery.finite.lane {
62            return Err(Error::InvalidConfig(
63                "continuous and finite objects require distinct lanes",
64            ));
65        }
66        let manifest = decode_record(&self.recovery.manifest_record)?;
67        if manifest.kind != RecordKind::Manifest {
68            return Err(Error::InvalidConfig(
69                "recovery configuration requires a manifest record",
70            ));
71        }
72        if self.recovery.anchor_interval == 0 {
73            return Err(Error::InvalidConfig("anchor interval must be nonzero"));
74        }
75        Ok(())
76    }
77}

121impl<T: CuStreamTx> KeyFrameAnchorSink<T> {
122    pub fn new(transport: T, config: RecoverySenderConfig) -> Result<Self> {
123        if config.anchor_interval == 0 {
124            return Err(Error::InvalidConfig("anchor interval must be nonzero"));
125        }
126        let manifest = decode_record(&config.manifest_record)?;
127        if manifest.kind != RecordKind::Manifest {
128            return Err(Error::InvalidConfig(

130            ));
131        }
132        let manifest_object_id = manifest.object_id;
133        let manifest_record_digest = manifest.digest;
134        Ok(Self {
135            transport,
136            encoder: FiniteObjectEncoder::new(config.finite)?,
137            manifest_record: config.manifest_record,
138            manifest_object_id,
139            manifest_record_digest,
140            anchor_interval: config.anchor_interval,
141            stats: RecoverySenderStats::default(),
142        })
143    }

171fn emit_finite_record<T: CuStreamTx>(
172    encoder: &mut FiniteObjectEncoder,
173    transport: &mut T,
174    stats: &mut RecoverySenderStats,
175    record: &[u8],
176) -> Result<()> {
177    let datagrams = encoder.push_record_with(record, |datagram| {
178        stats.object_datagrams = stats.object_datagrams.saturating_add(1);
179        match transport.try_send(datagram) {
180            Ok(()) => stats.datagrams_sent = stats.datagrams_sent.saturating_add(1),
181            Err(CuStreamTxError::WouldBlock) => {
182                stats.datagrams_dropped = stats.datagrams_dropped.saturating_add(1);
183            }
184            Err(CuStreamTxError::Failed(message)) => return Err(Error::Transport(message)),
185        }
186        Ok(())
187    })?;
188    debug_assert!(datagrams > 0);
189    Ok(())
190}

192impl<T: CuStreamTx> WriteStream<KeyFrame> for KeyFrameAnchorSink<T> {
193    fn log(&mut self, keyframe: &KeyFrame) -> CuResult<()> {
194        if !keyframe
195            .culistid
196            .is_multiple_of(u64::from(self.anchor_interval))
197        {
198            return Ok(());
199        }
200        let (keyframe_record, anchor_record) = encode_keyframe_and_anchor(
201            keyframe,
202            self.manifest_object_id,
203            self.manifest_record_digest,
204        )
205        .map_err(|error| CuError::from(error.to_string()))?;
206        emit_finite_record(
207            &mut self.encoder,
208            &mut self.transport,
209            &mut self.stats,
210            &self.manifest_record,
211        )
212        .and_then(|()| self.emit_record(&keyframe_record))
213        .and_then(|()| self.emit_record(&anchor_record))
214        .map_err(|error| CuError::from(error.to_string()))?;
215        self.stats.keyframes_encoded = self.stats.keyframes_encoded.saturating_add(1);
216        Ok(())
217    }
218}

256impl<P, T, const MAX_SYMBOL_SIZE: usize, const MAX_WINDOW_SYMBOLS: usize>
257    ContinuousCopperListSink<P, T, MAX_SYMBOL_SIZE, MAX_WINDOW_SYMBOLS>
258where
259    P: CopperListTuple,
260    T: CuStreamTx,
261{
262    pub fn new(transport: T, config: ContinuousSenderConfig) -> Result<Self> {
263        if config.repair_every_source_symbols == 0 {
264            return Err(Error::InvalidConfig(
265                "repair interval must contain at least one source symbol",
266            ));
267        }
268        let datagram_bytes = PACKET_HEADER_LEN
269            .checked_add(config.fec.symbol_size())
270            .ok_or(Error::InvalidConfig("datagram length overflow"))?;
271        let encoder = ContinuousEncoder::new(
272            config.identity,
273            config.first_packet_sequence,
274            config.lane,
275            config.fec,
276            config.max_record_bytes,
277            config.initial_esi,
278        )?;
279        Ok(Self {
280            transport,
281            encoder: Box::new(encoder),
282            record: vec![0; config.max_record_bytes],
283            datagram: vec![0; datagram_bytes],
284            repair_schedule: RepairSchedule::new(
285                config.repair_every_source_symbols,
286                config.first_repair_key,
287                config.repair_density,
288            ),
289            stats: ContinuousSenderStats::default(),
290            _payload: PhantomData,
291        })
292    }
293
294    pub const fn stats(&self) -> ContinuousSenderStats {
295        self.stats
296    }
297
298    pub fn transport(&self) -> &T {
299        &self.transport
300    }
301
302    pub fn transport_mut(&mut self) -> &mut T {
303        &mut self.transport
304    }
305
306    pub fn into_transport(self) -> T {
307        self.transport
308    }
309
310    fn emit_source_record(&mut self, record_len: usize) -> Result<()> {
311        let transport = &mut self.transport;
312        let stats = &mut self.stats;
313        let (source_datagrams, repair_datagrams) = self.encoder.push_record_with_repairs(
314            &self.record[..record_len],
315            &mut self.datagram,
316            &mut |datagram| submit_datagram(transport, stats, datagram),
317            &mut self.repair_schedule,
318        )?;
319        stats.source_datagrams = stats
320            .source_datagrams
321            .saturating_add(source_datagrams as u64);
322        stats.repair_datagrams = stats
323            .repair_datagrams
324            .saturating_add(repair_datagrams as u64);
325        Ok(())
326    }
327}

329impl<P, T, const MAX_SYMBOL_SIZE: usize, const MAX_WINDOW_SYMBOLS: usize> WriteStream<CopperList<P>>
330    for ContinuousCopperListSink<P, T, MAX_SYMBOL_SIZE, MAX_WINDOW_SYMBOLS>
331where
332    P: CopperListTuple + Send + Sync,
333    T: CuStreamTx,
334{
335    fn log(&mut self, copperlist: &CopperList<P>) -> CuResult<()> {
336        let record_len = encode_copperlist_record_into(copperlist, &mut self.record)
337            .map_err(|error| CuError::from(error.to_string()))?;
338        self.emit_source_record(record_len)
339            .map_err(|error| CuError::from(error.to_string()))?;
340        self.stats.records_encoded = self.stats.records_encoded.saturating_add(1);
341        Ok(())
342    }
343}

345fn submit_datagram<T: CuStreamTx>(
346    transport: &mut T,
347    stats: &mut ContinuousSenderStats,
348    datagram: &[u8],
349) -> Result<()> {
350    match transport.try_send(datagram) {
351        Ok(()) => stats.datagrams_sent = stats.datagrams_sent.saturating_add(1),
352        Err(CuStreamTxError::WouldBlock) => {
353            stats.datagrams_dropped = stats.datagrams_dropped.saturating_add(1);
354        }
355        Err(CuStreamTxError::Failed(message)) => return Err(Error::Transport(message)),
356    }
357    Ok(())
