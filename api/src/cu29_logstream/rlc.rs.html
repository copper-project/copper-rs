63impl ContinuousRecoveryStats {
64    /// Percentage of expected systematic symbols that did not arrive intact.
65    pub const fn destruction_basis_points(self, expected_source_symbols: usize) -> u16 {
66        if expected_source_symbols == 0 {
67            return 0;
68        }
69        let destroyed = expected_source_symbols.saturating_sub(self.source_symbols_received);
70        ((destroyed as u64 * 10_000) / expected_source_symbols as u64) as u16
71    }
72
73    /// Percentage of missing systematic symbols reconstructed by FEC.
74    pub const fn source_recovery_basis_points(self, expected_source_symbols: usize) -> u16 {
75        let missing = expected_source_symbols.saturating_sub(self.source_symbols_received);
76        if missing == 0 {
77            return 10_000;
78        }
79        let recovered = if self.source_symbols_recovered < missing {
80            self.source_symbols_recovered
81        } else {
82            missing
83        };
84        ((recovered as u64 * 10_000) / missing as u64) as u16
85    }
86
87    /// Percentage of expected semantic records reconstructed and digest-verified.
88    pub const fn semantic_recovery_basis_points(self, expected_records: usize) -> u16 {
89        if expected_records == 0 {
90            return 0;
91        }
92        let recovered = if self.records_recovered < expected_records {
93            self.records_recovered
94        } else {
95            expected_records
96        };
97        ((recovered as u64 * 10_000) / expected_records as u64) as u16
98    }
99}

155    pub fn push_record(&mut self, record: &[u8], output: &mut Vec<Vec<u8>>) -> Result<usize> {
156        let mut datagram = vec![0; PACKET_HEADER_LEN + self.config().symbol_size()];

175    pub(crate) fn push_record_with_repairs(
176        &mut self,
177        record: &[u8],
178        datagram: &mut [u8],
179        emit: &mut impl FnMut(&[u8]) -> Result<()>,
180        repair_schedule: &mut RepairSchedule,
181    ) -> Result<(usize, usize)> {

185    fn push_record_inner(
186        &mut self,
187        record: &[u8],
188        datagram: &mut [u8],
189        emit: &mut impl FnMut(&[u8]) -> Result<()>,
190        mut repair_schedule: Option<&mut RepairSchedule>,
191    ) -> Result<(usize, usize)> {
192        let decoded = decode_record(record)?;
193        if decoded.kind != RecordKind::CopperList {
194            return Err(Error::InconsistentObject);
195        }
196        if record.len() > self.max_record_bytes {
197            return Err(Error::ObjectTooLarge {
198                actual: record.len() as u64,
199                maximum: self.max_record_bytes as u64,
200            });
201        }
202        let record_len = u32::try_from(record.len())
203            .map_err(|_| Error::InvalidConfig("record length exceeds u32"))?;
204        let fragment_capacity = fragment_capacity(self.config())?;
205        let fragment_count = record.len().div_ceil(fragment_capacity);
206        let fragment_count = u32::try_from(fragment_count)
207            .map_err(|_| Error::InvalidConfig("fragment count exceeds u32"))?;
208        let mut symbol = [0_u8; MAX_SYMBOL_SIZE];
209        let mut repair_symbols = 0usize;
210
211        for (fragment_index, chunk) in record.chunks(fragment_capacity).enumerate() {
212            let active_symbol = &mut symbol[..self.config().symbol_size()];
213            active_symbol.fill(0);
214            encode_fragment_header(
215                active_symbol,
216                decoded.kind,
217                decoded.object_id,
218                record_len,
219                fragment_index as u32,
220                fragment_count,
221                chunk.len() as u16,
222            );
223            active_symbol[FRAGMENT_HEADER_LEN..FRAGMENT_HEADER_LEN + chunk.len()]
224                .copy_from_slice(chunk);
225            let esi = self.fec.push_source(active_symbol)?;
226            let mut fec_metadata = [0_u8; 12];
227            fec_metadata[..4].copy_from_slice(&SourcePayloadId::new(esi).to_bytes());
228            let header = self.packet_header(
229                decoded.kind,
230                decoded.object_id,
231                fragment_count,
232                FecSymbolKind::Source,
233                fec_metadata,
234            );
235            self.emit_packet_with(header, active_symbol, datagram, emit)?;
236            if let Some(schedule) = repair_schedule.as_mut() {
237                schedule.source_symbols_since_repair =
238                    schedule.source_symbols_since_repair.saturating_add(1);
239                if schedule.source_symbols_since_repair >= schedule.every {
240                    schedule.source_symbols_since_repair -= schedule.every;
241                    self.push_repair_with(
242                        RepairParameters::new(schedule.next_repair_key, schedule.density),
243                        datagram,
244                        &mut *emit,
245                    )?;
246                    schedule.next_repair_key = schedule.next_repair_key.wrapping_add(1);
247                    repair_symbols = repair_symbols.saturating_add(1);
248                }
249            }
250        }
251        Ok((fragment_count as usize, repair_symbols))
252    }

257        parameters: RepairParameters,

259    ) -> Result<RepairPayloadId> {
260        let mut datagram = vec![0; PACKET_HEADER_LEN + self.config().symbol_size()];

262            output.push(packet.to_vec());
263            Ok(())

267    /// Emits one repair packet immediately from reusable caller-owned storage.
268    pub fn push_repair_with(
269        &mut self,
270        parameters: RepairParameters,
271        datagram: &mut [u8],
272        mut emit: impl FnMut(&[u8]) -> Result<()>,
273    ) -> Result<RepairPayloadId> {
274        let mut symbol = [0_u8; MAX_SYMBOL_SIZE];
275        let active_symbol = &mut symbol[..self.config().symbol_size()];
276        let id = self.fec.encode_repair(parameters, active_symbol)?;
277        let mut fec_metadata = [0_u8; 12];
278        fec_metadata[..8].copy_from_slice(&id.to_bytes());
279        let header = self.packet_header(
280            RecordKind::CopperList,
281            0,
282            0,
283            FecSymbolKind::Repair,
284            fec_metadata,
285        );
286        self.emit_packet_with(header, active_symbol, datagram, &mut emit)?;
287        Ok(id)
288    }

305            packet_sequence: self.packet_sequence,
306            object_id,

312    fn emit_packet_with(
313        &mut self,
314        header: WireHeader,
315        payload: &[u8],
316        datagram: &mut [u8],
317        emit: &mut impl FnMut(&[u8]) -> Result<()>,
318    ) -> Result<()> {
319        let encoded = encode_packet_into(header, payload, datagram)?;
320        emit(&datagram[..encoded])?;
321        self.packet_sequence = self.packet_sequence.wrapping_add(1);
322        Ok(())
323    }

405    /// Validates and submits one datagram. Corrupt or foreign packets are counted
406    /// and ignored before they can contaminate the FEC decoder.
407    pub fn receive_datagram(&mut self, datagram: &[u8]) -> Result<()> {
408        self.stats.datagrams_seen = self.stats.datagrams_seen.saturating_add(1);
409        if self.stats.datagrams_seen > self.limits.max_datagrams {
410            return Err(Error::TooManyDatagrams {
411                actual: self.stats.datagrams_seen,
412                maximum: self.limits.max_datagrams,
413            });
414        }
415        let packet = match WirePacket::decode(datagram) {
416            Ok(packet) => packet,
417            Err(_) => {
418                self.stats.invalid_datagrams = self.stats.invalid_datagrams.saturating_add(1);
419                return Ok(());
420            }
421        };
422        if packet.header.session_id != self.identity.session_id
423            || packet.header.sender_id != self.identity.sender_id
424            || packet.header.lane != self.lane
425            || packet.header.fec_scheme != fec_scheme(self.fec.config().field())
426            || packet.header.record_kind != RecordKind::CopperList
427        {
428            self.stats.inconsistent_datagrams = self.stats.inconsistent_datagrams.saturating_add(1);
429            return Ok(());
430        }
431        if packet.payload.len() != self.fec.config().symbol_size() {
432            self.stats.invalid_datagrams = self.stats.invalid_datagrams.saturating_add(1);
433            return Ok(());
434        }
435
436        let accepted = match packet.header.symbol_kind {
437            FecSymbolKind::Source => self.receive_source_packet(&packet)?,
438            FecSymbolKind::Repair => self.receive_repair_packet(&packet)?,
439        };
440        if !accepted {
441            return Ok(());
442        }
443        self.stats.valid_datagrams = self.stats.valid_datagrams.saturating_add(1);
444        self.process_new_symbols()
445    }
446
447    fn receive_source_packet(&mut self, packet: &WirePacket) -> Result<bool> {
448        if packet.header.fec_metadata[4..]
449            .iter()
450            .any(|byte| *byte != 0)
451        {
452            self.stats.invalid_datagrams = self.stats.invalid_datagrams.saturating_add(1);
453            return Ok(false);
454        }
455        let id = SourcePayloadId::from_bytes(packet.header.fec_metadata[..4].try_into().unwrap());
456        let fragment = match decode_fragment(&packet.payload, fragment_capacity(self.fec.config())?)
457        {
458            Ok(fragment) => fragment,
459            Err(_) => {
460                self.stats.invalid_datagrams = self.stats.invalid_datagrams.saturating_add(1);
461                return Ok(false);
462            }
463        };
464        if packet.header.object_id != fragment.object_id
465            || packet.header.fragment_count as usize != fragment.fragment_count
466        {
467            self.stats.inconsistent_datagrams = self.stats.inconsistent_datagrams.saturating_add(1);
468            return Ok(false);
469        }
470        if self.source_ids.contains(&id.esi()) {
471            self.stats.duplicate_symbols = self.stats.duplicate_symbols.saturating_add(1);
472            return Ok(false);
473        }
474        self.source_ids.push(id.esi());
475        self.stats.source_symbols_received = self.stats.source_symbols_received.saturating_add(1);
476        let report = self.fec.receive_source(id.esi(), &packet.payload)?;
477        self.stats.source_symbols_recovered = self
478            .stats
479            .source_symbols_recovered
480            .saturating_add(report.recovered());
481        Ok(true)
482    }
483
484    fn receive_repair_packet(&mut self, packet: &WirePacket) -> Result<bool> {
485        if packet.header.object_id != 0
486            || packet.header.fragment_count != 0
487            || packet.header.fec_metadata[8..]
488                .iter()
489                .any(|byte| *byte != 0)
490        {
491            self.stats.invalid_datagrams = self.stats.invalid_datagrams.saturating_add(1);
492            return Ok(false);
493        }
494        let raw_id: [u8; 8] = packet.header.fec_metadata[..8].try_into().unwrap();
495        if self.repair_ids.contains(&raw_id) {
496            self.stats.duplicate_symbols = self.stats.duplicate_symbols.saturating_add(1);
497            return Ok(false);
498        }
499        let id = RepairPayloadId::from_bytes(raw_id)?;
500        let report = self.fec.receive_repair(id, &packet.payload)?;
501        self.repair_ids.push(raw_id);
502        self.stats.repair_symbols_received = self.stats.repair_symbols_received.saturating_add(1);
503        if report.is_innovative() {
504            self.stats.innovative_repairs = self.stats.innovative_repairs.saturating_add(1);
505        }
506        self.stats.source_symbols_recovered = self
507            .stats
508            .source_symbols_recovered
509            .saturating_add(report.recovered());
510        Ok(true)
511    }
512
513    fn process_new_symbols(&mut self) -> Result<()> {
514        let pending = self
515            .fec
516            .known_symbols()
517            .filter(|(esi, _)| !self.processed_symbols.contains(esi))
518            .map(|(esi, symbol)| (esi, symbol.to_vec()))
519            .collect::<Vec<_>>();
520        for (esi, symbol) in pending {
521            let symbol_count = self.processed_symbols.len().saturating_add(1);
522            if symbol_count > self.limits.max_source_symbols {
523                return Err(Error::TooManySourceSymbols {
524                    actual: symbol_count,
525                    maximum: self.limits.max_source_symbols,
526                });
527            }
528            self.process_fragment(&symbol)?;
529            self.processed_symbols.push(esi);
530        }
531        Ok(())
532    }
533
534    fn process_fragment(&mut self, symbol: &[u8]) -> Result<()> {
535        let fragment = decode_fragment(symbol, fragment_capacity(self.fec.config())?)?;
536        if fragment.record_len > self.limits.max_record_bytes {
537            return Err(Error::ObjectTooLarge {
538                actual: fragment.record_len as u64,
539                maximum: self.limits.max_record_bytes as u64,
540            });
541        }
542        let assembly_index = match self.assemblies.iter().position(|assembly| {
543            assembly.kind == fragment.kind && assembly.object_id == fragment.object_id
544        }) {
545            Some(index) => index,
546            None => {
547                let count = self.assemblies.len().saturating_add(1);
548                if count > self.limits.max_incomplete_records {
549                    return Err(Error::TooManyRecords {
550                        actual: count,
551                        maximum: self.limits.max_incomplete_records,
552                    });
553                }
554                self.assemblies.push(RecordAssembly::new(&fragment));
555                self.assemblies.len() - 1
556            }
557        };
558        let complete = self.assemblies[assembly_index].insert(&fragment)?;
559        if !complete {
560            return Ok(());
561        }
562
563        let assembly = self.assemblies.swap_remove(assembly_index);
564        let decoded = decode_record(&assembly.bytes)?;
565        if decoded.kind != assembly.kind || decoded.object_id != assembly.object_id {
566            return Err(Error::InconsistentObject);
567        }
568        self.recovered.push(RecoveredRecord {
569            bytes: assembly.bytes,
570        });
571        self.stats.records_recovered = self.stats.records_recovered.saturating_add(1);
572        Ok(())
573    }
574}

576#[derive(Debug)]
577struct Fragment<'a> {
578    kind: RecordKind,
579    object_id: u64,
580    record_len: usize,
581    fragment_index: usize,
582    fragment_count: usize,
583    fragment_capacity: usize,
584    payload: &'a [u8],
585}
586
587struct RecordAssembly {
588    kind: RecordKind,
589    object_id: u64,
590    bytes: Vec<u8>,
591    received: Vec<bool>,
592    received_count: usize,
593    fragment_capacity: usize,
594}
595
596impl RecordAssembly {
597    fn new(fragment: &Fragment<'_>) -> Self {
598        Self {
599            kind: fragment.kind,
600            object_id: fragment.object_id,
601            bytes: vec![0; fragment.record_len],
602            received: vec![false; fragment.fragment_count],
603            received_count: 0,
604            fragment_capacity: fragment.fragment_capacity,
605        }
606    }
607
608    fn insert(&mut self, fragment: &Fragment<'_>) -> Result<bool> {
609        if self.bytes.len() != fragment.record_len || self.received.len() != fragment.fragment_count
610        {
611            return Err(Error::InvalidFragment("record geometry changed"));
612        }
613        let offset = fragment
614            .fragment_index
615            .checked_mul(self.fragment_capacity)
616            .ok_or(Error::InvalidFragment("fragment offset overflow"))?;
617        let end = offset
618            .checked_add(fragment.payload.len())
619            .ok_or(Error::InvalidFragment("fragment length overflow"))?;
620        if end > self.bytes.len() {
621            return Err(Error::InvalidFragment("fragment exceeds record"));
622        }
623        if self.received[fragment.fragment_index] {
624            if self.bytes[offset..end] != *fragment.payload {
625                return Err(Error::InvalidFragment("duplicate fragment differs"));
626            }
627            return Ok(self.received_count == self.received.len());
628        }
629        self.bytes[offset..end].copy_from_slice(fragment.payload);
630        self.received[fragment.fragment_index] = true;
631        self.received_count += 1;
632        Ok(self.received_count == self.received.len())
633    }
634}

636fn validate_config(config: RlcConfig, max_record_bytes: usize) -> Result<()> {
637    let _ = fragment_capacity(config)?;
638    if max_record_bytes == 0 {
639        return Err(Error::InvalidConfig("maximum record size must be nonzero"));
640    }
641    if max_record_bytes > u32::MAX as usize {
642        return Err(Error::InvalidConfig("maximum record size exceeds u32"));
643    }
644    Ok(())
645}
646
647fn fragment_capacity(config: RlcConfig) -> Result<usize> {
648    config
649        .symbol_size()
650        .checked_sub(FRAGMENT_HEADER_LEN)
651        .filter(|capacity| *capacity > 0 && *capacity <= u16::MAX as usize)
652        .ok_or(Error::InvalidConfig(
653            "RLC symbol cannot hold the fragment header and payload",
654        ))
655}
656
657fn fec_scheme(field: Field) -> FecScheme {
658    match field {
659        Field::Gf2 => FecScheme::RlcGf2,
660        Field::Gf256 => FecScheme::RlcGf256,
661    }
662}
663
664fn encode_fragment_header(
665    symbol: &mut [u8],
666    kind: RecordKind,
667    object_id: u64,
668    record_len: u32,
669    fragment_index: u32,
670    fragment_count: u32,
671    fragment_len: u16,
672) {
673    symbol[..4].copy_from_slice(&FRAGMENT_MAGIC);
674    symbol[4] = FRAGMENT_VERSION;
675    symbol[5] = kind as u8;
676    symbol[6..8].fill(0);
677    symbol[8..16].copy_from_slice(&object_id.to_be_bytes());
678    symbol[16..20].copy_from_slice(&record_len.to_be_bytes());
679    symbol[20..24].copy_from_slice(&fragment_index.to_be_bytes());
680    symbol[24..28].copy_from_slice(&fragment_count.to_be_bytes());
681    symbol[28..30].copy_from_slice(&fragment_len.to_be_bytes());
682    symbol[30..32].fill(0);
683}
684
685fn decode_fragment(symbol: &[u8], fragment_capacity: usize) -> Result<Fragment<'_>> {
686    if symbol.len() < FRAGMENT_HEADER_LEN || symbol[..4] != FRAGMENT_MAGIC {
687        return Err(Error::InvalidFragment("missing fragment header"));
688    }
689    if symbol[4] != FRAGMENT_VERSION {
690        return Err(Error::UnsupportedVersion(symbol[4]));
691    }
692    if symbol[6..8].iter().any(|byte| *byte != 0) || symbol[30..32].iter().any(|byte| *byte != 0) {
693        return Err(Error::InvalidFragment("reserved bytes are nonzero"));
694    }
695    let kind = RecordKind::try_from(symbol[5])?;
696    if kind != RecordKind::CopperList {
697        return Err(Error::InvalidFragment(
698            "continuous lane requires CopperList records",
699        ));
700    }
701    let object_id = u64::from_be_bytes(symbol[8..16].try_into().unwrap());
702    let record_len = u32::from_be_bytes(symbol[16..20].try_into().unwrap()) as usize;
703    let fragment_index = u32::from_be_bytes(symbol[20..24].try_into().unwrap()) as usize;
704    let fragment_count = u32::from_be_bytes(symbol[24..28].try_into().unwrap()) as usize;
705    let fragment_len = u16::from_be_bytes(symbol[28..30].try_into().unwrap()) as usize;
706    if record_len == 0 || fragment_count == 0 || fragment_index >= fragment_count {
707        return Err(Error::InvalidFragment("invalid fragment geometry"));
708    }
709    let expected_count = record_len.div_ceil(fragment_capacity);
710    if fragment_count != expected_count {
711        return Err(Error::InvalidFragment(
712            "fragment count does not match record length",
713        ));
714    }
715    let expected_len = if fragment_index + 1 == fragment_count {
716        record_len - fragment_index * fragment_capacity
717    } else {
718        fragment_capacity
719    };
720    if fragment_len != expected_len || FRAGMENT_HEADER_LEN + fragment_len > symbol.len() {
721        return Err(Error::InvalidFragment(
722            "fragment length does not match geometry",
723        ));
724    }
725    if symbol[FRAGMENT_HEADER_LEN + fragment_len..]
726        .iter()
727        .any(|byte| *byte != 0)
728    {
729        return Err(Error::InvalidFragment("fragment padding is nonzero"));
730    }
731    Ok(Fragment {
732        kind,
733        object_id,
734        record_len,
735        fragment_index,
736        fragment_count,
737        fragment_capacity,
738        payload: &symbol[FRAGMENT_HEADER_LEN..FRAGMENT_HEADER_LEN + fragment_len],
739    })
740}
