132impl ContinuousRecoveryStats {
133    /// Percentage of expected systematic symbols that did not arrive intact.
134    pub const fn destruction_basis_points(self, expected_source_symbols: usize) -> u16 {
135        if expected_source_symbols == 0 {
136            return 0;
137        }
138        let destroyed = expected_source_symbols.saturating_sub(self.source_symbols_received);
139        ((destroyed as u64 * 10_000) / expected_source_symbols as u64) as u16
140    }
141
142    /// Percentage of missing systematic symbols reconstructed by FEC.
143    pub const fn source_recovery_basis_points(self, expected_source_symbols: usize) -> u16 {
144        let missing = expected_source_symbols.saturating_sub(self.source_symbols_received);
145        if missing == 0 {
146            return 10_000;
147        }
148        let recovered = if self.source_symbols_recovered < missing {
149            self.source_symbols_recovered
150        } else {
151            missing
152        };
153        ((recovered as u64 * 10_000) / missing as u64) as u16
154    }
155
156    /// Percentage of expected semantic records reconstructed and digest-verified.
157    pub const fn semantic_recovery_basis_points(self, expected_records: usize) -> u16 {
158        if expected_records == 0 {
159            return 0;
160        }
161        let recovered = if self.records_recovered < expected_records {
162            self.records_recovered
163        } else {
164            expected_records
165        };
166        ((recovered as u64 * 10_000) / expected_records as u64) as u16
167    }

244    pub(crate) fn push_record_with_repairs(
245        &mut self,
246        record: &[u8],
247        datagram: &mut [u8],
248        emit: &mut impl FnMut(&[u8]) -> Result<()>,
249        repair_schedule: &mut RepairSchedule,
250    ) -> Result<(usize, usize)> {

252    }

254    fn push_record_inner(
255        &mut self,
256        record: &[u8],
257        datagram: &mut [u8],
258        emit: &mut impl FnMut(&[u8]) -> Result<()>,
259        mut repair_schedule: Option<&mut RepairSchedule>,
260    ) -> Result<(usize, usize)> {
261        let decoded = decode_record(record)?;
262        if decoded.kind != RecordKind::CopperList {
263            return Err(Error::InconsistentObject);
264        }
265        if record.len() > self.max_record_bytes {
266            return Err(Error::ObjectTooLarge {
267                actual: record.len() as u64,
268                maximum: self.max_record_bytes as u64,
269            });
270        }
271        let record_len = u32::try_from(record.len())
272            .map_err(|_| Error::InvalidConfig("record length exceeds u32"))?;
273        let fragment_capacity = fragment_capacity(self.config())?;
274        let fragment_count = record.len().div_ceil(fragment_capacity);
275        let fragment_count = u32::try_from(fragment_count)
276            .map_err(|_| Error::InvalidConfig("fragment count exceeds u32"))?;
277        let mut symbol = [0_u8; MAX_SYMBOL_SIZE];
278        let mut repair_symbols = 0usize;
279
280        for (fragment_index, chunk) in record.chunks(fragment_capacity).enumerate() {
281            let active_symbol = &mut symbol[..self.config().symbol_size()];
282            active_symbol.fill(0);
283            encode_fragment_header(
284                active_symbol,
285                decoded.kind,
286                decoded.object_id,
287                record_len,
288                fragment_index as u32,
289                fragment_count,
290                chunk.len() as u16,
291            );
292            active_symbol[FRAGMENT_HEADER_LEN..FRAGMENT_HEADER_LEN + chunk.len()]
293                .copy_from_slice(chunk);
294            let esi = self.fec.push_source(active_symbol)?;
295            let mut fec_metadata = [0_u8; 12];
296            fec_metadata[..4].copy_from_slice(&SourcePayloadId::new(esi).to_bytes());
297            let header = self.packet_header(
298                decoded.kind,
299                decoded.object_id,
300                fragment_count,
301                FecSymbolKind::Source,
302                fec_metadata,
303            );
304            self.emit_packet_with(header, active_symbol, datagram, emit)?;
305            if let Some(schedule) = repair_schedule.as_mut() {
306                schedule.source_symbols_since_repair =
307                    schedule.source_symbols_since_repair.saturating_add(1);
308                if schedule.source_symbols_since_repair >= schedule.every {
309                    schedule.source_symbols_since_repair -= schedule.every;
310                    self.push_repair_with(
311                        RepairParameters::new(schedule.next_repair_key, schedule.density),
312                        datagram,
313                        &mut *emit,
314                    )?;
315                    schedule.next_repair_key = schedule.next_repair_key.wrapping_add(1);
316                    repair_symbols = repair_symbols.saturating_add(1);
317                }
318            }
319        }
320        Ok((fragment_count as usize, repair_symbols))
321    }

336    /// Emits one repair packet immediately from reusable caller-owned storage.
337    pub fn push_repair_with(
338        &mut self,
339        parameters: RepairParameters,
340        datagram: &mut [u8],
341        mut emit: impl FnMut(&[u8]) -> Result<()>,
342    ) -> Result<RepairPayloadId> {
343        let mut symbol = [0_u8; MAX_SYMBOL_SIZE];
344        let active_symbol = &mut symbol[..self.config().symbol_size()];
345        let id = self.fec.encode_repair(parameters, active_symbol)?;
346        let mut fec_metadata = [0_u8; 12];
347        fec_metadata[..8].copy_from_slice(&id.to_bytes());
348        let header = self.packet_header(
349            RecordKind::CopperList,
350            0,
351            0,
352            FecSymbolKind::Repair,
353            fec_metadata,
354        );
355        self.emit_packet_with(header, active_symbol, datagram, &mut emit)?;
356        Ok(id)
357    }

384        payload: &[u8],
385        datagram: &mut [u8],

437impl<const MAX_SYMBOL_SIZE: usize, const MAX_WINDOW_SYMBOLS: usize, const MAX_EQUATIONS: usize>

441        identity: StreamIdentity,
442        lane: Lane,

463            observed_window_base: None,
464            retention_floor: None,
465            stats: ContinuousRecoveryStats::default(),

481    /// Retries delivery of ordered records or gaps already ready in receiver state.
482    ///
483    /// This is useful after a consumer error and when shutting down a transport
484    /// after its final datagram. A consumer failure leaves the current event
485    /// pending so a later call can retry it.
486    pub fn drain_events<E>(
487        &mut self,
488        mut emit: impl FnMut(ContinuousReceiveEvent<'_>) -> core::result::Result<(), E>,
489    ) -> core::result::Result<(), ReceiveError<E>> {
490        self.expire_and_emit(&mut emit)
491    }

493    /// Finalizes ordered delivery through an inclusive CopperList identifier.
494    ///
495    /// Any unresolved record or wholly absent identifier through `last_object_id`
496    /// is emitted as a terminal gap. Calling this method is an explicit promise
497    /// that no more useful repair symbols for that range will arrive.
498    pub fn finish_through<E>(
499        &mut self,
500        last_object_id: u64,
501        mut emit: impl FnMut(ContinuousReceiveEvent<'_>) -> core::result::Result<(), E>,
502    ) -> core::result::Result<(), ReceiveError<E>> {
503        self.process_new_symbols()?;
504        self.emit_through(last_object_id, GapReason::SessionEnded, &mut emit)?;
505        let unresolved = self
506            .assemblies
507            .iter()
508            .filter(|assembly| assembly.object_id <= last_object_id)
509            .count();
510        self.assemblies
511            .retain(|assembly| assembly.object_id > last_object_id);
512        self.stats.records_expired = self.stats.records_expired.saturating_add(unresolved);
513        self.discard_stale_ready();
514        Ok(())
515    }

517    /// Validates and submits one datagram. Corrupt or foreign packets are counted
518    /// and ignored before they can contaminate the FEC decoder. Recovered records
519    /// and definitive gaps are emitted in CopperList-id order.
520    pub fn receive_datagram<E>(
521        &mut self,
522        datagram: &[u8],
523        mut emit: impl FnMut(ContinuousReceiveEvent<'_>) -> core::result::Result<(), E>,
524    ) -> core::result::Result<(), ReceiveError<E>> {
525        self.stats.datagrams_seen = self.stats.datagrams_seen.saturating_add(1);
526        let packet = match WirePacket::decode(datagram) {
527            Ok(packet) => packet,
528            Err(_) => {
529                self.stats.invalid_datagrams = self.stats.invalid_datagrams.saturating_add(1);
530                return Ok(());
531            }
532        };
533        if packet.header.session_id != self.identity.session_id
534            || packet.header.sender_id != self.identity.sender_id
535            || packet.header.lane != self.lane
536            || packet.header.fec_scheme != fec_scheme(self.fec.config().field())
537            || packet.header.record_kind != RecordKind::CopperList
538        {
539            self.stats.inconsistent_datagrams = self.stats.inconsistent_datagrams.saturating_add(1);
540            return Ok(());
541        }
542        if packet.payload.len() != self.fec.config().symbol_size() {
543            self.stats.invalid_datagrams = self.stats.invalid_datagrams.saturating_add(1);
544            return Ok(());
545        }
546
547        let accepted = match packet.header.symbol_kind {
548            FecSymbolKind::Source => self.receive_source_packet(&packet)?,
549            FecSymbolKind::Repair => self.receive_repair_packet(&packet)?,
550        };
551        if !accepted {
552            return Ok(());
553        }
554        self.stats.valid_datagrams = self.stats.valid_datagrams.saturating_add(1);
555        self.update_retention_floor();
556        self.expire_and_emit(&mut emit)?;
557        self.process_new_symbols()?;
558        self.expire_and_emit(&mut emit)
559    }

561    fn receive_source_packet(&mut self, packet: &WirePacket) -> Result<bool> {
562        if packet.header.fec_metadata[4..]
563            .iter()
564            .any(|byte| *byte != 0)
565        {
566            self.stats.invalid_datagrams = self.stats.invalid_datagrams.saturating_add(1);
567            return Ok(false);
568        }
569        let id = SourcePayloadId::from_bytes(packet.header.fec_metadata[..4].try_into().unwrap());
570        let fragment = match decode_fragment(&packet.payload, fragment_capacity(self.fec.config())?)
571        {
572            Ok(fragment) => fragment,
573            Err(_) => {
574                self.stats.invalid_datagrams = self.stats.invalid_datagrams.saturating_add(1);
575                return Ok(false);
576            }
577        };
578        if packet.header.object_id != fragment.object_id
579            || packet.header.fragment_count as usize != fragment.fragment_count
580        {
581            self.stats.inconsistent_datagrams = self.stats.inconsistent_datagrams.saturating_add(1);
582            return Ok(false);
583        }
584        let report = match self.fec.receive_source(id.esi(), &packet.payload) {
585            Ok(report) => report,
586            Err(cu_fec::Error::WindowExpired) => {
587                self.stats.expired_datagrams = self.stats.expired_datagrams.saturating_add(1);
588                return Ok(false);
589            }
590            Err(error) => return Err(error.into()),
591        };
592        if report.status() == SourceStatus::Duplicate {
593            self.stats.duplicate_symbols = self.stats.duplicate_symbols.saturating_add(1);
594            return Ok(false);
595        }
596        self.stats.source_symbols_received = self.stats.source_symbols_received.saturating_add(1);
597        self.stats.source_symbols_recovered = self
598            .stats
599            .source_symbols_recovered
600            .saturating_add(report.recovered());
601        Ok(true)
602    }

604    fn receive_repair_packet(&mut self, packet: &WirePacket) -> Result<bool> {
605        if packet.header.object_id != 0
606            || packet.header.fragment_count != 0
607            || packet.header.fec_metadata[8..]
608                .iter()
609                .any(|byte| *byte != 0)
610        {
611            self.stats.invalid_datagrams = self.stats.invalid_datagrams.saturating_add(1);
612            return Ok(false);
613        }
614        let raw_id: [u8; 8] = packet.header.fec_metadata[..8].try_into().unwrap();
615        let id = RepairPayloadId::from_bytes(raw_id)?;
616        let report = match self.fec.receive_repair(id, &packet.payload) {
617            Ok(report) => report,
618            Err(cu_fec::Error::WindowExpired) => {
619                self.stats.expired_datagrams = self.stats.expired_datagrams.saturating_add(1);
620                return Ok(false);
621            }
622            Err(error) => return Err(error.into()),
623        };
624        self.stats.repair_symbols_received = self.stats.repair_symbols_received.saturating_add(1);
625        if report.is_innovative() {
626            self.stats.innovative_repairs = self.stats.innovative_repairs.saturating_add(1);
627        } else {
628            self.stats.redundant_repairs = self.stats.redundant_repairs.saturating_add(1);
629        }
630        self.stats.source_symbols_recovered = self
631            .stats
632            .source_symbols_recovered
633            .saturating_add(report.recovered());
634        Ok(true)
635    }
636
637    fn process_new_symbols(&mut self) -> Result<()> {
638        while let Some(esi) = self
639            .fec
640            .known_symbols()
641            .map(|(esi, _)| esi)
642            .find(|esi| !self.processed_symbols.contains(esi))
643        {
644            let symbol = self
645                .fec
646                .symbol(esi)
647                .expect("known symbol disappeared from the retained RLC window")
648                .to_vec();
649            self.process_fragment(esi, &symbol)?;
650            self.processed_symbols.push(esi);
651        }
652        Ok(())
653    }
654
655    fn process_fragment(&mut self, esi: EncodingSymbolId, symbol: &[u8]) -> Result<()> {
656        let fragment = decode_fragment(symbol, fragment_capacity(self.fec.config())?)?;
657        if fragment.record_len > self.limits.max_record_bytes {
658            return Err(Error::ObjectTooLarge {
659                actual: fragment.record_len as u64,
660                maximum: self.limits.max_record_bytes as u64,
661            });
662        }
663        let assembly_index = match self.assemblies.iter().position(|assembly| {
664            assembly.kind == fragment.kind && assembly.object_id == fragment.object_id
665        }) {
666            Some(index) => index,
667            None => {
668                let count = self
669                    .assemblies
670                    .len()
671                    .saturating_add(self.ready.len())
672                    .saturating_add(1);
673                if count > self.limits.max_buffered_records {
674                    return Err(Error::TooManyRecords {
675                        actual: count,
676                        maximum: self.limits.max_buffered_records,
677                    });
678                }
679                self.assemblies.push(RecordAssembly::new(esi, &fragment));
680                self.observe_peak_buffered_records();
681                self.assemblies.len() - 1
682            }
683        };
684        let complete = self.assemblies[assembly_index].insert(esi, &fragment)?;
685        if !complete {
686            return Ok(());
687        }
688
689        let assembly = self.assemblies.swap_remove(assembly_index);
690        let decoded = decode_record(&assembly.bytes)?;
691        if decoded.kind != assembly.kind || decoded.object_id != assembly.object_id {
692            return Err(Error::InconsistentObject);
693        }
694        self.ready.push(ReadyRecord {
695            object_id: assembly.object_id,
696            first_esi: assembly.first_esi,
697            record: RecoveredRecord {
698                bytes: assembly.bytes,
699            },
700        });
701        self.stats.records_recovered = self.stats.records_recovered.saturating_add(1);
702        self.observe_peak_buffered_records();
703        Ok(())
704    }
705
706    fn expire_and_emit<E>(
707        &mut self,
708        emit: &mut impl FnMut(ContinuousReceiveEvent<'_>) -> core::result::Result<(), E>,
709    ) -> core::result::Result<(), ReceiveError<E>> {
710        let Some((base_esi, _)) = self.fec.window() else {
711            return Ok(());
712        };
713        self.processed_symbols
714            .retain(|esi| !esi_is_before(*esi, base_esi));
715
716        if let Some(retention_floor) = self.retention_floor {
717            loop {
718                let expired = self
719                    .assemblies
720                    .iter()
721                    .enumerate()
722                    .filter(|(_, assembly)| assembly.has_expired_fragment(retention_floor))
723                    .min_by_key(|(_, assembly)| assembly.object_id)
724                    .map(|(index, _)| index);
725                let Some(index) = expired else {
726                    break;
727                };
728                let object_id = self.assemblies[index].object_id;
729                self.emit_through(object_id, GapReason::RlcWindowExpired, emit)?;
730                self.assemblies.swap_remove(index);
731                self.stats.records_expired = self.stats.records_expired.saturating_add(1);
732            }
733        }
734
735        self.emit_ready(self.retention_floor, emit)
736    }
737
738    fn emit_ready<E>(
739        &mut self,
740        retention_floor: Option<EncodingSymbolId>,
741        emit: &mut impl FnMut(ContinuousReceiveEvent<'_>) -> core::result::Result<(), E>,
742    ) -> core::result::Result<(), ReceiveError<E>> {
743        loop {
744            self.discard_stale_ready();
745            if self.emit_next_ready(emit)? {
746                continue;
747            }
748
749            let Some(retention_floor) = retention_floor else {
750                return Ok(());
751            };
752            let expired_boundary = self
753                .ready
754                .iter()
755                .filter(|record| {
756                    record.object_id > self.next_object_id
757                        && !esi_is_after(record.first_esi, retention_floor)
758                })
759                .min_by_key(|record| record.object_id)
760                .map(|record| record.object_id);
761            let Some(boundary) = expired_boundary else {
762                return Ok(());
763            };
764            self.emit_gap(boundary - 1, GapReason::RlcWindowExpired, emit)?;
765        }
766    }
767
768    fn emit_through<E>(
769        &mut self,
770        last_id: u64,
771        reason: GapReason,
772        emit: &mut impl FnMut(ContinuousReceiveEvent<'_>) -> core::result::Result<(), E>,
773    ) -> core::result::Result<(), ReceiveError<E>> {
774        while self.next_object_id <= last_id {
775            self.discard_stale_ready();
776            if self.emit_next_ready(emit)? {
777                continue;
778            }
779            let next_ready = self
780                .ready
781                .iter()
782                .map(|record| record.object_id)
783                .filter(|object_id| *object_id <= last_id)
784                .min();
785            let gap_last = next_ready.map_or(last_id, |object_id| object_id - 1);
786            self.emit_gap(gap_last, reason, emit)?;
787        }
788        Ok(())
789    }
790
791    fn emit_next_ready<E>(
792        &mut self,
793        emit: &mut impl FnMut(ContinuousReceiveEvent<'_>) -> core::result::Result<(), E>,
794    ) -> core::result::Result<bool, ReceiveError<E>> {
795        let Some(index) = self
796            .ready
797            .iter()
798            .position(|record| record.object_id == self.next_object_id)
799        else {
800            return Ok(false);
801        };
802        emit(ContinuousReceiveEvent::Record(&self.ready[index].record))
803            .map_err(ReceiveError::Consumer)?;
804        self.ready.swap_remove(index);
805        self.stats.records_emitted = self.stats.records_emitted.saturating_add(1);
806        self.next_object_id = self.next_object_id.wrapping_add(1);
807        Ok(true)
808    }
809
810    fn emit_gap<E>(
811        &mut self,
812        last_id: u64,
813        reason: GapReason,
814        emit: &mut impl FnMut(ContinuousReceiveEvent<'_>) -> core::result::Result<(), E>,
815    ) -> core::result::Result<(), ReceiveError<E>> {
816        if last_id < self.next_object_id {
817            return Ok(());
818        }
819        let first_id = self.next_object_id;
820        emit(ContinuousReceiveEvent::Gap(CopperListGap {
821            first_id,
822            last_id,
823            reason,
824        }))
825        .map_err(ReceiveError::Consumer)?;
826        self.stats.gaps_emitted = self.stats.gaps_emitted.saturating_add(1);
827        self.stats.missing_copperlists = self
828            .stats
829            .missing_copperlists
830            .saturating_add(last_id - first_id + 1);
831        self.next_object_id = last_id.wrapping_add(1);
832        Ok(())
833    }
834
835    fn discard_stale_ready(&mut self) {
836        self.ready
837            .retain(|record| record.object_id >= self.next_object_id);
838    }
839
840    fn observe_peak_buffered_records(&mut self) {
841        self.stats.peak_buffered_records = self
842            .stats
843            .peak_buffered_records
844            .max(self.assemblies.len().saturating_add(self.ready.len()));
845    }
846
847    fn update_retention_floor(&mut self) {
848        let Some((base_esi, _)) = self.fec.window() else {
849            return;
850        };
851        if self
852            .observed_window_base
853            .is_some_and(|previous| esi_is_after(base_esi, previous))
854        {
855            self.retention_floor = Some(base_esi);
856        }
857        self.observed_window_base = Some(base_esi);
858    }
859}

868#[derive(Debug)]
869struct Fragment<'a> {
870    kind: RecordKind,
871    object_id: u64,
872    record_len: usize,
873    fragment_index: usize,
874    fragment_count: usize,
875    fragment_capacity: usize,
876    payload: &'a [u8],
877}
878
879struct RecordAssembly {
880    kind: RecordKind,
881    object_id: u64,
882    first_esi: EncodingSymbolId,
883    bytes: Vec<u8>,
884    received: Vec<bool>,
885    received_count: usize,
886    fragment_capacity: usize,
887}
888
889impl RecordAssembly {
890    fn new(esi: EncodingSymbolId, fragment: &Fragment<'_>) -> Self {
891        Self {
892            kind: fragment.kind,
893            object_id: fragment.object_id,
894            first_esi: esi.wrapping_sub(fragment.fragment_index as u32),
895            bytes: vec![0; fragment.record_len],
896            received: vec![false; fragment.fragment_count],
897            received_count: 0,
898            fragment_capacity: fragment.fragment_capacity,
899        }
900    }
901
902    fn insert(&mut self, esi: EncodingSymbolId, fragment: &Fragment<'_>) -> Result<bool> {
903        if self.bytes.len() != fragment.record_len || self.received.len() != fragment.fragment_count
904        {
905            return Err(Error::InvalidFragment("record geometry changed"));
906        }
907        if self.first_esi.wrapping_add(fragment.fragment_index as u32) != esi {
908            return Err(Error::InvalidFragment(
909                "fragment ESI does not match record geometry",
910            ));
911        }
912        let offset = fragment
913            .fragment_index
914            .checked_mul(self.fragment_capacity)
915            .ok_or(Error::InvalidFragment("fragment offset overflow"))?;
916        let end = offset
917            .checked_add(fragment.payload.len())
918            .ok_or(Error::InvalidFragment("fragment length overflow"))?;
919        if end > self.bytes.len() {
920            return Err(Error::InvalidFragment("fragment exceeds record"));
921        }
922        if self.received[fragment.fragment_index] {
923            if self.bytes[offset..end] != *fragment.payload {
924                return Err(Error::InvalidFragment("duplicate fragment differs"));
925            }
926            return Ok(self.received_count == self.received.len());
927        }
928        self.bytes[offset..end].copy_from_slice(fragment.payload);
929        self.received[fragment.fragment_index] = true;
930        self.received_count += 1;
931        Ok(self.received_count == self.received.len())
932    }
933
934    fn has_expired_fragment(&self, base_esi: EncodingSymbolId) -> bool {
935        self.received.iter().enumerate().any(|(index, received)| {
936            !received && esi_is_before(self.first_esi.wrapping_add(index as u32), base_esi)
937        })
938    }
939}
940
941fn esi_is_before(candidate: EncodingSymbolId, reference: EncodingSymbolId) -> bool {
942    (candidate.get().wrapping_sub(reference.get()) as i32) < 0
943}
944
945fn esi_is_after(candidate: EncodingSymbolId, reference: EncodingSymbolId) -> bool {
946    (candidate.get().wrapping_sub(reference.get()) as i32) > 0
947}

949fn validate_config(config: RlcConfig, max_record_bytes: usize) -> Result<()> {
950    let _ = fragment_capacity(config)?;
951    if max_record_bytes == 0 {
952        return Err(Error::InvalidConfig("maximum record size must be nonzero"));
953    }
954    if max_record_bytes > u32::MAX as usize {
955        return Err(Error::InvalidConfig("maximum record size exceeds u32"));
956    }
957    Ok(())
958}
959
960fn fragment_capacity(config: RlcConfig) -> Result<usize> {
961    config
962        .symbol_size()
963        .checked_sub(FRAGMENT_HEADER_LEN)
964        .filter(|capacity| *capacity > 0 && *capacity <= u16::MAX as usize)
965        .ok_or(Error::InvalidConfig(
966            "RLC symbol cannot hold the fragment header and payload",
967        ))
968}

977fn encode_fragment_header(
978    symbol: &mut [u8],
979    kind: RecordKind,
980    object_id: u64,
981    record_len: u32,
982    fragment_index: u32,
983    fragment_count: u32,
984    fragment_len: u16,
985) {
986    symbol[..4].copy_from_slice(&FRAGMENT_MAGIC);
987    symbol[4] = FRAGMENT_VERSION;
988    symbol[5] = kind as u8;
989    symbol[6..8].fill(0);
990    symbol[8..16].copy_from_slice(&object_id.to_be_bytes());
991    symbol[16..20].copy_from_slice(&record_len.to_be_bytes());
992    symbol[20..24].copy_from_slice(&fragment_index.to_be_bytes());
993    symbol[24..28].copy_from_slice(&fragment_count.to_be_bytes());
994    symbol[28..30].copy_from_slice(&fragment_len.to_be_bytes());
995    symbol[30..32].fill(0);
996}
997
998fn decode_fragment(symbol: &[u8], fragment_capacity: usize) -> Result<Fragment<'_>> {
999    if symbol.len() < FRAGMENT_HEADER_LEN || symbol[..4] != FRAGMENT_MAGIC {
1000        return Err(Error::InvalidFragment("missing fragment header"));
1001    }
1002    if symbol[4] != FRAGMENT_VERSION {
1003        return Err(Error::UnsupportedVersion(symbol[4]));
1004    }
1005    if symbol[6..8].iter().any(|byte| *byte != 0) || symbol[30..32].iter().any(|byte| *byte != 0) {
1006        return Err(Error::InvalidFragment("reserved bytes are nonzero"));
1007    }
1008    let kind = RecordKind::try_from(symbol[5])?;
1009    if kind != RecordKind::CopperList {
1010        return Err(Error::InvalidFragment(
1011            "continuous lane requires CopperList records",
1012        ));
1013    }
1014    let object_id = u64::from_be_bytes(symbol[8..16].try_into().unwrap());
1015    let record_len = u32::from_be_bytes(symbol[16..20].try_into().unwrap()) as usize;
1016    let fragment_index = u32::from_be_bytes(symbol[20..24].try_into().unwrap()) as usize;
1017    let fragment_count = u32::from_be_bytes(symbol[24..28].try_into().unwrap()) as usize;
1018    let fragment_len = u16::from_be_bytes(symbol[28..30].try_into().unwrap()) as usize;
1019    if record_len == 0 || fragment_count == 0 || fragment_index >= fragment_count {
1020        return Err(Error::InvalidFragment("invalid fragment geometry"));
1021    }
1022    let expected_count = record_len.div_ceil(fragment_capacity);
1023    if fragment_count != expected_count {
1024        return Err(Error::InvalidFragment(
1025            "fragment count does not match record length",
1026        ));
1027    }
1028    let expected_len = if fragment_index + 1 == fragment_count {
1029        record_len - fragment_index * fragment_capacity
1030    } else {
1031        fragment_capacity
1032    };
1033    if fragment_len != expected_len || FRAGMENT_HEADER_LEN + fragment_len > symbol.len() {
1034        return Err(Error::InvalidFragment(
1035            "fragment length does not match geometry",
1036        ));
1037    }
1038    if symbol[FRAGMENT_HEADER_LEN + fragment_len..]
1039        .iter()
1040        .any(|byte| *byte != 0)
1041    {
1042        return Err(Error::InvalidFragment("fragment padding is nonzero"));
1043    }
1044    Ok(Fragment {
1045        kind,
1046        object_id,
1047        record_len,
1048        fragment_index,
1049        fragment_count,
1050        fragment_capacity,
1051        payload: &symbol[FRAGMENT_HEADER_LEN..FRAGMENT_HEADER_LEN + fragment_len],
1052    })
1053}
