61/// routing, archive, and recovery plumbing; callers supply a packet transport.

90    /// Sets the size of each archive backing file (slab), in bytes. Defaults to 16 MiB.
91    ///
92    /// The archive adds another slab when the current one fills; this is not a
93    /// limit on total recording size. A slab must fit the largest section used
94    /// to store CopperLists, keyframes, or structured logs. Prefer a slab roughly
95    /// 10–100 times larger than a section to hold many sections per file and
96    /// reduce file turnover. Larger slabs reserve more disk space at a time.
97    /// Both slab and section sizes must be multiples of 512 bytes.
98    pub fn with_slab_size(mut self, bytes: usize) -> Self {
99        self.slab_bytes = bytes;
100        self
101    }
102
103    /// Sets the size of an archive section, in bytes. Defaults to 128 KiB.
104    ///
105    /// Sections group records such as CopperLists, keyframes, and structured log
106    /// entries inside a slab. A complete record must fit within one section.
107    /// Choose at least the larger of `max_record_bytes` and
108    /// `finite_objects.max_object_bytes` from [`Self::with_receiver_limits`],
109    /// plus 512 bytes for the section header and 32 bytes for the archive's
110    /// continuity envelope. Round up to a multiple of 512 bytes.
111    ///
112    /// Increase this alongside receive bounds when captures or task-state
113    /// snapshots grow. The section must fit in [`Self::with_slab_size`]; keeping
114    /// slabs much larger lets each file hold many sections.
115    pub fn with_section_size(mut self, bytes: usize) -> Self {
116        self.section_bytes = bytes;
117        self
118    }
119
120    /// Sets receive size and buffering limits for one sender session.
121    /// Defaults to [`SessionRouterLimits::default`].
122    ///
123    /// `max_record_bytes` bounds one complete serialized CopperList record
124    /// (default: 4 KiB). `finite_objects.max_object_bytes` bounds one manifest,
125    /// keyframe, or structured log record (default: 64 KiB). Match or exceed
126    /// the sender's advertised bounds so its captures and recovery snapshots
127    /// can be received, and increase [`Self::with_section_size`] to fit them.
128    ///
129    /// The remaining limits bound records, packets, recovery data, and objects
130    /// buffered during decoding. Larger sizes and concurrency limits increase
131    /// receiver memory use; they are not a total process memory budget.
132    /// `max_sessions` must be 1. The packet profile remains a 1200-byte MTU with
133    /// at most 1128-byte symbols and 64 FEC equations.
134    pub fn with_receiver_limits(mut self, limits: SessionRouterLimits) -> Self {
135        self.receiver_limits = limits;
136        self
137    }
138
139    /// Sets buffering between recording and task replay. Defaults to 32.
140    ///
141    /// This bounds both the queued replay events (captured CopperLists and
142    /// recovery keyframes) and the pending CopperLists waiting for matching
143    /// replay state. One recovery keyframe and one executing CopperList are
144    /// retained in addition to these buffers.
145    ///
146    /// Increase this to absorb short replay slowdowns; reduce it to retain fewer
147    /// payloads in memory. Overflow interrupts reconstruction until a matching
148    /// recovery point is available. Received data continues to be recorded in
149    /// the archive. This setting is unused with [`Self::archive_only`].
150    pub fn with_replay_capacity(mut self, capacity: NonZeroUsize) -> Self {
151        self.replay_capacity = capacity;
152        self
153    }
154
155    /// Sets the number of unread structured log entries buffered for
156    /// [`CuTwin::take_log_reader`]. Defaults to 64 entries.
157    ///
158    /// These are the robot's structured logging calls (such as `info!` and
159    /// `debug!`), published after being saved in the archive. Increase this to
160    /// let the log reader pause longer at the cost of retaining more entries in
161    /// memory. When full, the buffer replaces the oldest unread entry. Reading
162    /// slowly never blocks recording; overwritten entries remain in the archive.
163    pub fn with_log_capacity(mut self, capacity: NonZeroUsize) -> Self {
164        self.log_capacity = capacity;
165        self
166    }
167
168    /// Sets the number of unread reconstructed CopperLists buffered for the
169    /// returned [`CuTwinReader`]. Defaults to 64.
170    ///
171    /// Each [`TwinFrame`] contains one CopperList with its captured inputs and
172    /// locally reconstructed outputs, plus sender identity and receive time.
173    /// Increase this to absorb pauses in a UI or analysis reader; reduce it to
174    /// retain fewer message payloads in memory. When full, the buffer replaces
175    /// the oldest unread CopperList and the reader reports how many it missed.
176    /// Recording and task replay continue independently of reader consumption.
177    /// Network packets and recovery keyframes have separate receive/replay
178    /// buffers; this capacity counts completed CopperLists.
179    pub fn with_frame_capacity(mut self, capacity: NonZeroUsize) -> Self {
180        self.frame_capacity = capacity;
181        self
182    }

184    /// Record the native capture without starting task reconstruction.
185    pub fn archive_only(mut self) -> Self {
186        self.reconstruct = false;
187        self
188    }

190    pub fn spawn(self) -> CuResult<(CuTwin<A>, CuTwinReader<A>)> {
191        validate_sizes(self.slab_bytes, self.section_bytes, self.receiver_limits)?;
192        // Validate and construct routing before creating directories or starting workers.
193        let mut router =
194            SessionRouter::<1128, 64, 64>::new(self.receiver_limits).map_err(stream_error)?;
195        let path = self
196            .log_base
197            .ok_or_else(|| CuError::from("Copper twin requires a log path"))?;
198        if cu29_runtime::replay::first_slab_path(&path)?.exists() || path.exists() {
199            return Err(CuError::from(
200                "Copper twin archive already exists; choose a fresh log path",
201            ));
202        }
203        if let Some(parent) = path
204            .parent()
205            .filter(|parent| !parent.as_os_str().is_empty())
206        {
207            std::fs::create_dir_all(parent)
208                .map_err(|e| CuError::new_with_cause("Create twin log directory", e))?;
209        }
210        let (publisher, reader) = telemetry_channel(self.frame_capacity, CuTwinStatus::default());
211        let (mut log_publisher, log_reader) = telemetry_channel(self.log_capacity, ());
212        let stop = Arc::new(AtomicBool::new(false));
213        let stopping = stop.clone();
214        let (ready_tx, ready_rx) = std::sync::mpsc::sync_channel(1);
215        let worker = thread::Builder::new()
216            .name("copper-twin-receiver".into())
217            .spawn(move || {
218                let mut status = CuTwinStatus::default();
219                let (replay, mut publisher) = if self.reconstruct {
220                    match TwinWorker::spawn::<A>(self.replay_capacity, publisher) {
221                        Ok(replay) => (Some(replay), None),
222                        Err(error) => {
223                            let _ = ready_tx.send(Err(error.clone()));
224                            return Err(error);
225                        }
226                    }
227                } else {
228                    (None, Some(publisher))
229                };
230                let mut publish_status = |status| {
231                    if let Some(replay) = &replay {
232                        replay.set_status(status);
233                    }
234                    if let Some(publisher) = &mut publisher {
235                        publisher.set_status(status);
236                    }
237                };
238                let mut archive = None;
239                let result = (|| {
240                    let _ = ready_tx.send(Ok(()));
241                    let mut rx = self.rx;
242                    let mut feedback_tx = self.feedback_tx;
243                    let feedback_clock = cu29_clock::RobotClock::new();
244                    let receiver_id = crate::new_session_id();
245                    let mut reporter = None;
246                    let mut feedback_packet = [0; crate::feedback::FEEDBACK_BUFFER_BYTES];
247                    let mut packet = [0; 1200];
248                    while !stopping.load(Ordering::Acquire) {
249                        if let Some(len) = rx
250                            .try_recv(&mut packet)
251                            .map_err(|e| CuError::from(format!("Twin receive: {e:?}")))?
252                        {
253                            let bytes = packet.get(..len).ok_or_else(|| {
254                                CuError::from("Transport returned an invalid packet length")
255                            })?;
256                            status.last_packet = Some(Instant::now());
257                            router
258                                .receive_datagram(bytes, |event| {
259                                    // Recovery objects can arrive before the manifest.
260                                    // The router retains them and emits VerifiedRecoveryPoint
261                                    // once their manifest/keyframe references agree.
262                                    if matches!(&event, SessionEvent::Object { record, .. } if record.decoded().kind != crate::RecordKind::StructuredLog) {
263                                        return Ok(());
264                                    }
265                                    if let SessionEvent::Manifest(manifest) = &event {
266                                        status.identity = Some(manifest.manifest().identity);
267                                        if feedback_tx.is_some() {
268                                            reporter = crate::feedback::FeedbackReporter::new(
269                                                manifest.manifest(),
270                                                receiver_id,
271                                                feedback_clock.now(),
272                                            );
273                                        }
274                                        archive = Some(CaptureArchive::<A::DataSet>::new(
275                                            &path,
276                                            manifest,
277                                            self.slab_bytes,
278                                            self.section_bytes,
279                                        )?);
280                                    }
281                                    let writer =
282                                        archive.as_mut().ok_or(crate::Error::InvalidConfig(
283                                            "capture arrived before its manifest",
284                                        ))?;
285                                    let capture = writer.accept(&event)?;
286                                    if let Some(entry) = writer.take_structured_log() {
287                                        status.structured_logs += 1;
288                                        if let SessionEvent::Object { identity, record } = &event {
289                                            log_publisher.publish(ReceivedStructuredLog {
290                                                identity: *identity, sequence: record.decoded().object_id, entry,
291                                            });
292                                        }
293                                    }
294                                    if let Some(capture) = &capture {
295                                        status.latest = Some(capture.copperlist.id);
296                                        status.archived += 1;
297                                        status.state = CuTwinRecordingState::Recording;
298                                    }
299                                    if let Some(replay) = &replay {
300                                        replay.accept(&event, capture)?;
301                                    }
302                                    match event {
303                                        SessionEvent::Gap { .. } => status.gaps += 1,
304                                        SessionEvent::VerifiedRecoveryPoint {
305                                            recovery_point,
306                                            ..
307                                        } => {
308                                            status.recovery_point =
309                                                Some(recovery_point.copperlist_id)
310                                        }
311                                        _ => {}
312                                    }
313                                    Ok::<(), crate::Error>(())
314                                })
315                                .map_err(|e| CuError::from(format!("Twin receive: {e}")))?;
316                            status.packets = router.stats().datagrams_seen;
317                            publish_status(status);
318                        } else {
319                            thread::park_timeout(POLL_INTERVAL);
320                        }
321                        if let (Some(reporter), Some(tx), Some(identity)) =
322                            (&mut reporter, &mut feedback_tx, status.identity)
323                            && let Some(mut counters) = router.feedback_counters(identity)
324                        {
325                            counters.latest_copperlist = status.latest;
326                            if let Some(report) = reporter.report(feedback_clock.now(), counters) {
327                                let len = report
328                                    .encode_into(&mut feedback_packet)
329                                    .map_err(stream_error)?;
330                                match tx.try_send_feedback(&feedback_packet[..len]) {
331                                    Ok(()) => status.feedback_reports_sent += 1,
332                                    Err(crate::CuStreamTxError::WouldBlock) => {
333                                        status.feedback_reports_dropped += 1
334                                    }
335                                    Err(crate::CuStreamTxError::Failed(_)) => {
336                                        feedback_tx = None;
337                                        status.feedback_failed = true;
338                                    }
339                                }
340                                publish_status(status);
341                            }
342                        }
343                    }
344                    if let Some(archive) = archive.take() {
345                        archive.finish().map_err(stream_error)?;
346                    }
347                    Ok(())
348                })();
349                status.state = if result.is_ok() {
350                    CuTwinRecordingState::Closed
351                } else {
352                    CuTwinRecordingState::Failed
353                };
354                publish_status(status);
355                // Drain admitted replay before returning the final receiver status.
356                if let Some(replay) = replay {
357                    status.twin = replay.finish()?;
358                }
359                result.map(|()| status)
360            })
361            .map_err(|e| CuError::new_with_cause("Start Copper twin", e))?;
362        if let Err(error) = ready_rx
363            .recv()
364            .map_err(|_| CuError::from("Copper twin initialization failed"))
365            .and_then(|ready| ready)
366        {
367            stop.store(true, Ordering::Release);
368            let _ = worker.join();
369            return Err(error);
370        }
371        Ok((
372            CuTwin {
373                stop,
374                worker: Some(worker),
375                final_status: None,
376                log_reader: Some(log_reader),
377                app: PhantomData,
378            },
379            reader,
380        ))
381    }
382}

402fn validate_sizes(
403    slab_bytes: usize,
404    section_bytes: usize,
405    limits: SessionRouterLimits,
406) -> CuResult<()> {
407    let header = usize::from(cu29_unifiedlog::SECTION_HEADER_COMPACT_SIZE);
408    if slab_bytes < 1024 || !slab_bytes.is_multiple_of(header) {
409        return Err(CuError::from(
410            "Twin slab size must be at least 1024 bytes and a multiple of 512",
411        ));
412    }
413    if section_bytes <= header
414        || !section_bytes.is_multiple_of(header)
415        || section_bytes > slab_bytes
416    {
417        return Err(CuError::from(
418            "Twin section size must exceed 512 bytes, be a multiple of 512, and fit in its slab",
419        ));
420    }
421    if section_bytes - header > u32::MAX as usize {
422        return Err(CuError::from("Twin section payload size exceeds u32"));
423    }
424    if limits.max_sessions != 1 {
425        return Err(CuError::from(
426            "Twin receiver limits must allow exactly one sender session",
427        ));
428    }
429    if limits.equation_capacity > 64 || usize::from(limits.finite_objects.max_symbol_size) > 1128 {
430        return Err(CuError::from(
431            "Twin receiver limits exceed the 1128-byte-symbol, 64-equation FEC profile",
432        ));
433    }
434    crate::object::validate_limits(limits.finite_objects).map_err(stream_error)?;
435    if limits.max_record_bytes > u32::MAX as usize {
436        return Err(CuError::from("Twin maximum record size exceeds u32"));
437    }
438    let object_bytes = usize::try_from(limits.finite_objects.max_object_bytes)
439        .map_err(|_| CuError::from("Twin maximum object size exceeds usize"))?;
440    let required = limits
441        .max_record_bytes
442        .max(object_bytes)
443        .checked_add(CONTINUITY_OVERHEAD)
444        .and_then(|bytes| bytes.checked_add(header))
445        .ok_or_else(|| CuError::from("Twin section size requirement overflow"))?;
446    if section_bytes < required {
447        return Err(CuError::from(format!(
448            "Twin section size is {section_bytes} bytes; configured record/object bounds require at least {required} bytes including header and continuity envelope"
449        )));
450    }
451    Ok(())
452}

454/// Running Copper twin. The separately owned reader can be paused or dropped
455/// without affecting recording. `stop()` closes the archive and drains admitted
456/// replay; dropping the handle also stops and joins Copper's workers.
457pub struct CuTwin<A: LiveReplay> {
458    stop: Arc<AtomicBool>,
459    worker: Option<JoinHandle<CuResult<CuTwinStatus>>>,
460    final_status: Option<CuTwinStatus>,
461    log_reader: Option<CuTwinLogReader>,
462    app: PhantomData<fn() -> A>,
463}
464
465impl<A: LiveReplay> CuTwin<A> {
466    pub fn builder<R: CuStreamRx + 'static>(rx: R) -> CuTwinBuilder<A, R> {
467        CuTwinBuilder {
468            rx,
469            feedback_tx: None,
470            log_base: None,
471            frame_capacity: FRAME_CAPACITY,
472            log_capacity: FRAME_CAPACITY,
473            replay_capacity: REPLAY_CAPACITY,
474            slab_bytes: SLAB_BYTES,
475            section_bytes: SECTION_BYTES,
476            receiver_limits: SessionRouterLimits::default(),
477            reconstruct: true,
478            app: PhantomData,
479        }
480    }
481
482    /// Take the independent bounded log view once (64 entries by default).
483    /// Entries retain sender string IDs; use the producing application's string index to render text.
484    pub fn take_log_reader(&mut self) -> Option<CuTwinLogReader> {
485        self.log_reader.take()
486    }
487
488    pub fn stop(&mut self) -> CuResult<CuTwinStatus> {
489        self.stop.store(true, Ordering::Release);
490        if let Some(worker) = self.worker.take() {
491            worker.thread().unpark();
492            self.final_status = Some(
493                worker
494                    .join()
495                    .map_err(|_| CuError::from("Copper twin receiver panicked"))??,
496            );
497        }
498        self.final_status
499            .ok_or_else(|| CuError::from("Copper twin stopped after a worker failure"))
500    }
501}
502
503impl<A: LiveReplay> Drop for CuTwin<A> {
504    fn drop(&mut self) {
505        let _ = self.stop();
506    }
