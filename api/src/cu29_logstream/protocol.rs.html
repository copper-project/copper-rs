1//! Native capture records exchanged between a generated runtime and a live twin.
2//!
3//! A protocol instance serves one session. It owns packetization, integrity,
4//! reassembly and duplicate suppression. Both peers use the matching application
5//! and capture encoding. Each byte slice is one complete native payload.
6
7use crate::{ApplicationSchema, StreamIdentity};
8use bincode::{Decode, Encode};

13/// Identifies the robot construction, active mission and selected output slots.

15pub struct CuTelemetrySession {
16    pub identity: StreamIdentity,
17    pub context: SectionContext,
18    pub schema: ApplicationSchema,
19}

21/// Borrowed for one synchronous handoff. No packet or FEC header is included.
22#[derive(Clone, Copy, Debug)]
23pub enum CuTelemetryRecord<'a> {
24    Session(&'a CuTelemetrySession),
25    CopperList { id: u64, capture: &'a [u8] },
26    KeyFrame { copperlist_id: u64, state: &'a [u8] },
27    StructuredLog { sequence: u64, entry: &'a [u8] },
28}

30/// Output worker lifecycle. During drain the worker enforces a finite deadline.
31#[derive(Clone, Copy, Debug, PartialEq, Eq)]
32pub enum TxMode {
33    Running,
34    Draining,
35}
36
37/// Robot-side protocol, called on the output worker. The first record opens
38/// the session. A retained payload must be moved into protocol-owned storage.
39pub trait CuTelemetryTx: Send {
40    /// Accept one borrowed record.
41    fn send(&mut self, record: CuTelemetryRecord<'_>) -> CuResult<()>;
42
43    /// Perform bounded, nonblocking work and return the next absolute wakeup
44    /// on the transport clock supplied at setup.
45    /// The worker may also poll early after receiving a record. In draining
46    /// mode, stop periodic repetition and return None when pending work is done.
47    fn poll(&mut self, _mode: TxMode) -> CuResult<Option<CuTime>> {
48        Ok(None)
49    }
50}
51
52/// Deliver each borrowed record to both protocols on the same output worker.
53/// Both children are called on error. The first error in tuple order propagates.
54pub struct Fanout<A, B>(pub A, pub B);
55
56impl<A: CuTelemetryTx, B: CuTelemetryTx> CuTelemetryTx for Fanout<A, B> {
57    fn send(&mut self, record: CuTelemetryRecord<'_>) -> CuResult<()> {
58        let first = self.0.send(record);
59        let second = self.1.send(record);
60        first.and(second)
61    }
62
63    fn poll(&mut self, mode: TxMode) -> CuResult<Option<CuTime>> {
64        let first = self.0.poll(mode);
65        let second = self.1.poll(mode);
66        Ok(first?.into_iter().chain(second?).min())
67    }
68}

70/// Disable an optional output and report its first send or poll error on the worker.
71/// Both the output and its error callback must be nonblocking.
72pub struct BestEffort<T, F>(Option<(T, F)>);
73
74impl<T: CuTelemetryTx, F: FnOnce(CuError) + Send> BestEffort<T, F> {
75    pub fn new(tx: T, on_error: F) -> Self {
76        Self(Some((tx, on_error)))
77    }
78
79    fn attempt<R: Default>(&mut self, op: impl FnOnce(&mut T) -> CuResult<R>) -> R {
80        if let Some((tx, _)) = &mut self.0 {
81            match op(tx) {
82                Ok(value) => return value,
83                Err(error) => {
84                    if let Some((_, report)) = self.0.take() {
85                        report(error);
86                    }
87                }
88            }
89        }
90        R::default()
91    }
92}
93
94impl<T: CuTelemetryTx, F: FnOnce(CuError) + Send> CuTelemetryTx for BestEffort<T, F> {
95    fn send(&mut self, record: CuTelemetryRecord<'_>) -> CuResult<()> {
96        self.attempt(|tx| tx.send(record));
97        Ok(())
98    }
99
100    fn poll(&mut self, mode: TxMode) -> CuResult<Option<CuTime>> {
101        Ok(self.attempt(|tx| tx.poll(mode)))
102    }
103}

105/// Delivery from a single receiver. Report missing CopperLists in order,
106/// immediately before the next delivered list. Gap bounds are inclusive.
107#[derive(Clone, Copy, Debug)]
108pub enum CuTelemetryEvent<'a> {
109    Record(CuTelemetryRecord<'a>),
110    Gap {
111        first_id: u64,
112        last_id: u64,
113        reason: cu29_runtime::continuity::SourceGapReason,
114    },
115}

117/// Outcome of one bounded nonblocking receive operation.
118#[derive(Clone, Copy, Debug, PartialEq, Eq)]
119pub enum CuTelemetryPoll {
120    /// No input is available yet.
121    Pending,
122    /// Input was processed, possibly yielding zero or multiple events.
123    Progress,
124    /// All known input and final gap events have been delivered.
125    End,
126}

128/// Ground-side protocol. Deliver one session header before its records,
129/// CopperLists in ID order starting at zero (or report an initial gap), and
130/// snapshots at their original boundaries. The callback may borrow directly
131/// from reassembly storage and its first error must terminate delivery.
132pub trait CuTelemetryRx: Send {
133    fn try_recv(
134        &mut self,
135        deliver: impl FnMut(CuTelemetryEvent<'_>) -> CuResult<()>,
136    ) -> CuResult<CuTelemetryPoll>;
137}
