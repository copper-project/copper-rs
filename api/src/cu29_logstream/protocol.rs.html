1//! Native capture records exchanged between a generated runtime and a live twin.
2//!
3//! A protocol instance serves one session. It owns packetization, integrity,
4//! reassembly and duplicate suppression. Both peers use the matching application
5//! and capture encoding. Each byte slice is one complete native payload.
6

13/// Identifies the robot construction, active mission and selected output slots.

23/// Borrowed for one synchronous handoff. No packet or FEC header is included.
24#[derive(Clone, Copy, Debug)]
25pub enum CuTelemetryRecord<'a> {
26    Session(&'a CuTelemetrySession),
27    CopperList { id: u64, capture: &'a [u8] },
28    KeyFrame { copperlist_id: u64, state: &'a [u8] },
29    StructuredLog { sequence: u64, entry: &'a [u8] },

32/// Output worker lifecycle. During drain the worker enforces a finite deadline.
33#[derive(Clone, Copy, Debug, PartialEq, Eq)]
34pub enum TxMode {
35    Running,
36    Draining,
37}
38
39/// Robot-side protocol, driven by a host worker or explicitly between firmware iterations. The first record opens
40/// the session. A retained payload must be moved into protocol-owned storage.
41pub trait CuTelemetryTx: Send {
42    /// Accept one borrowed record with bounded, nonblocking work.
43    fn send(&mut self, record: CuTelemetryRecord<'_>) -> CuResult<()>;
44
45    /// Perform bounded, nonblocking work and return the next absolute wakeup
46    /// on the transport clock supplied at setup.
47    /// The worker may also poll early after receiving a record. In draining
48    /// mode, stop periodic repetition and return None when pending work is done.
49    fn poll(&mut self, _mode: TxMode) -> CuResult<Option<CuTime>> {
50        Ok(None)
51    }
52}
53
54/// Run one bounded service pass. Always poll after processing the available records.
55#[cfg(any(feature = "std", feature = "telemetry"))]
56pub(crate) fn service_tx<T: CuTelemetryTx + ?Sized>(
57    tx: &mut T,
58    mode: TxMode,
59    budget: usize,
60    mut send_next: impl FnMut(&mut T) -> CuResult<bool>,
61) -> CuResult<(usize, Option<CuTime>)> {
62    let mut sent = 0;
63    while sent < budget && send_next(tx)? {
64        sent += 1;
65    }
66    Ok((sent, tx.poll(mode)?))
67}

69/// Deliver each borrowed record to both protocols on the same output worker.
70/// Both children are called on error. The first error in tuple order propagates.
71pub struct Fanout<A, B>(pub A, pub B);
72
73impl<A: CuTelemetryTx, B: CuTelemetryTx> CuTelemetryTx for Fanout<A, B> {
74    fn send(&mut self, record: CuTelemetryRecord<'_>) -> CuResult<()> {
75        let first = self.0.send(record);
76        let second = self.1.send(record);
77        first.and(second)
78    }
79
80    fn poll(&mut self, mode: TxMode) -> CuResult<Option<CuTime>> {
81        let first = self.0.poll(mode);
82        let second = self.1.poll(mode);
83        Ok(first?.into_iter().chain(second?).min())
84    }
85}

87/// Disable an optional output and report its first send or poll error on the worker.
88/// Both the output and its error callback must be nonblocking.
89pub struct BestEffort<T, F>(Option<(T, F)>);
90
91impl<T: CuTelemetryTx, F: FnOnce(CuError) + Send> BestEffort<T, F> {
92    pub fn new(tx: T, on_error: F) -> Self {
93        Self(Some((tx, on_error)))
94    }
95
96    fn attempt<R: Default>(&mut self, op: impl FnOnce(&mut T) -> CuResult<R>) -> R {
97        if let Some((tx, _)) = &mut self.0 {
98            match op(tx) {
99                Ok(value) => return value,
100                Err(error) => {
101                    if let Some((_, report)) = self.0.take() {
102                        report(error);
103                    }
104                }
105            }
106        }
107        R::default()
108    }
109}
110
111impl<T: CuTelemetryTx, F: FnOnce(CuError) + Send> CuTelemetryTx for BestEffort<T, F> {
112    fn send(&mut self, record: CuTelemetryRecord<'_>) -> CuResult<()> {
113        self.attempt(|tx| tx.send(record));
114        Ok(())
115    }
116
117    fn poll(&mut self, mode: TxMode) -> CuResult<Option<CuTime>> {
118        Ok(self.attempt(|tx| tx.poll(mode)))
119    }
120}

122/// Delivery from a single receiver. Report missing CopperLists in order,
123/// immediately before the next delivered list. Gap bounds are inclusive.
124#[derive(Clone, Copy, Debug)]
125pub enum CuTelemetryEvent<'a> {
126    Record(CuTelemetryRecord<'a>),
127    Gap {
128        first_id: u64,
129        last_id: u64,
130        reason: cu29_runtime::continuity::SourceGapReason,
131    },
132}

134/// Outcome of one bounded nonblocking receive operation.
135#[derive(Clone, Copy, Debug, PartialEq, Eq)]
136pub enum CuTelemetryPoll {
137    /// No input is available yet.
138    Pending,
139    /// Input was processed, possibly yielding zero or multiple events.
140    Progress,
141    /// All known input and final gap events have been delivered.
142    End,
143}

145/// Ground-side protocol. Deliver one session header before its records,
146/// CopperLists in ID order starting at zero (or report an initial gap), and
147/// snapshots at their original boundaries. The callback may borrow directly
148/// from reassembly storage and its first error must terminate delivery.
149pub trait CuTelemetryRx: Send {
150    fn try_recv(
151        &mut self,
152        deliver: impl FnMut(CuTelemetryEvent<'_>) -> CuResult<()>,
153    ) -> CuResult<CuTelemetryPoll>;
154}
