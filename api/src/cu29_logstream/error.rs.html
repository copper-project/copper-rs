34impl Display for Error {
35    fn fmt(&self, formatter: &mut Formatter<'_>) -> core::fmt::Result {
36        match self {
37            Self::InvalidConfig(message) => write!(formatter, "invalid configuration: {message}"),
38            Self::ObjectTooLarge { actual, maximum } => {
39                write!(formatter, "object has {actual} bytes; maximum is {maximum}")
40            }
41            Self::TooManyDatagrams { actual, maximum } => {
42                write!(
43                    formatter,
44                    "object needs {actual} datagrams; maximum is {maximum}"
45                )
46            }
47            Self::TooManyRecords { actual, maximum } => {
48                write!(
49                    formatter,
50                    "stream has {actual} records; maximum is {maximum}"
51                )
52            }
53            Self::TooManySourceSymbols { actual, maximum } => {
54                write!(
55                    formatter,
56                    "stream has {actual} source symbols; maximum is {maximum}"
57                )
58            }
59            Self::BufferTooSmall { needed, available } => {
60                write!(
61                    formatter,
62                    "buffer needs {needed} bytes; only {available} available"
63                )
64            }
65            Self::TruncatedPacket => formatter.write_str("truncated logstream packet"),
66            Self::InvalidMagic => formatter.write_str("invalid logstream magic"),
67            Self::UnsupportedVersion(version) => {
68                write!(formatter, "unsupported logstream version {version}")
69            }
70            Self::InvalidHeaderLength(length) => {
71                write!(formatter, "invalid logstream header length {length}")
72            }
73            Self::UnknownLane(value) => write!(formatter, "unknown logstream lane {value}"),
74            Self::UnknownRecordKind(value) => write!(formatter, "unknown record kind {value}"),
75            Self::UnknownFecScheme(value) => write!(formatter, "unknown FEC scheme {value}"),
76            Self::UnknownSymbolKind(value) => write!(formatter, "unknown FEC symbol kind {value}"),
77            Self::PayloadLengthMismatch => formatter.write_str("packet payload length mismatch"),
78            Self::CrcMismatch => formatter.write_str("packet CRC32C mismatch"),
79            Self::InvalidFecMetadata(message) => {
80                write!(formatter, "invalid FEC metadata: {message}")
81            }
82            Self::InvalidFragment(message) => write!(formatter, "invalid fragment: {message}"),
83            Self::InconsistentObject => formatter.write_str("packet belongs to a different object"),
84            Self::TruncatedRecord => formatter.write_str("truncated semantic record"),
85            Self::RecordLengthMismatch => formatter.write_str("semantic record length mismatch"),
86            Self::RecordDigestMismatch => formatter.write_str("semantic record digest mismatch"),
87            Self::Codec(message) => write!(formatter, "semantic codec failed: {message}"),
88            Self::Transport(message) => write!(formatter, "datagram transport failed: {message}"),
89        }
90    }
91}
