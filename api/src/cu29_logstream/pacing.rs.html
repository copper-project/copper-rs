44/// Fixed-capacity packet storage; filling and replacing it never grows its allocation.
45struct Packets {
46    bytes: Vec<u8>,
47    lengths: Vec<usize>,
48    times: Vec<CuTime>,
49    ids: Vec<u64>,
50    mtu: usize,
51    head: usize,
52    len: usize,
53}
54
55impl Packets {
56    fn new(capacity: usize, mtu: usize) -> Self {
57        Self {
58            bytes: vec![0; capacity * mtu],
59            lengths: vec![0; capacity],
60            times: vec![CuTime::default(); capacity],
61            ids: vec![0; capacity],
62            mtu,
63            head: 0,
64            len: 0,
65        }
66    }
67
68    fn clear(&mut self) {
69        self.head = 0;
70        self.len = 0;
71    }
72
73    fn push(&mut self, bytes: &[u8], now: CuTime, id: u64) -> bool {
74        if self.len == self.lengths.len() || bytes.len() > self.mtu {
75            return false;
76        }
77        let slot = (self.head + self.len) % self.lengths.len();
78        self.bytes[slot * self.mtu..slot * self.mtu + bytes.len()].copy_from_slice(bytes);
79        self.lengths[slot] = bytes.len();
80        self.times[slot] = now;
81        self.ids[slot] = id;
82        self.len += 1;
83        true
84    }
85
86    fn get(&self, index: usize) -> &[u8] {
87        let slot = (self.head + index) % self.lengths.len();
88        &self.bytes[slot * self.mtu..slot * self.mtu + self.lengths[slot]]
89    }
90
91    fn pop(&mut self) {
92        self.head = (self.head + 1) % self.lengths.len();
93        self.len -= 1;
94    }
95}

107/// A destination's single packet scheduler and retained recovery state.
108///
109/// `now` always belongs to the same local RobotClock. The caller owns transport
110/// and wakeups; this core neither reads a global clock nor sleeps. Runtime objects
111/// are never retained here. Optional feedback can later request the same retained
112/// jobs through `request_recovery`, without bypassing this scheduler's budget.
113pub struct SenderCore {
114    config: LogStreamSenderConfig,
115    continuous: Box<ContinuousEncoder<1128, 64>>,
116    finite: FiniteObjectEncoder,
117    repairs: RepairSchedule,
118    scratch: Vec<u8>,
119    data: Packets,
120    manifest: Packets,
121    latest: RecoveryPackets,
122    pending_cl: Vec<PendingPackets>,
123    pending_kf: Vec<PendingPackets>,
124    latest_id: Option<u64>,
125    manifest_cursor: usize,
126    recovery_cursor: usize,
127    next_repeat: CuTime,
128    last_now: CuTime,
129    credit: u128,
130    capacity: u128,
131    // Byte deficit fairness: three MTUs of replay for one MTU of recovery.
132    deficit: [usize; 2],
133    lane: usize,
134    stopping: bool,
135    transport_pending: bool,
136    stats: SenderStats,
137}

149    /// Validate all storage arithmetic before allocating. `reserved_bytes` covers
150    /// the driver's owned-record pool, which shares the destination budget.
151    pub fn new(config: LogStreamSenderConfig, now: CuTime, reserved_bytes: usize) -> Result<Self> {
152        config.validate()?;
153        let policy = config.pacing;
154        if policy.bitrate_bps == 0
155            || policy.burst_packets == 0
156            || policy.max_latency.as_nanos() == 0
157        {
158            return Err(Error::InvalidConfig("pacing bounds must be nonzero"));
159        }
160        let mtu = crate::PACKET_HEADER_LEN + config.continuous.fec.symbol_size();
161        let object_bytes = usize::try_from(config.recovery.finite.max_object_bytes)
162            .map_err(|_| Error::InvalidConfig("object limit exceeds usize"))?;
163        let symbol = usize::from(config.recovery.finite.symbol_size);
164        if symbol != config.continuous.fec.symbol_size()
165            || symbol <= crate::rlc::FRAGMENT_HEADER_LEN
166            || config.continuous.repair_every_source_symbols == 0
167            || object_bytes < 256
168        {
169            return Err(Error::InvalidConfig(
170                "scheduled lanes require matching valid symbols and object bounds",
171            ));
172        }
173        // One source block keeps finite encoding and its retained packet bound explicit.
174        if object_bytes.div_ceil(symbol) > 56_403
175            || raptorq::ObjectTransmissionInformation::with_defaults(
176                object_bytes as u64,
177                symbol as u16,
178            )
179            .source_blocks()
180                != 1
181        {
182            return Err(Error::InvalidConfig(
183                "scheduled objects must fit one RaptorQ source block",
184            ));
185        }
186        let repairs_per_object = config.recovery.finite.repair_symbols_per_block as usize;
187        let finite_capacity = |bytes: usize| bytes.div_ceil(symbol).checked_add(repairs_per_object);
188        let manifest_capacity = finite_capacity(config.recovery.manifest_record.len())
189            .ok_or(Error::InvalidConfig("manifest packet bound overflow"))?;
190        let control_capacity = finite_capacity(object_bytes)
191            .and_then(|v| finite_capacity(256).and_then(|a| v.checked_add(a)))
192            .ok_or(Error::InvalidConfig("recovery packet bound overflow"))?;
193        let source_capacity = config
194            .continuous
195            .max_record_bytes
196            .div_ceil(symbol - crate::rlc::FRAGMENT_HEADER_LEN);
197        let boundary_capacity = source_capacity
198            .checked_mul(2)
199            .ok_or(Error::InvalidConfig("boundary packet bound overflow"))?;
200        let packet_bytes = mtu + size_of::<usize>() + size_of::<CuTime>() + size_of::<u64>();
201        let retained_packets = control_capacity
202            .checked_mul(PENDING_KEYFRAMES + 1)
203            .and_then(|v| {
204                boundary_capacity
205                    .checked_mul(PENDING_BOUNDARIES + 1)
206                    .and_then(|b| v.checked_add(b))
207            })
208            .and_then(|v| v.checked_add(manifest_capacity))
209            .ok_or(Error::InvalidConfig("retained packet bound overflow"))?;
210        let fixed = retained_packets
211            .checked_mul(packet_bytes)
212            .and_then(|v| v.checked_add(size_of::<ContinuousEncoder<1128, 64>>()))
213            .and_then(|v| v.checked_add(size_of::<Self>()))
214            .and_then(|v| {
215                v.checked_add(
216                    (PENDING_BOUNDARIES + PENDING_KEYFRAMES) * size_of::<PendingPackets>(),
217                )
218            })
219            .and_then(|v| v.checked_add(mtu))
220            .and_then(|v| v.checked_add(config.recovery.manifest_record.len()))
221            .and_then(|v| v.checked_add(reserved_bytes))
222            .ok_or(Error::InvalidConfig("sender storage bound overflow"))?;
223        let queue_capacity = policy
224            .memory_budget_bytes
225            .checked_sub(fixed)
226            .map(|remaining| remaining / packet_bytes)
227            .filter(|capacity| *capacity >= boundary_capacity.max(1))
228            .ok_or(Error::InvalidConfig(
229                "sender buffers exceed destination memory budget",
230            ))?;
231        let continuous = Box::new(ContinuousEncoder::new(
232            config.continuous.identity,
233            config.continuous.lane,
234            config.continuous.fec,
235            config.continuous.max_record_bytes,
236            config.continuous.initial_esi,
237        )?);
238        let mut finite = FiniteObjectEncoder::new(config.recovery.finite)?;
239        let mut manifest = Packets::new(manifest_capacity, mtu);
240        let manifest_id = decode_record(&config.recovery.manifest_record)?.object_id;
241        finite.push_record_with(&config.recovery.manifest_record, |packet| {
242            if !manifest.push(packet, now, manifest_id) {
243                return Err(Error::InvalidConfig("manifest packet capacity"));
244            }
245            Ok(())
246        })?;
247        let recovery = || RecoveryPackets {
248            control: Packets::new(control_capacity, mtu),
249            boundary: Packets::new(boundary_capacity, mtu),
250        };
251        let capacity = u128::from(policy.burst_packets) * mtu as u128 * 8 * 1_000_000_000;
252        Ok(Self {
253            repairs: RepairSchedule::new(
254                config.continuous.repair_every_source_symbols,
255                config.continuous.first_repair_key,
256                config.continuous.repair_density,
257            ),
258            config,
259            continuous,
260            finite,
261            scratch: vec![0; mtu],
262            data: Packets::new(queue_capacity, mtu),
263            manifest,
264            latest: recovery(),
265            pending_cl: (0..PENDING_BOUNDARIES)
266                .map(|_| PendingPackets {
267                    id: None,
268                    packets: Packets::new(boundary_capacity, mtu),
269                })
270                .collect(),
271            pending_kf: (0..PENDING_KEYFRAMES)
272                .map(|_| PendingPackets {
273                    id: None,
274                    packets: Packets::new(control_capacity, mtu),
275                })
276                .collect(),
277            latest_id: None,
278            manifest_cursor: 0,
279            recovery_cursor: 0,
280            next_repeat: now + RECOVERY_REPEAT_INTERVAL,
281            last_now: now,
282            credit: capacity,
283            capacity,
284            deficit: [0; 2],
285            lane: 0,
286            stopping: false,
287            transport_pending: false,
288            stats: SenderStats::default(),
289        })
290    }

296    /// Admit an already encoded record from a bounded worker buffer. No borrow
297    /// escapes this call. CL packets shed on overflow; retained recovery replaces
298    /// only a complete matching boundary/keyframe pair.
299    pub fn accept_record(&mut self, record: &[u8], now: CuTime) -> Result<()> {
300        let decoded = decode_record(record)?;
301        match decoded.kind {
302            RecordKind::CopperList => {
303                let retain = decoded
304                    .object_id
305                    .is_multiple_of(u64::from(self.config.recovery.recovery_interval));
306                let slot = if retain {
307                    Some(Self::reserve(
308                        &mut self.pending_cl,
309                        decoded.object_id,
310                        &mut self.stats,
311                    ))
312                } else {
313                    None
314                };
315                let stats = &mut self.stats;
316                let pending = &mut self.pending_cl;
317                let data = &mut self.data;
318                self.continuous.push_record_with_repairs(
319                    record,
320                    &mut self.scratch,
321                    &mut |packet| {
322                        if let Some(slot) = slot
323                            && !pending[slot].packets.push(packet, now, decoded.object_id)
324                        {
325                            return Err(Error::InvalidConfig("boundary packet capacity"));
326                        }
327                        if !data.push(packet, now, decoded.object_id) {
328                            stats.queue_drops = stats.queue_drops.saturating_add(1);
329                        }
330                        stats.queue_peak = stats.queue_peak.max(data.len);
331                        Ok(())
332                    },
333                    &mut self.repairs,
334                )?;
335            }
336            RecordKind::KeyFrame => {
337                if !decoded
338                    .object_id
339                    .is_multiple_of(u64::from(self.config.recovery.recovery_interval))
340                {
341                    return Ok(());
342                }
343                let manifest = decode_record(&self.config.recovery.manifest_record)?;
344                let recovery_point = encode_record(
345                    RecordKind::RecoveryPoint,
346                    decoded.object_id,
347                    &encode_recovery_point(&RecoveryPoint {
348                        manifest_object_id: manifest.object_id,
349                        manifest_record_digest: manifest.digest,
350                        copperlist_id: decoded.object_id,
351                        keyframe_object_id: decoded.object_id,
352                        keyframe_record_digest: decoded.digest,
353                    })?,
354                )?;
355                let slot = Self::reserve(&mut self.pending_kf, decoded.object_id, &mut self.stats);
356                for bytes in [record, recovery_point.as_slice()] {
357                    self.finite.push_record_with(bytes, |packet| {
358                        if !self.pending_kf[slot]
359                            .packets
360                            .push(packet, now, decoded.object_id)
361                        {
362                            return Err(Error::InvalidConfig("recovery packet capacity"));
363                        }
364                        Ok(())
365                    })?;
366                }
367            }
368            _ => {
369                return Err(Error::InvalidConfig(
370                    "sender inbox accepts CLs and keyframes",
371                ));
372            }
373        }
374        self.promote_recovery();
375        Ok(())
376    }

378    fn reserve(slots: &mut [PendingPackets], id: u64, stats: &mut SenderStats) -> usize {
379        let index = slots
380            .iter()
381            .position(|s| s.id.is_none() || s.id == Some(id))
382            .unwrap_or_else(|| {
383                slots
384                    .iter()
385                    .enumerate()
386                    .min_by_key(|(_, s)| s.id)
387                    .unwrap()
388                    .0
389            });
390        if slots[index].id.is_some() {
391            stats.recovery_superseded += 1;
392        }
393        slots[index].id = Some(id);
394        slots[index].packets.clear();
395        index
396    }

398    fn promote_recovery(&mut self) {
399        // Finish an in-flight bundle before replacing it; frequent captures must
400        // not continually restart a large transfer on a slow link.
401        if self.recovery_cursor < self.recovery_len() {
402            return;
403        }
404        let pair = self
405            .pending_cl
406            .iter()
407            .enumerate()
408            .filter_map(|(cl, entry)| {
409                let id = entry.id?;
410                // A recovery point must not make the receiver skip older source packets
411                // still waiting in our own queue. Expiry releases this fence too.
412                if self.data.len > 0 && self.data.ids[self.data.head] < id {
413                    return None;
414                }
415                if self.latest_id.is_some_and(|latest| id <= latest) {
416                    return None;
417                }
418                self.pending_kf
419                    .iter()
420                    .position(|kf| kf.id == Some(id))
421                    .map(|kf| (id, cl, kf))
422            })
423            .min_by_key(|(id, _, _)| *id);
424        if let Some((id, cl, kf)) = pair {
425            core::mem::swap(&mut self.latest.boundary, &mut self.pending_cl[cl].packets);
426            core::mem::swap(&mut self.latest.control, &mut self.pending_kf[kf].packets);
427            self.pending_cl[cl].id = None;
428            self.pending_kf[kf].id = None;
429            self.latest_id = Some(id);
430            self.recovery_cursor = 0;
431            self.request_recovery();
432        }
433    }

435    /// Coalesced request for retained bootstrap data. Future advisory feedback
436    /// uses this same operation; it grants neither extra bandwidth nor retention.
437    pub fn request_recovery(&mut self) {
438        if self.stopping || self.control_packet().is_some() {
439            return;
440        }
441        if self.manifest_cursor >= self.manifest.len {
442            self.manifest_cursor = 0;
443        }
444        if self.recovery_cursor >= self.recovery_len() {
445            self.recovery_cursor = 0;
446        }
447    }
448
449    fn recovery_len(&self) -> usize {
450        self.latest.control.len + self.latest.boundary.len
451    }
452
453    fn control_packet(&self) -> Option<&[u8]> {
454        if self.manifest_cursor < self.manifest.len {
455            return Some(self.manifest.get(self.manifest_cursor));
456        }
457        if self.recovery_cursor < self.latest.control.len {
458            return Some(self.latest.control.get(self.recovery_cursor));
459        }
460        let boundary = self.recovery_cursor - self.latest.control.len;
461        (boundary < self.latest.boundary.len).then(|| self.latest.boundary.get(boundary))
462    }

469    /// Account for work abandoned at the driver's finite shutdown deadline.
470    pub fn discard_pending(&mut self) {
471        self.stats.shutdown_drops += (self.data.len
472            + self.manifest.len.saturating_sub(self.manifest_cursor)
473            + self.recovery_len().saturating_sub(self.recovery_cursor))
474            as u64;
475        self.data.clear();
476        self.manifest_cursor = self.manifest.len;
477        self.recovery_cursor = self.recovery_len();
478    }

484    /// Send a bounded amount of eligible work and return the next local deadline.
485    /// Carrier WouldBlock consumes the attempt's budget and drops the packet.
486    pub fn poll<T: CuStreamTx>(
487        &mut self,
488        now: CuTime,
489        transport: &mut T,
490    ) -> Result<Option<CuTime>> {
491        let next = self.poll_packets(now, transport)?;
492        self.transport_pending = transport.poll_pending().map_err(|error| match error {
493            CuStreamTxError::WouldBlock => {
494                Error::Transport("Transport poll must report pending instead of WouldBlock")
495            }
496            CuStreamTxError::Failed(message) => Error::Transport(message),
497        })?;
498        Ok(if self.transport_pending {
499            let retry = now + CuDuration::from_millis(1);
500            Some(next.map_or(retry, |deadline| deadline.min(retry)))
501        } else {
502            next
503        })
504    }

506    fn poll_packets<T: CuStreamTx>(
507        &mut self,
508        now: CuTime,
509        transport: &mut T,
510    ) -> Result<Option<CuTime>> {
511        if now < self.last_now {
512            return Err(Error::InvalidConfig("sender clock moved backwards"));
513        }
514        let elapsed = (now - self.last_now).as_nanos();
515        self.credit = self.capacity.min(
516            self.credit
517                .saturating_add(u128::from(elapsed) * u128::from(self.config.pacing.bitrate_bps)),
518        );
519        self.last_now = now;
520        self.promote_recovery();
521        if !self.stopping && now >= self.next_repeat {
522            self.request_recovery();
523            self.stats.recovery_rounds = self.stats.recovery_rounds.saturating_add(1);
524            self.next_repeat = now + RECOVERY_REPEAT_INTERVAL;
525        }
526        let mut work = 0;
527        while self.data.len > 0
528            && now >= self.data.times[self.data.head] + self.config.pacing.max_latency
529            && work < MAX_SENDS_PER_POLL
530        {
531            self.data.pop();
532            self.stats.expired_packets += 1;
533            work += 1;
534        }
535        while work < MAX_SENDS_PER_POLL {
536            let has_data = self.data.len > 0;
537            let has_control = self.control_packet().is_some();
538            if !has_data && !has_control {
539                break;
540            }
541            if (self.lane == 0 && !has_data) || (self.lane == 1 && !has_control) {
542                self.deficit[self.lane] = 0;
543                self.lane ^= 1;
544            }
545            let packet = if self.lane == 0 {
546                self.data.get(0)
547            } else {
548                self.control_packet().unwrap()
549            };
550            let len = packet.len();
551            if self.deficit[self.lane] < len {
552                self.deficit[self.lane] += self.scratch.len() * if self.lane == 0 { 3 } else { 1 };
553                self.lane ^= 1;
554                continue;
555            }
556            let cost = len as u128 * 8 * 1_000_000_000;
557            if self.credit < cost {
558                let wait =
559                    (cost - self.credit).div_ceil(u128::from(self.config.pacing.bitrate_bps));
560                let mut deadline = now + CuDuration(wait.min(u128::from(u64::MAX)) as u64);
561                if self.data.len > 0 {
562                    deadline = deadline
563                        .min(self.data.times[self.data.head] + self.config.pacing.max_latency);
564                }
565                return Ok(Some(if self.stopping {
566                    deadline
567                } else {
568                    deadline.min(self.next_repeat)
569                }));
570            }
571            match transport.try_send(packet) {
572                Ok(()) => {
573                    self.stats.packets_sent += 1;
574                    self.stats.bytes_sent += len as u64;
575                }
576                Err(CuStreamTxError::WouldBlock) => self.stats.transport_drops += 1,
577                Err(CuStreamTxError::Failed(message)) => return Err(Error::Transport(message)),
578            }
579            self.credit -= cost;
580            self.deficit[self.lane] -= len;
581            if self.lane == 0 {
582                self.data.pop();
583            } else if self.manifest_cursor < self.manifest.len {
584                self.manifest_cursor += 1;
585            } else {
586                self.recovery_cursor += 1;
587            }
588            work += 1;
589            self.promote_recovery();
590        }
591        Ok(if self.data.len != 0 || self.control_packet().is_some() {
592            Some(now)
593        } else if self.stopping {
594            None
595        } else {
596            Some(self.next_repeat)
597        })
598    }
