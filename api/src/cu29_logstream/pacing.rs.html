11/// Autonomous repetition period, measured by the sender's local RobotClock.
12/// Repetition is coalesced while the previous transmission is still pending.
13pub const RECOVERY_REPEAT_INTERVAL: CuDuration = CuDuration(250_000_000);
14const MAX_SENDS_PER_POLL: usize = 64;
15const PENDING_BOUNDARIES: usize = 4;
16const PENDING_KEYFRAMES: usize = 2;

43/// Fixed-capacity packet storage; filling and replacing it never grows its allocation.
44struct Packets {
45    bytes: Vec<u8>,
46    lengths: Vec<usize>,
47    times: Vec<CuTime>,
48    ids: Vec<u64>,
49    mtu: usize,
50    head: usize,
51    len: usize,
52}
53
54impl Packets {
55    fn new(capacity: usize, mtu: usize) -> Self {
56        Self {
57            bytes: vec![0; capacity * mtu],
58            lengths: vec![0; capacity],
59            times: vec![CuTime::default(); capacity],
60            ids: vec![0; capacity],
61            mtu,
62            head: 0,
63            len: 0,
64        }
65    }
66
67    fn clear(&mut self) {
68        self.head = 0;
69        self.len = 0;
70    }
71
72    fn push(&mut self, bytes: &[u8], now: CuTime, id: u64) -> bool {
73        if self.len == self.lengths.len() || bytes.len() > self.mtu {
74            return false;
75        }
76        let slot = (self.head + self.len) % self.lengths.len();
77        self.bytes[slot * self.mtu..slot * self.mtu + bytes.len()].copy_from_slice(bytes);
78        self.lengths[slot] = bytes.len();
79        self.times[slot] = now;
80        self.ids[slot] = id;
81        self.len += 1;
82        true
83    }
84
85    fn get(&self, index: usize) -> &[u8] {
86        let slot = (self.head + index) % self.lengths.len();
87        &self.bytes[slot * self.mtu..slot * self.mtu + self.lengths[slot]]
88    }
89
90    fn pop(&mut self) {
91        self.head = (self.head + 1) % self.lengths.len();
92        self.len -= 1;
93    }
94}

106/// A destination's single packet scheduler and retained recovery state.
107///
108/// `now` always belongs to the same local RobotClock. The caller owns transport
109/// and wakeups; this core neither reads a global clock nor sleeps. Runtime objects
110/// are never retained here. Optional feedback can later request the same retained
111/// jobs through `request_recovery`, without bypassing this scheduler's budget.
112pub struct SenderCore {
113    config: LogStreamSenderConfig,
114    continuous: Box<ContinuousEncoder<1128, 64>>,
115    finite: FiniteObjectEncoder,
116    repairs: RepairSchedule,
117    scratch: Vec<u8>,
118    data: Packets,
119    manifest: Packets,
120    latest: RecoveryPackets,
121    pending_cl: Vec<PendingPackets>,
122    pending_kf: Vec<PendingPackets>,
123    latest_id: Option<u64>,
124    manifest_cursor: usize,
125    recovery_cursor: usize,
126    next_repeat: CuTime,
127    last_now: CuTime,
128    credit: u128,
129    capacity: u128,
130    // Byte deficit fairness: three MTUs of replay for one MTU of recovery.
131    deficit: [usize; 2],
132    lane: usize,
133    stopping: bool,
134    stats: SenderStats,
135}

138    /// Validate all storage arithmetic before allocating. `reserved_bytes` covers
139    /// the driver's owned-record pool, which shares the destination budget.
140    pub fn new(config: LogStreamSenderConfig, now: CuTime, reserved_bytes: usize) -> Result<Self> {
141        config.validate()?;
142        let policy = config.pacing;
143        if policy.bitrate_bps == 0
144            || policy.burst_packets == 0
145            || policy.max_latency.as_nanos() == 0
146        {
147            return Err(Error::InvalidConfig("pacing bounds must be nonzero"));
148        }
149        let mtu = crate::PACKET_HEADER_LEN + config.continuous.fec.symbol_size();
150        let object_bytes = usize::try_from(config.recovery.finite.max_object_bytes)
151            .map_err(|_| Error::InvalidConfig("object limit exceeds usize"))?;
152        let symbol = usize::from(config.recovery.finite.symbol_size);
153        if symbol != config.continuous.fec.symbol_size()
154            || symbol <= crate::rlc::FRAGMENT_HEADER_LEN
155            || config.continuous.repair_every_source_symbols == 0
156            || object_bytes < 256
157        {
158            return Err(Error::InvalidConfig(
159                "scheduled lanes require matching valid symbols and object bounds",
160            ));
161        }
162        // One source block keeps finite encoding and its retained packet bound explicit.
163        if object_bytes.div_ceil(symbol) > 56_403
164            || raptorq::ObjectTransmissionInformation::with_defaults(
165                object_bytes as u64,
166                symbol as u16,
167            )
168            .source_blocks()
169                != 1
170        {
171            return Err(Error::InvalidConfig(
172                "scheduled objects must fit one RaptorQ source block",
173            ));
174        }
175        let repairs_per_object = config.recovery.finite.repair_symbols_per_block as usize;
176        let finite_capacity = |bytes: usize| bytes.div_ceil(symbol).checked_add(repairs_per_object);
177        let manifest_capacity = finite_capacity(config.recovery.manifest_record.len())
178            .ok_or(Error::InvalidConfig("manifest packet bound overflow"))?;
179        let control_capacity = finite_capacity(object_bytes)
180            .and_then(|v| finite_capacity(256).and_then(|a| v.checked_add(a)))
181            .ok_or(Error::InvalidConfig("recovery packet bound overflow"))?;
182        let source_capacity = config
183            .continuous
184            .max_record_bytes
185            .div_ceil(symbol - crate::rlc::FRAGMENT_HEADER_LEN);
186        let boundary_capacity = source_capacity
187            .checked_mul(2)
188            .ok_or(Error::InvalidConfig("boundary packet bound overflow"))?;
189        let packet_bytes = mtu + size_of::<usize>() + size_of::<CuTime>() + size_of::<u64>();
190        let retained_packets = control_capacity
191            .checked_mul(PENDING_KEYFRAMES + 1)
192            .and_then(|v| {
193                boundary_capacity
194                    .checked_mul(PENDING_BOUNDARIES + 1)
195                    .and_then(|b| v.checked_add(b))
196            })
197            .and_then(|v| v.checked_add(manifest_capacity))
198            .ok_or(Error::InvalidConfig("retained packet bound overflow"))?;
199        let fixed = retained_packets
200            .checked_mul(packet_bytes)
201            .and_then(|v| v.checked_add(size_of::<ContinuousEncoder<1128, 64>>()))
202            .and_then(|v| v.checked_add(size_of::<Self>()))
203            .and_then(|v| {
204                v.checked_add(
205                    (PENDING_BOUNDARIES + PENDING_KEYFRAMES) * size_of::<PendingPackets>(),
206                )
207            })
208            .and_then(|v| v.checked_add(mtu))
209            .and_then(|v| v.checked_add(config.recovery.manifest_record.len()))
210            .and_then(|v| v.checked_add(reserved_bytes))
211            .ok_or(Error::InvalidConfig("sender storage bound overflow"))?;
212        let queue_capacity = policy
213            .memory_budget_bytes
214            .checked_sub(fixed)
215            .map(|remaining| remaining / packet_bytes)
216            .filter(|capacity| *capacity >= boundary_capacity.max(1))
217            .ok_or(Error::InvalidConfig(
218                "sender buffers exceed destination memory budget",
219            ))?;
220        let continuous = Box::new(ContinuousEncoder::new(
221            config.continuous.identity,
222            config.continuous.first_packet_sequence,
223            config.continuous.lane,
224            config.continuous.fec,
225            config.continuous.max_record_bytes,
226            config.continuous.initial_esi,
227        )?);
228        let mut finite = FiniteObjectEncoder::new(config.recovery.finite)?;
229        let mut manifest = Packets::new(manifest_capacity, mtu);
230        let manifest_id = decode_record(&config.recovery.manifest_record)?.object_id;
231        finite.push_record_with(&config.recovery.manifest_record, |packet| {
232            if !manifest.push(packet, now, manifest_id) {
233                return Err(Error::InvalidConfig("manifest packet capacity"));
234            }
235            Ok(())
236        })?;
237        let recovery = || RecoveryPackets {
238            control: Packets::new(control_capacity, mtu),
239            boundary: Packets::new(boundary_capacity, mtu),
240        };
241        let capacity = u128::from(policy.burst_packets) * mtu as u128 * 8 * 1_000_000_000;
242        Ok(Self {
243            repairs: RepairSchedule::new(
244                config.continuous.repair_every_source_symbols,
245                config.continuous.first_repair_key,
246                config.continuous.repair_density,
247            ),
248            config,
249            continuous,
250            finite,
251            scratch: vec![0; mtu],
252            data: Packets::new(queue_capacity, mtu),
253            manifest,
254            latest: recovery(),
255            pending_cl: (0..PENDING_BOUNDARIES)
256                .map(|_| PendingPackets {
257                    id: None,
258                    packets: Packets::new(boundary_capacity, mtu),
259                })
260                .collect(),
261            pending_kf: (0..PENDING_KEYFRAMES)
262                .map(|_| PendingPackets {
263                    id: None,
264                    packets: Packets::new(control_capacity, mtu),
265                })
266                .collect(),
267            latest_id: None,
268            manifest_cursor: 0,
269            recovery_cursor: 0,
270            next_repeat: now + RECOVERY_REPEAT_INTERVAL,
271            last_now: now,
272            credit: capacity,
273            capacity,
274            deficit: [0; 2],
275            lane: 0,
276            stopping: false,
277            stats: SenderStats::default(),
278        })
279    }

285    /// Admit an already encoded record from a bounded worker buffer. No borrow
286    /// escapes this call. CL packets shed on overflow; retained recovery replaces
287    /// only a complete matching boundary/keyframe pair.
288    pub fn accept_record(&mut self, record: &[u8], now: CuTime) -> Result<()> {
289        let decoded = decode_record(record)?;
290        match decoded.kind {
291            RecordKind::CopperList => {
292                let retain = decoded
293                    .object_id
294                    .is_multiple_of(u64::from(self.config.recovery.anchor_interval));
295                let slot = if retain {
296                    Some(Self::reserve(
297                        &mut self.pending_cl,
298                        decoded.object_id,
299                        &mut self.stats,
300                    ))
301                } else {
302                    None
303                };
304                let stats = &mut self.stats;
305                let pending = &mut self.pending_cl;
306                let data = &mut self.data;
307                self.continuous.push_record_with_repairs(
308                    record,
309                    &mut self.scratch,
310                    &mut |packet| {
311                        if let Some(slot) = slot
312                            && !pending[slot].packets.push(packet, now, decoded.object_id)
313                        {
314                            return Err(Error::InvalidConfig("boundary packet capacity"));
315                        }
316                        if !data.push(packet, now, decoded.object_id) {
317                            stats.queue_drops = stats.queue_drops.saturating_add(1);
318                        }
319                        stats.queue_peak = stats.queue_peak.max(data.len);
320                        Ok(())
321                    },
322                    &mut self.repairs,
323                )?;
324            }
325            RecordKind::KeyFrame => {
326                if !decoded
327                    .object_id
328                    .is_multiple_of(u64::from(self.config.recovery.anchor_interval))
329                {
330                    return Ok(());
331                }
332                let manifest = decode_record(&self.config.recovery.manifest_record)?;
333                let anchor = encode_record(
334                    RecordKind::Anchor,
335                    decoded.object_id,
336                    &encode_anchor(&Anchor {
337                        manifest_object_id: manifest.object_id,
338                        manifest_record_digest: manifest.digest,
339                        copperlist_id: decoded.object_id,
340                        keyframe_object_id: decoded.object_id,
341                        keyframe_record_digest: decoded.digest,
342                    })?,
343                )?;
344                let slot = Self::reserve(&mut self.pending_kf, decoded.object_id, &mut self.stats);
345                for bytes in [record, anchor.as_slice()] {
346                    self.finite.push_record_with(bytes, |packet| {
347                        if !self.pending_kf[slot]
348                            .packets
349                            .push(packet, now, decoded.object_id)
350                        {
351                            return Err(Error::InvalidConfig("recovery packet capacity"));
352                        }
353                        Ok(())
354                    })?;
355                }
356            }
357            _ => {
358                return Err(Error::InvalidConfig(
359                    "sender inbox accepts CLs and keyframes",
360                ));
361            }
362        }
363        self.promote_recovery();
364        Ok(())
365    }

367    fn reserve(slots: &mut [PendingPackets], id: u64, stats: &mut SenderStats) -> usize {
368        let index = slots
369            .iter()
370            .position(|s| s.id.is_none() || s.id == Some(id))
371            .unwrap_or_else(|| {
372                slots
373                    .iter()
374                    .enumerate()
375                    .min_by_key(|(_, s)| s.id)
376                    .unwrap()
377                    .0
378            });
379        if slots[index].id.is_some() {
380            stats.recovery_superseded += 1;
381        }
382        slots[index].id = Some(id);
383        slots[index].packets.clear();
384        index
385    }

387    fn promote_recovery(&mut self) {
388        // Finish an in-flight bundle before replacing it; frequent captures must
389        // not continually restart a large transfer on a slow link.
390        if self.recovery_cursor < self.recovery_len() {
391            return;
392        }
393        let pair = self
394            .pending_cl
395            .iter()
396            .enumerate()
397            .filter_map(|(cl, entry)| {
398                let id = entry.id?;
399                // An anchor must not make the receiver skip older source packets
400                // still waiting in our own queue. Expiry releases this fence too.
401                if self.data.len > 0 && self.data.ids[self.data.head] < id {
402                    return None;
403                }
404                if self.latest_id.is_some_and(|latest| id <= latest) {
405                    return None;
406                }
407                self.pending_kf
408                    .iter()
409                    .position(|kf| kf.id == Some(id))
410                    .map(|kf| (id, cl, kf))
411            })
412            .min_by_key(|(id, _, _)| *id);
413        if let Some((id, cl, kf)) = pair {
414            core::mem::swap(&mut self.latest.boundary, &mut self.pending_cl[cl].packets);
415            core::mem::swap(&mut self.latest.control, &mut self.pending_kf[kf].packets);
416            self.pending_cl[cl].id = None;
417            self.pending_kf[kf].id = None;
418            self.latest_id = Some(id);
419            self.recovery_cursor = 0;
420            self.request_recovery();
421        }
422    }

424    /// Coalesced request for retained bootstrap data. Future advisory feedback
425    /// uses this same operation; it grants neither extra bandwidth nor retention.
426    pub fn request_recovery(&mut self) {
427        if self.stopping || self.control_packet().is_some() {
428            return;
429        }
430        if self.manifest_cursor >= self.manifest.len {
431            self.manifest_cursor = 0;
432        }
433        if self.recovery_cursor >= self.recovery_len() {
434            self.recovery_cursor = 0;
435        }
436    }
437
438    fn recovery_len(&self) -> usize {
439        self.latest.control.len + self.latest.boundary.len
440    }
441
442    fn control_packet(&self) -> Option<&[u8]> {
443        if self.manifest_cursor < self.manifest.len {
444            return Some(self.manifest.get(self.manifest_cursor));
445        }
446        if self.recovery_cursor < self.latest.control.len {
447            return Some(self.latest.control.get(self.recovery_cursor));
448        }
449        let boundary = self.recovery_cursor - self.latest.control.len;
450        (boundary < self.latest.boundary.len).then(|| self.latest.boundary.get(boundary))
451    }

453    /// Disable periodic work. The driver enforces the finite shutdown deadline.

458    /// Account for work abandoned at the driver's finite shutdown deadline.
459    pub fn discard_pending(&mut self) {
460        self.stats.shutdown_drops += (self.data.len
461            + self.manifest.len.saturating_sub(self.manifest_cursor)
462            + self.recovery_len().saturating_sub(self.recovery_cursor))
463            as u64;
464        self.data.clear();
465        self.manifest_cursor = self.manifest.len;
466        self.recovery_cursor = self.recovery_len();
467    }

469    pub fn is_idle(&self) -> bool {
470        self.data.len == 0 && self.control_packet().is_none()
471    }

473    /// Send a bounded amount of eligible work and return the next local deadline.
474    /// Carrier WouldBlock consumes the attempt's budget and drops the packet.
475    pub fn poll<T: CuStreamTx>(
476        &mut self,
477        now: CuTime,
478        transport: &mut T,
479    ) -> Result<Option<CuTime>> {
480        if now < self.last_now {
481            return Err(Error::InvalidConfig("sender clock moved backwards"));
482        }
483        let elapsed = (now - self.last_now).as_nanos();
484        self.credit = self.capacity.min(
485            self.credit
486                .saturating_add(u128::from(elapsed) * u128::from(self.config.pacing.bitrate_bps)),
487        );
488        self.last_now = now;
489        self.promote_recovery();
490        if !self.stopping && now >= self.next_repeat {
491            self.request_recovery();
492            self.stats.recovery_rounds = self.stats.recovery_rounds.saturating_add(1);
493            self.next_repeat = now + RECOVERY_REPEAT_INTERVAL;
494        }
495        let mut work = 0;
496        while self.data.len > 0
497            && now >= self.data.times[self.data.head] + self.config.pacing.max_latency
498            && work < MAX_SENDS_PER_POLL
499        {
500            self.data.pop();
501            self.stats.expired_packets += 1;
502            work += 1;
503        }
504        while work < MAX_SENDS_PER_POLL {
505            let has_data = self.data.len > 0;
506            let has_control = self.control_packet().is_some();
507            if !has_data && !has_control {
508                break;
509            }
510            if (self.lane == 0 && !has_data) || (self.lane == 1 && !has_control) {
511                self.deficit[self.lane] = 0;
512                self.lane ^= 1;
513            }
514            let packet = if self.lane == 0 {
515                self.data.get(0)
516            } else {
517                self.control_packet().unwrap()
518            };
519            let len = packet.len();
520            if self.deficit[self.lane] < len {
521                self.deficit[self.lane] += self.scratch.len() * if self.lane == 0 { 3 } else { 1 };
522                self.lane ^= 1;
523                continue;
524            }
525            let cost = len as u128 * 8 * 1_000_000_000;
526            if self.credit < cost {
527                let wait =
528                    (cost - self.credit).div_ceil(u128::from(self.config.pacing.bitrate_bps));
529                let mut deadline = now + CuDuration(wait.min(u128::from(u64::MAX)) as u64);
530                if self.data.len > 0 {
531                    deadline = deadline
532                        .min(self.data.times[self.data.head] + self.config.pacing.max_latency);
533                }
534                return Ok(Some(if self.stopping {
535                    deadline
536                } else {
537                    deadline.min(self.next_repeat)
538                }));
539            }
540            match transport.try_send(packet) {
541                Ok(()) => {
542                    self.stats.packets_sent += 1;
543                    self.stats.bytes_sent += len as u64;
544                }
545                Err(CuStreamTxError::WouldBlock) => self.stats.transport_drops += 1,
546                Err(CuStreamTxError::Failed(message)) => return Err(Error::Transport(message)),
547            }
548            self.credit -= cost;
549            self.deficit[self.lane] -= len;
550            if self.lane == 0 {
551                self.data.pop();
552            } else if self.manifest_cursor < self.manifest.len {
553                self.manifest_cursor += 1;
554            } else {
555                self.recovery_cursor += 1;
556            }
557            work += 1;
558            self.promote_recovery();
559        }
560        Ok(if !self.is_idle() {
561            Some(now)
562        } else if self.stopping {
563            None
564        } else {
565            Some(self.next_repeat)
566        })
567    }
