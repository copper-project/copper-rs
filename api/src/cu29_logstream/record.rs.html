4const RECORD_MAGIC: [u8; 4] = *b"CUSR";
5const RECORD_VERSION: u8 = 1;
6pub const RECORD_HEADER_LEN: usize = 56;
7const RECORD_DIGEST_OFFSET: usize = 24;
8
9/// Semantic record families carried by the log stream.
10#[derive(Clone, Copy, Debug, PartialEq, Eq)]
11#[repr(u8)]
12pub enum RecordKind {
13    Manifest = 0,
14    CopperList = 1,
15    KeyFrame = 2,
16    StructuredLog = 3,
17    Lifecycle = 4,
18    Gap = 5,
19    Anchor = 6,
20}
21
22impl TryFrom<u8> for RecordKind {
23    type Error = Error;
24
25    fn try_from(value: u8) -> Result<Self> {
26        match value {
27            0 => Ok(Self::Manifest),
28            1 => Ok(Self::CopperList),
29            2 => Ok(Self::KeyFrame),
30            3 => Ok(Self::StructuredLog),
31            4 => Ok(Self::Lifecycle),
32            5 => Ok(Self::Gap),
33            6 => Ok(Self::Anchor),
34            _ => Err(Error::UnknownRecordKind(value)),
35        }
36    }
37}

39/// A verified semantic record borrowing its recovered payload.
40#[derive(Clone, Copy, Debug, PartialEq, Eq)]
41pub struct DecodedRecord<'a> {
42    pub kind: RecordKind,
43    pub object_id: u64,
44    /// Digest binding the record version, kind, identity, length, and payload.
45    pub digest: [u8; 32],
46    pub payload: &'a [u8],
47}
48
49/// Frame and digest one semantic payload before FEC is applied.
50pub fn encode_record(kind: RecordKind, object_id: u64, payload: &[u8]) -> Result<Vec<u8>> {
51    let payload_len = u64::try_from(payload.len())
52        .map_err(|_| Error::InvalidConfig("record payload length exceeds u64"))?;
53    let capacity = RECORD_HEADER_LEN
54        .checked_add(payload.len())
55        .ok_or(Error::InvalidConfig("record length overflow"))?;
56    let mut record = Vec::with_capacity(capacity);
57    record.extend_from_slice(&RECORD_MAGIC);
58    record.push(RECORD_VERSION);
59    record.push(kind as u8);
60    record.extend_from_slice(&0_u16.to_be_bytes());
61    record.extend_from_slice(&object_id.to_be_bytes());
62    record.extend_from_slice(&payload_len.to_be_bytes());
63    record.extend_from_slice(&[0_u8; 32]);
64    record.extend_from_slice(payload);
65
66    let digest = record_digest(kind, object_id, payload_len, payload);
67    record[RECORD_DIGEST_OFFSET..RECORD_HEADER_LEN].copy_from_slice(digest.as_bytes());
68    Ok(record)
69}

71/// Writes a semantic-record header for a payload already stored immediately
72/// after `header` in the caller's record buffer.
73pub(crate) fn encode_record_header(
74    kind: RecordKind,
75    object_id: u64,
76    payload: &[u8],
77    header: &mut [u8],
78) -> Result<()> {
79    if header.len() < RECORD_HEADER_LEN {
80        return Err(Error::BufferTooSmall {
81            needed: RECORD_HEADER_LEN,
82            available: header.len(),
83        });
84    }
85    let payload_len = u64::try_from(payload.len())
86        .map_err(|_| Error::InvalidConfig("record payload length exceeds u64"))?;
87    let header = &mut header[..RECORD_HEADER_LEN];
88    header.fill(0);
89    header[..4].copy_from_slice(&RECORD_MAGIC);
90    header[4] = RECORD_VERSION;
91    header[5] = kind as u8;
92    header[8..16].copy_from_slice(&object_id.to_be_bytes());
93    header[16..24].copy_from_slice(&payload_len.to_be_bytes());
94    let digest = record_digest(kind, object_id, payload_len, payload);
95    header[RECORD_DIGEST_OFFSET..RECORD_HEADER_LEN].copy_from_slice(digest.as_bytes());
96    Ok(())
97}

99/// Validates and decodes one complete semantic record envelope.
100pub fn decode_record(record: &[u8]) -> Result<DecodedRecord<'_>> {
101    if record.len() < RECORD_HEADER_LEN {
102        return Err(Error::TruncatedRecord);
103    }
104    if record[..4] != RECORD_MAGIC {
105        return Err(Error::InvalidMagic);
106    }
107    let version = record[4];
108    if version != RECORD_VERSION {
109        return Err(Error::UnsupportedVersion(version));
110    }
111    let kind = RecordKind::try_from(record[5])?;
112    let object_id = u64::from_be_bytes(record[8..16].try_into().unwrap());
113    let payload_len = u64::from_be_bytes(record[16..24].try_into().unwrap());
114    let payload_len = usize::try_from(payload_len).map_err(|_| Error::RecordLengthMismatch)?;
115    let expected_len = RECORD_HEADER_LEN
116        .checked_add(payload_len)
117        .ok_or(Error::RecordLengthMismatch)?;
118    if record.len() != expected_len {
119        return Err(Error::RecordLengthMismatch);
120    }
121    let payload = &record[RECORD_HEADER_LEN..];
122    let expected_digest = record_digest(kind, object_id, payload_len as u64, payload);
123    if record[RECORD_DIGEST_OFFSET..RECORD_HEADER_LEN] != expected_digest.as_bytes()[..] {
124        return Err(Error::RecordDigestMismatch);
125    }
126    Ok(DecodedRecord {
127        kind,
128        object_id,
129        digest: *expected_digest.as_bytes(),
130        payload,
131    })
132}

134fn record_digest(
135    kind: RecordKind,
136    object_id: u64,
137    payload_len: u64,
138    payload: &[u8],
139) -> blake3::Hash {
140    let mut hasher = blake3::Hasher::new();
141    hasher.update(&[RECORD_VERSION, kind as u8]);
142    hasher.update(&object_id.to_be_bytes());
143    hasher.update(&payload_len.to_be_bytes());
144    hasher.update(payload);
145    hasher.finalize()
146}

148#[cfg(test)]
149mod tests {
150    use super::*;
151
152    #[test]
153    fn record_digest_binds_identity_and_payload() {
154        let encoded = encode_record(RecordKind::KeyFrame, 42, b"state").unwrap();
155        let decoded = decode_record(&encoded).unwrap();
156        assert_eq!(decoded.kind, RecordKind::KeyFrame);
157        assert_eq!(decoded.object_id, 42);
158        assert_eq!(decoded.payload, b"state");
159
160        let mut corrupted = encoded;
161        *corrupted.last_mut().unwrap() ^= 0x80;
162        assert_eq!(decode_record(&corrupted), Err(Error::RecordDigestMismatch));
163    }
