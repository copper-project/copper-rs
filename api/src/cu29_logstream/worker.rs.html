37/// Final counters remain readable after both output sinks have been dropped.
38#[derive(Clone, Debug)]
39pub struct SenderMonitor(Arc<Shared>);
40
41#[derive(Debug, Default)]
42struct Shared {
43    stop: AtomicBool,
44    failed: AtomicBool,
45    inbox_drops: AtomicU64,
46    final_stats: Mutex<Option<SenderStats>>,
47}
48
49impl SenderMonitor {
50    pub fn inbox_drops(&self) -> u64 {
51        self.0.inbox_drops.load(Ordering::Relaxed)
52    }
53    pub fn failed(&self) -> bool {
54        self.0.failed.load(Ordering::Acquire)
55    }
56    pub fn final_stats(&self) -> Option<SenderStats> {
57        *self.0.final_stats.lock().expect("sender stats poisoned")
58    }
59}

61#[derive(Debug)]
62struct Worker {
63    shared: Arc<Shared>,
64    thread: Thread,
65    handle: Option<JoinHandle<()>>,
66}
67
68impl Drop for Worker {
69    fn drop(&mut self) {
70        self.shared.stop.store(true, Ordering::Release);
71        self.thread.unpark();
72        if self
73            .handle
74            .take()
75            .is_some_and(|handle| handle.join().is_err())
76        {
77            self.shared.failed.store(true, Ordering::Release);
78        }
79    }
80}

82#[derive(Debug)]
83struct Inbox {
84    clock: RobotClock,
85    worker: Arc<Worker>,
86    pending: SyncSender<Record>,
87    // Mutex supplies Sync for WriteStream; exclusive log() uses get_mut, never lock().
88    free: Mutex<Receiver<Box<[u8]>>>,
89    recycled: SyncSender<Box<[u8]>>,
90}
91
92impl Inbox {
93    fn submit(
94        &mut self,
95        keyframe: bool,
96        encode: impl FnOnce(&mut [u8]) -> CuResult<usize>,
97    ) -> CuResult<()> {
98        if self.worker.shared.failed.load(Ordering::Acquire) {
99            return Err(CuError::from("Logstream sender worker failed"));
100        }
101        let mut bytes = match self.free.get_mut().expect("exclusive inbox").try_recv() {
102            Ok(bytes) => bytes,
103            Err(TryRecvError::Empty) => {
104                self.worker
105                    .shared
106                    .inbox_drops
107                    .fetch_add(1, Ordering::Relaxed);
108                return Ok(());
109            }
110            Err(TryRecvError::Disconnected) => {
111                return Err(CuError::from("Logstream sender worker stopped"));
112            }
113        };
114        let queued_at = self.clock.now();
115        let len = match encode(&mut bytes) {
116            Ok(len) => len,
117            Err(error) => {
118                let _ = self.recycled.try_send(bytes);
119                return Err(error);
120            }
121        };
122        // Capacity equals the total buffer count, so an owned free buffer always
123        // has a corresponding inbox slot. Neither this nor pool exhaustion waits.
124        self.pending
125            .try_send(Record {
126                bytes,
127                len,
128                keyframe,
129                queued_at,
130            })
131            .map_err(|_| CuError::from("Logstream sender inbox disconnected"))?;
132        self.worker.thread.unpark();
133        Ok(())
134    }
135}

137/// Encodes directly into a sender-owned buffer on the existing CL output worker.
138pub struct ScheduledCopperListSink<P: CopperListTuple> {
139    inbox: Inbox,
140    encoder: fn(&CopperList<P>, &mut [u8]) -> crate::Result<usize>,
141    _payload: PhantomData<fn() -> P>,
142}
143impl<P: CopperListTuple> ScheduledCopperListSink<P> {
144    /// Select the generated capture encoder before handing the sink to its output worker.
145    pub fn with_encoder(
146        mut self,
147        encoder: fn(&CopperList<P>, &mut [u8]) -> crate::Result<usize>,
148    ) -> Self
149    where
150        P: CopperListTuple,
151    {
152        self.encoder = encoder;
153        self
154    }
155}
156impl<P: CopperListTuple> Debug for ScheduledCopperListSink<P> {
157    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
158        f.debug_struct("ScheduledCopperListSink")
159            .finish_non_exhaustive()
160    }
161}
162impl<P: CopperListTuple + Send + Sync> WriteStream<CopperList<P>> for ScheduledCopperListSink<P> {
163    fn log(&mut self, record: &CopperList<P>) -> CuResult<()> {
164        self.inbox.submit(false, |bytes| {
165            (self.encoder)(record, bytes).map_err(|error| CuError::from(error.to_string()))
166        })
167    }
168}

170/// Bounded keyframe encoding; capture objects are released before pacing begins.
171#[derive(Debug)]
172pub struct ScheduledKeyFrameSink {
173    inbox: Inbox,
174    interval: u32,
175}
176impl WriteStream<KeyFrame> for ScheduledKeyFrameSink {
177    fn log(&mut self, keyframe: &KeyFrame) -> CuResult<()> {
178        if !keyframe.culistid.is_multiple_of(u64::from(self.interval)) {
179            return Ok(());
180        }
181        self.inbox.submit(true, |bytes| {
182            if bytes.len() < crate::record::RECORD_HEADER_LEN {
183                return Err(CuError::from("Keyframe buffer too small"));
184            }
185            let (header, payload) = bytes.split_at_mut(crate::record::RECORD_HEADER_LEN);
186            let len = bincode::encode_into_slice(keyframe, payload, bincode::config::standard())
187                .map_err(|error| CuError::from(error.to_string()))?;
188            crate::record::encode_record_header(
189                crate::RecordKind::KeyFrame,
190                keyframe.culistid,
191                &payload[..len],
192                header,
193            )
194            .map_err(|error| CuError::from(error.to_string()))?;
195            Ok(header.len() + len)
196        })
197    }
198}

200/// Start one autonomous worker with an explicitly selected local RobotClock.
201/// A real carrier must use a running clock even when application time is mocked.
202/// Drops of both sinks stop repetition, then drain at most max_latency in local
203/// clock time. No application object, serialization, or wait is added to the RT path.
204pub fn scheduled_sinks<P, T>(
205    transport: T,
206    config: LogStreamSenderConfig,
207    clock: RobotClock,
208) -> CuResult<(
209    ScheduledCopperListSink<P>,
210    ScheduledKeyFrameSink,
211    SenderMonitor,
212)>
213where
214    P: CopperListTuple + Send + Sync,
215    T: CuStreamTx + 'static,
216{
217    let cl_bytes = config.continuous.max_record_bytes;
218    let kf_bytes = usize::try_from(config.recovery.finite.max_object_bytes)
219        .map_err(|_| CuError::from("Keyframe limit exceeds usize"))?;
220    let reserved = cl_bytes
221        .checked_mul(CL_BUFFERS)
222        .and_then(|v| {
223            kf_bytes
224                .checked_mul(KF_BUFFERS)
225                .and_then(|k| v.checked_add(k))
226        })
227        .ok_or_else(|| CuError::from("Sender pool size overflow"))?;
228    let shutdown_clock = if clock.is_mock() {
229        RobotClock::new()
230    } else {
231        clock.clone()
232    };
233    let interval = config.recovery.recovery_interval;
234    let drain_time = config.pacing.max_latency;
235    let mut core = SenderCore::new(config, clock.now(), reserved)
236        .map_err(|error| CuError::from(error.to_string()))?;
237    let (pending, records) = mpsc::sync_channel::<Record>(CL_BUFFERS + KF_BUFFERS);
238    let (cl_return, cl_free) = mpsc::sync_channel(CL_BUFFERS);
239    let (kf_return, kf_free) = mpsc::sync_channel(KF_BUFFERS);
240    for _ in 0..CL_BUFFERS {
241        cl_return
242            .try_send(vec![0; cl_bytes].into_boxed_slice())
243            .expect("new pool");
244    }
245    for _ in 0..KF_BUFFERS {
246        kf_return
247            .try_send(vec![0; kf_bytes].into_boxed_slice())
248            .expect("new pool");
249    }
250    let cl_clock = clock.clone();
251    let kf_clock = clock.clone();
252    let shared = Arc::new(Shared::default());
253    let status = shared.clone();
254    let cl_recycled = cl_return.clone();
255    let kf_recycled = kf_return.clone();
256    let handle = std::thread::Builder::new()
257        .name("cu-logstream".into())
258        .spawn(move || {
259            let mut transport = OneWay::new(transport);
260            let mut stop_at = None;
261            // Teardown remains finite even for deliberately frozen scheduler test clocks.
262            let result = (|| -> crate::Result<()> {
263                loop {
264                    if status.stop.load(Ordering::Acquire) && stop_at.is_none() {
265                        core.begin_shutdown();
266                        stop_at = Some(shutdown_clock.now() + drain_time);
267                    }
268                    let mut received = 0;
269                    for _ in 0..CL_BUFFERS + KF_BUFFERS {
270                        let Ok(record) = records.try_recv() else {
271                            break;
272                        };
273                        let result =
274                            core.accept_record(&record.bytes[..record.len], record.queued_at);
275                        let returned = if record.keyframe {
276                            &kf_return
277                        } else {
278                            &cl_return
279                        };
280                        let _ = returned.try_send(record.bytes);
281                        result?;
282                        received += 1;
283                    }
284                    let next = core.poll(clock.now(), &mut transport)?;
285                    if stop_at.is_some_and(|deadline| shutdown_clock.now() >= deadline)
286                        || (stop_at.is_some() && received == 0 && core.is_idle())
287                    {
288                        break;
289                    }
290                    if received > 0 {
291                        continue;
292                    }
293                    if let Some(deadline) = next {
294                        let now = clock.now();
295                        if deadline > now {
296                            let duration = (deadline - now).min(CuDuration::from_millis(50));
297                            std::thread::park_timeout(std::time::Duration::from_nanos(
298                                duration.as_nanos(),
299                            ));
300                        }
301                    } else {
302                        std::thread::park_timeout(std::time::Duration::from_millis(50));
303                    }
304                }
305                Ok(())
306            })();
307            core.discard_pending();
308            let stats = core.stats();
309            *status.final_stats.lock().expect("sender stats") = Some(stats);
310            cu29_log_derive::info!("Logstream sender stopped: sent={} bytes={} queue_drops={} expired={} transport_drops={} inbox_drops={} shutdown_drops={}",
311                stats.packets_sent, stats.bytes_sent, stats.queue_drops, stats.expired_packets, stats.transport_drops,
312                status.inbox_drops.load(Ordering::Relaxed), stats.shutdown_drops);
313            if let Err(error) = result {
314                status.failed.store(true, Ordering::Release);
315                cu29_log_derive::error!("Logstream sender failed: {}", error.to_string());
316            }
317        })
318        .map_err(|error| CuError::new_with_cause("Start logstream sender", error))?;
319    let worker = Arc::new(Worker {
320        shared: shared.clone(),
321        thread: handle.thread().clone(),
322        handle: Some(handle),
323    });
324    Ok((
325        ScheduledCopperListSink {
326            encoder: crate::encode_copperlist_record_into,
327            inbox: Inbox {
328                clock: cl_clock,
329                worker: worker.clone(),
330                pending: pending.clone(),
331                free: Mutex::new(cl_free),
332                recycled: cl_recycled,
333            },
334            _payload: PhantomData,
335        },
336        ScheduledKeyFrameSink {
337            inbox: Inbox {
338                clock: kf_clock,
339                worker,
340                pending,
341                free: Mutex::new(kf_free),
342                recycled: kf_recycled,
343            },
344            interval,
345        },
346        SenderMonitor(shared),
347    ))
