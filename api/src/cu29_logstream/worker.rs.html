55/// Final counters remain readable after both output sinks have been dropped.
56#[derive(Clone, Debug)]
57pub struct SenderMonitor(Arc<Shared>);

59#[derive(Debug, Default)]
60struct Shared {
61    stop: AtomicBool,
62    failed: AtomicBool,
63    inbox_drops: AtomicU64,
64    final_stats: Mutex<Option<SenderStats>>,
65    snapshot: Mutex<SenderSnapshot>,
66}
67
68impl SenderMonitor {
69    pub fn snapshot(&self) -> SenderSnapshot {
70        let mut snapshot = *self.0.snapshot.lock().expect("sender snapshot poisoned");
71        snapshot.failed |= self.failed();
72        snapshot
73    }
74
75    pub fn inbox_drops(&self) -> u64 {
76        self.0.inbox_drops.load(Ordering::Relaxed)
77    }
78    pub fn failed(&self) -> bool {
79        self.0.failed.load(Ordering::Acquire)
80    }
81    pub fn final_stats(&self) -> Option<SenderStats> {
82        *self.0.final_stats.lock().expect("sender stats poisoned")
83    }
84}

86#[derive(Debug)]
87struct Worker {
88    shared: Arc<Shared>,
89    thread: Thread,
90    handle: Option<JoinHandle<()>>,
91}
92
93impl Drop for Worker {
94    fn drop(&mut self) {
95        self.shared.stop.store(true, Ordering::Release);
96        self.thread.unpark();
97        if self
98            .handle
99            .take()
100            .is_some_and(|handle| handle.join().is_err())
101        {
102            self.shared.failed.store(true, Ordering::Release);
103        }
104    }
105}

107#[derive(Debug)]
108struct Inbox {
109    clock: RobotClock,
110    worker: Arc<Worker>,
111    pending: SyncSender<Record>,
112    // Mutex supplies Sync for WriteStream; exclusive log() uses get_mut, never lock().
113    free: Mutex<Receiver<Box<[u8]>>>,
114    recycled: SyncSender<Box<[u8]>>,
115}
116
117impl Inbox {
118    fn submit(
119        &mut self,
120        keyframe: bool,
121        encode: impl FnOnce(&mut [u8]) -> CuResult<usize>,
122    ) -> CuResult<()> {
123        if self.worker.shared.failed.load(Ordering::Acquire) {
124            return Err(CuError::from("Logstream sender worker failed"));
125        }
126        let mut bytes = match self.free.get_mut().expect("exclusive inbox").try_recv() {
127            Ok(bytes) => bytes,
128            Err(TryRecvError::Empty) => {
129                self.worker
130                    .shared
131                    .inbox_drops
132                    .fetch_add(1, Ordering::Relaxed);
133                return Ok(());
134            }
135            Err(TryRecvError::Disconnected) => {
136                return Err(CuError::from("Logstream sender worker stopped"));
137            }
138        };
139        let queued_at = self.clock.now();
140        let len = match encode(&mut bytes) {
141            Ok(len) => len,
142            Err(error) => {
143                let _ = self.recycled.try_send(bytes);
144                return Err(error);
145            }
146        };
147        // Capacity equals the total buffer count, so an owned free buffer always
148        // has a corresponding inbox slot. Neither this nor pool exhaustion waits.
149        self.pending
150            .try_send(Record {
151                bytes,
152                len,
153                keyframe,
154                queued_at,
155            })
156            .map_err(|_| CuError::from("Logstream sender inbox disconnected"))?;
157        self.worker.thread.unpark();
158        Ok(())
159    }
160}

162/// Encodes directly into a sender-owned buffer on the existing CL output worker.
163pub struct ScheduledCopperListSink<P: CopperListTuple> {
164    inbox: Inbox,
165    encoder: fn(&CopperList<P>, &mut [u8]) -> crate::Result<usize>,
166    _payload: PhantomData<fn() -> P>,
167}
168impl<P: CopperListTuple> ScheduledCopperListSink<P> {
169    /// Select the generated capture encoder before handing the sink to its output worker.
170    pub fn with_encoder(
171        mut self,
172        encoder: fn(&CopperList<P>, &mut [u8]) -> crate::Result<usize>,
173    ) -> Self
174    where
175        P: CopperListTuple,
176    {
177        self.encoder = encoder;
178        self
179    }
180}
181impl<P: CopperListTuple> Debug for ScheduledCopperListSink<P> {
182    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
183        f.debug_struct("ScheduledCopperListSink")
184            .finish_non_exhaustive()
185    }
186}
187impl<P: CopperListTuple + Send + Sync> WriteStream<CopperList<P>> for ScheduledCopperListSink<P> {
188    fn log(&mut self, record: &CopperList<P>) -> CuResult<()> {
189        self.inbox.submit(false, |bytes| {
190            (self.encoder)(record, bytes).map_err(|error| CuError::from(error.to_string()))
191        })
192    }

195/// Bounded keyframe encoding; capture objects are released before pacing begins.
196#[derive(Debug)]
197pub struct ScheduledKeyFrameSink {
198    inbox: Inbox,
199    interval: u32,
200}
201impl WriteStream<KeyFrame> for ScheduledKeyFrameSink {
202    fn log(&mut self, keyframe: &KeyFrame) -> CuResult<()> {
203        if !keyframe.culistid.is_multiple_of(u64::from(self.interval)) {
204            return Ok(());
205        }
206        self.inbox.submit(true, |bytes| {
207            if bytes.len() < crate::record::RECORD_HEADER_LEN {
208                return Err(CuError::from("Keyframe buffer too small"));
209            }
210            let (header, payload) = bytes.split_at_mut(crate::record::RECORD_HEADER_LEN);
211            let len = bincode::encode_into_slice(keyframe, payload, bincode::config::standard())
212                .map_err(|error| CuError::from(error.to_string()))?;
213            crate::record::encode_record_header(
214                crate::RecordKind::KeyFrame,
215                keyframe.culistid,
216                &payload[..len],
217                header,
218            )
219            .map_err(|error| CuError::from(error.to_string()))?;
220            Ok(header.len() + len)
221        })
222    }
223}

262fn scheduled_worker<P, T>(
263    transport: T,
264    config: LogStreamSenderConfig,
265    clock: RobotClock,
266    feedback_enabled: bool,
267) -> CuResult<(
268    ScheduledCopperListSink<P>,
269    ScheduledKeyFrameSink,
270    SenderMonitor,
271)>
272where
273    P: CopperListTuple + Send + Sync,
274    T: CuStreamTx + CuFeedbackRx + 'static,
275{
276    let manifest = crate::SessionManifest::decode_record(&config.recovery.manifest_record);
277    let policy = manifest
278        .as_ref()
279        .ok()
280        .and_then(|manifest| manifest.plan.feedback);
281    if policy.is_some() != feedback_enabled {
282        return Err(CuError::from(
283            "Feedback transport wiring must match manifest capability",
284        ));
285    }
286    if policy.is_some() {
287        let manifest = manifest.as_ref().expect("validated feedback manifest");
288        if manifest.identity != config.continuous.identity
289            || usize::from(manifest.plan.continuous.repair_every_source_symbols)
290                != config.continuous.repair_every_source_symbols
291        {
292            return Err(CuError::from(
293                "Feedback manifest identity and baseline must match the sender",
294            ));
295        }
296    }
297    let mut feedback = policy
298        .map(|policy| {
299            let manifest = manifest.as_ref().expect("validated feedback manifest");
300            FeedbackController::new(
301                policy,
302                config.continuous.identity,
303                destination_key(&manifest.plan.destination_id),
304                config.continuous.repair_every_source_symbols as u16,
305            )
306        })
307        .transpose()
308        .map_err(|error| CuError::from(error.to_string()))?;
309    let cl_bytes = config.continuous.max_record_bytes;
310    let kf_bytes = usize::try_from(config.recovery.finite.max_object_bytes)
311        .map_err(|_| CuError::from("Keyframe limit exceeds usize"))?;
312    let reserved = cl_bytes
313        .checked_mul(CL_BUFFERS)
314        .and_then(|v| {
315            kf_bytes
316                .checked_mul(KF_BUFFERS)
317                .and_then(|k| v.checked_add(k))
318        })
319        .ok_or_else(|| CuError::from("Sender pool size overflow"))?;
320    let shutdown_clock = if clock.is_mock() {
321        RobotClock::new()
322    } else {
323        clock.clone()
324    };
325    let interval = config.recovery.recovery_interval;
326    let drain_time = config.pacing.max_latency;
327    let mut core = SenderCore::new(config, clock.now(), reserved)
328        .map_err(|error| CuError::from(error.to_string()))?;
329    let (pending, records) = mpsc::sync_channel::<Record>(CL_BUFFERS + KF_BUFFERS);
330    let (cl_return, cl_free) = mpsc::sync_channel(CL_BUFFERS);
331    let (kf_return, kf_free) = mpsc::sync_channel(KF_BUFFERS);
332    for _ in 0..CL_BUFFERS {
333        cl_return
334            .try_send(vec![0; cl_bytes].into_boxed_slice())
335            .expect("new pool");
336    }
337    for _ in 0..KF_BUFFERS {
338        kf_return
339            .try_send(vec![0; kf_bytes].into_boxed_slice())
340            .expect("new pool");
341    }
342    let cl_clock = clock.clone();
343    let kf_clock = clock.clone();
344    let shared = Arc::new(Shared::default());
345    *shared.snapshot.lock().expect("new snapshot") = SenderSnapshot {
346        sampled_at: clock.now(),
347        feedback: feedback.as_ref().map(FeedbackController::snapshot),
348        ..Default::default()
349    };
350    let status = shared.clone();
351    let cl_recycled = cl_return.clone();
352    let kf_recycled = kf_return.clone();
353    let handle = std::thread::Builder::new()
354        .name("cu-logstream".into())
355        .spawn(move || {
356            let mut transport = transport;
357            let mut feedback_packet = [0; FEEDBACK_BUFFER_BYTES];
358            let mut feedback_failed = false;
359            let mut next_snapshot = clock.now();
360            let mut stop_at = None;
361            // Teardown remains finite even for deliberately frozen scheduler test clocks.
362            let result = (|| -> crate::Result<()> {
363                loop {
364                    if status.stop.load(Ordering::Acquire) && stop_at.is_none() {
365                        core.begin_shutdown();
366                        stop_at = Some(shutdown_clock.now() + drain_time);
367                    }
368                    if let Some(controller) = &mut feedback {
369                        for _ in 0..8 {
370                            if feedback_failed { break; }
371                            match transport.try_recv_feedback(&mut feedback_packet) {
372                                Ok(Some(len)) => match feedback_packet.get(..len).and_then(|p| ReceiverReport::decode(p).ok()) {
373                                    Some(report) => { if controller.receive(report, clock.now()) { core.request_recovery(); } }
374                                    None => controller.invalid_report(),
375                                },
376                                Ok(None) => break,
377                                Err(CuStreamRxError::BufferTooSmall { .. }) => controller.invalid_report(),
378                                Err(CuStreamRxError::Failed(_)) => { feedback_failed = true; }
379                            }
380                        }
381                        controller.tick(clock.now());
382                        core.set_repair_interval(controller.snapshot().effective_repair_every_source_symbols)?;
383                    }
384                    let mut received = 0;
385                    for _ in 0..CL_BUFFERS + KF_BUFFERS {
386                        let Ok(record) = records.try_recv() else {
387                            break;
388                        };
389                        let result =
390                            core.accept_record(&record.bytes[..record.len], record.queued_at);
391                        let returned = if record.keyframe {
392                            &kf_return
393                        } else {
394                            &cl_return
395                        };
396                        let _ = returned.try_send(record.bytes);
397                        result?;
398                        received += 1;
399                    }
400                    let next = core.poll(clock.now(), &mut transport)?;
401                    let now = clock.now();
402                    if now >= next_snapshot {
403                        if let Ok(mut snapshot) = status.snapshot.try_lock() {
404                            *snapshot = SenderSnapshot { sampled_at: now, stats: core.stats(),
405                                inbox_drops: status.inbox_drops.load(Ordering::Relaxed),
406                                feedback: feedback.as_ref().map(FeedbackController::snapshot), feedback_failed,
407                                ..Default::default() };
408                        }
409                        next_snapshot = now + CuDuration::from_millis(100);
410                    }
411                    if stop_at.is_some_and(|deadline| shutdown_clock.now() >= deadline)
412                        || (stop_at.is_some() && received == 0 && core.is_idle())
413                    {
414                        break;
415                    }
416                    if received > 0 {
417                        continue;
418                    }
419                    if let Some(deadline) = next {
420                        let now = clock.now();
421                        if deadline > now {
422                            let duration = (deadline - now).min(CuDuration::from_millis(50));
423                            std::thread::park_timeout(std::time::Duration::from_nanos(
424                                duration.as_nanos(),
425                            ));
426                        }
427                    } else {
428                        std::thread::park_timeout(std::time::Duration::from_millis(50));
429                    }
430                }
431                Ok(())
432            })();
433            core.discard_pending();
434            let stats = core.stats();
435            *status.final_stats.lock().expect("sender stats") = Some(stats);
436            *status.snapshot.lock().expect("sender snapshot") = SenderSnapshot {
437                sampled_at: clock.now(), stats, inbox_drops: status.inbox_drops.load(Ordering::Relaxed),
438                stopped: true, failed: result.is_err(), feedback_failed,
439                feedback: feedback.as_ref().map(FeedbackController::snapshot),
440            };
441            cu29_log_derive::info!("Logstream sender stopped: sent={} bytes={} queue_drops={} expired={} transport_drops={} inbox_drops={} shutdown_drops={}",
442                stats.packets_sent, stats.bytes_sent, stats.queue_drops, stats.expired_packets, stats.transport_drops,
443                status.inbox_drops.load(Ordering::Relaxed), stats.shutdown_drops);
444            if let Err(error) = result {
445                status.failed.store(true, Ordering::Release);
446                cu29_log_derive::error!("Logstream sender failed: {}", error.to_string());
447            }
448        })
449        .map_err(|error| CuError::new_with_cause("Start logstream sender", error))?;
450    let worker = Arc::new(Worker {
451        shared: shared.clone(),
452        thread: handle.thread().clone(),
453        handle: Some(handle),
454    });
455    Ok((
456        ScheduledCopperListSink {
457            encoder: crate::encode_copperlist_record_into,
458            inbox: Inbox {
459                clock: cl_clock,
460                worker: worker.clone(),
461                pending: pending.clone(),
462                free: Mutex::new(cl_free),
463                recycled: cl_recycled,
464            },
465            _payload: PhantomData,
466        },
467        ScheduledKeyFrameSink {
468            inbox: Inbox {
469                clock: kf_clock,
470                worker,
471                pending,
472                free: Mutex::new(kf_free),
473                recycled: kf_recycled,
474            },
475            interval,
476        },
477        SenderMonitor(shared),
478    ))
479}

509            recovery_superseded: stats.recovery_superseded,
510            queue_peak: stats.queue_peak,
511            stopped: snapshot.stopped,
512            failed: snapshot.failed,
513            feedback: snapshot.feedback.map(|feedback| {
514                let report = feedback.report.unwrap_or_default();
515                LogStreamFeedbackStats {

529                    failed: snapshot.feedback_failed,

537                    finalized_symbols: report.sources.finalized,
538                    loss_basis_points: feedback.source_loss_basis_points,
539                    recovery_basis_points: feedback.source_recovery_basis_points,
540                    buffered_records: report.buffered_records,
541                    record_capacity: report.record_capacity,
542                    latest_copperlist: report.latest_copperlist,
543                    effective_repair_every_source_symbols: feedback
544                        .effective_repair_every_source_symbols,

548                }
