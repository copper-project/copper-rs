22/// Implemented by `#[copper_runtime(..., sim_mode = true)]` with logstream enabled.
23pub trait LiveReplay: Sized + 'static {
24    type DataSet: CaptureDataSet + Send + 'static;
25    fn build_twin() -> CuResult<(Self, RobotClockMock)>;
26    fn stop_twin(&mut self) -> CuResult<()> {
27        Ok(())
28    }
29    fn replay_capture(
30        &mut self,
31        clock: &RobotClockMock,
32        capture: &mut CopperList<Self::DataSet>,
33        keyframe: Option<&KeyFrame>,
34    ) -> CuResult<()>;
35}

38pub enum ReconstructionState {

67/// Synchronous replay engine, also usable for ordinary offline capture replay.
68pub struct LiveTwin<A: LiveReplay> {
69    app: A,
70    clock: RobotClockMock,
71    next: Option<u64>,
72    pub status: TwinStatus,
73}
74impl<A: LiveReplay> LiveTwin<A> {
75    pub fn new() -> CuResult<Self> {
76        let (app, clock) = A::build_twin()?;
77        Ok(Self {
78            app,
79            clock,
80            next: None,
81            status: TwinStatus::default(),
82        })
83    }
84    pub fn recover(&mut self) {
85        self.next = None;
86        self.status.state = ReconstructionState::Recovering;
87    }
88    pub fn reconstruct(
89        &mut self,
90        mut capture: CapturedList<A::DataSet>,
91        keyframe: Option<&KeyFrame>,
92    ) -> CuResult<Option<CopperList<A::DataSet>>> {
93        let id = capture.copperlist.id;
94        if self.next != Some(id) && keyframe.is_none_or(|k| k.culistid != id) {
95            if self.status.state != ReconstructionState::Diverged {
96                self.recover();
97            }
98            return Ok(None);
99        }
100        let result = self
101            .app
102            .replay_capture(&self.clock, &mut capture.copperlist, keyframe);
103        #[cfg(feature = "verify-reconstruction")]
104        let result = result.and_then(|()| {
105            capture
106                .verify()
107                .map_err(|e| cu29_traits::CuError::from(e.to_string()))
108        });
109        if let Err(error) = result {
110            self.next = None;
111            self.status.state = ReconstructionState::Diverged;
112            self.status.divergences += 1;
113            return Err(error);
114        }
115        self.next = id.checked_add(1);
116        self.status.state = ReconstructionState::Reconstructed;
117        #[cfg(feature = "verify-reconstruction")]
118        if capture.digest.is_some() {
119            self.status.state = ReconstructionState::Verified;
120        }
121        self.status.reconstructed += 1;
122        if self.status.state == ReconstructionState::Verified {
123            self.status.verified += 1;
124        }
125        Ok(Some(capture.copperlist))
126    }
127}

135enum Input<P: CopperListTuple> {
136    Capture {
137        capture: CapturedList<P>,
138        identity: StreamIdentity,
139        received_at: Instant,

141    Anchor(KeyFrame),
142}
143struct Work<P: CopperListTuple> {
144    input: Input<P>,
145    generation: u64,
146}
147struct Pending<P: CopperListTuple> {
148    capture: CapturedList<P>,
149    identity: StreamIdentity,
150    received_at: Instant,

153/// Nonblocking archive-to-replay handoff. The worker consumes every admitted
154/// iteration; only successfully reconstructed results enter the UI ring.
155/// Payload verification is opt-in through `verify-reconstruction`.
156/// Storage bounds are `capacity` queued events, `capacity` pending captures,
157/// one pending anchor and one executing frame, in addition to the telemetry ring.
158/// Overflow invalidates replay until an admitted matching anchor, never archival.
159#[doc(hidden)]
160pub struct TwinWorker<P: CopperListTuple, S> {
161    sender: Option<SyncSender<Work<P>>>,
162    statuses: Arc<ArrayQueue<S>>,
163    generation: Arc<AtomicU64>,
164    overflows: Arc<AtomicU64>,
165    worker: Option<JoinHandle<TwinStatus>>,
166}
167
168impl<P: CaptureDataSet + Send + 'static, S: TwinReceiverStatus> TwinWorker<P, S> {
169    pub fn spawn<A: LiveReplay<DataSet = P>>(
170        capacity: NonZeroUsize,
171        mut publisher: TelemetryPublisher<TwinFrame<P>, S>,
172    ) -> CuResult<Self> {
173        let (sender, receiver) = mpsc::sync_channel::<Work<P>>(capacity.get());
174        let statuses = Arc::new(ArrayQueue::new(1));
175        let pending_status = statuses.clone();
176        let generation = Arc::new(AtomicU64::new(0));
177        let epoch = generation.clone();
178        let overflows = Arc::new(AtomicU64::new(0));
179        let drops = overflows.clone();
180        let (ready_tx, ready_rx) = mpsc::sync_channel(1);
181        let worker = thread::Builder::new()
182            .name("copper-live-twin".into())
183            .spawn(move || {
184                let mut twin = match LiveTwin::<A>::new() {
185                    Ok(twin) => {
186                        let _ = ready_tx.send(Ok(()));
187                        twin
188                    }
189                    Err(e) => {
190                        let _ = ready_tx.send(Err(e));
191                        return TwinStatus::default();
192                    }
193                };
194                let mut current_generation = 0;
195                let mut pending: std::collections::VecDeque<Pending<P>> =
196                    std::collections::VecDeque::with_capacity(capacity.get());
197                let mut anchor: Option<KeyFrame> = None;
198                let mut receiver_status = publisher.status();
199                loop {
200                    if let Some(status) = pending_status.pop() {
201                        receiver_status = status;
202                    }
203                    let latest_generation = epoch.load(Ordering::Acquire);
204                    if current_generation != latest_generation {
205                        twin.recover();
206                        pending.clear();
207                        anchor = None;
208                        current_generation = latest_generation;
209                    }
210                    twin.status.queue_overflows = drops.load(Ordering::Relaxed);
211                    publisher.set_status(receiver_status.with_twin(twin.status));
212                    match receiver.recv_timeout(Duration::from_millis(10)) {
213                        Ok(work) => {
214                            let latest = epoch.load(Ordering::Acquire);
215                            if current_generation != latest {
216                                twin.recover();
217                                pending.clear();
218                                anchor = None;
219                                current_generation = latest;
220                            }
221                            if work.generation != current_generation {
222                                continue;
223                            }
224                            match work.input {
225                                Input::Capture {
226                                    capture,
227                                    identity,
228                                    received_at,
229                                } => {
230                                    if pending.len() == capacity.get() {
231                                        pending.pop_front();
232                                        drops.fetch_add(1, Ordering::Relaxed);
233                                        twin.recover();
234                                    }
235                                    pending.push_back(Pending {
236                                        capture,
237                                        identity,
238                                        received_at,
239                                    });
240                                }
241                                Input::Anchor(frame) => {
242                                    if anchor
243                                        .as_ref()
244                                        .is_none_or(|old| old.culistid < frame.culistid)
245                                    {
246                                        anchor = Some(frame);
247                                    }
248                                }
249                            }
250                            while let Some(first) = pending.front() {
251                                if twin.next != Some(first.capture.copperlist.id) {
252                                    let Some(frame) = &anchor else {
253                                        break;
254                                    };
255                                    let Some(position) = pending.iter().position(|item| {
256                                        item.capture.copperlist.id == frame.culistid
257                                    }) else {
258                                        break;
259                                    };
260                                    pending.drain(..position);
261                                }
262                                let item = pending.pop_front().unwrap();
263                                let keyframe = if anchor.as_ref().is_some_and(|frame| {
264                                    frame.culistid == item.capture.copperlist.id
265                                }) {
266                                    anchor.take()
267                                } else {
268                                    None
269                                };
270                                if let Ok(Some(copperlist)) =
271                                    twin.reconstruct(item.capture, keyframe.as_ref())
272                                    && epoch.load(Ordering::Acquire) == current_generation
273                                {
274                                    publisher.publish(TwinFrame {
275                                        identity: item.identity,
276                                        received_at: item.received_at,
277                                        copperlist,
278                                    });
279                                }
280                            }
281                        }
282                        Err(mpsc::RecvTimeoutError::Timeout) => {}
283                        Err(mpsc::RecvTimeoutError::Disconnected) => break,
284                    }
285                }
286                if let Some(status) = pending_status.pop() {
287                    receiver_status = status;
288                }
289                publisher.set_status(receiver_status.with_twin(twin.status));
290                twin.status
291            })
292            .map_err(|e| cu29_traits::CuError::new_with_cause("spawn live twin", e))?;
293        ready_rx
294            .recv()
295            .map_err(|_| cu29_traits::CuError::from("live twin initialization failed"))??;
296        Ok(Self {
297            sender: Some(sender),
298            statuses,
299            generation,
300            overflows,
301            worker: Some(worker),
302        })
303    }
304    pub fn finish(mut self) -> CuResult<TwinStatus> {
305        self.sender.take();
306        self.worker
307            .take()
308            .ok_or_else(|| cu29_traits::CuError::from("Twin worker already joined"))?
309            .join()
310            .map_err(|_| cu29_traits::CuError::from("Twin replay worker panicked"))
311    }

313    /// Route an already archived event into live replay. This is the usual
314    /// ground-station integration: `twin.accept(event, archive.accept(event)?)`.
315    /// Archival must succeed first; replay never retries or delays that writer.
316    pub fn accept(
317        &self,
318        event: &crate::SessionEvent,
319        capture: Option<CapturedList<P>>,
320    ) -> crate::Result<()> {
321        match event {
322            crate::SessionEvent::Manifest(_) | crate::SessionEvent::Gap { .. } => self.recover(),
323            crate::SessionEvent::VerifiedAnchor { keyframe, .. } => {
324                self.anchor(crate::decode_keyframe(keyframe.decoded()?.payload)?)
325            }
326            crate::SessionEvent::ContinuousRecord { identity, .. } => {
327                if let Some(capture) = capture {
328                    self.submit(capture, *identity, Instant::now());
329                }
330            }
331            _ => {}
332        }
333        Ok(())
334    }
335    pub fn set_status(&self, status: S) {
336        self.statuses.force_push(status);
337    }
338    pub fn recover(&self) {
339        self.generation.fetch_add(1, Ordering::AcqRel);
340    }
341    pub fn submit(&self, capture: CapturedList<P>, identity: StreamIdentity, received_at: Instant) {
342        self.send(Input::Capture {
343            capture,
344            identity,
345            received_at,
346        });
347    }
348    /// Accept only keyframes from the session router's VerifiedAnchor event.
349    /// Control objects may arrive before or after their captured boundary.
350    pub fn anchor(&self, frame: KeyFrame) {
351        self.send(Input::Anchor(frame));
352    }
353    fn send(&self, input: Input<P>) {
354        let work = Work {
355            input,
356            generation: self.generation.load(Ordering::Acquire),
357        };
358        if let Err(TrySendError::Full(_)) = self.sender.as_ref().unwrap().try_send(work) {
359            self.overflows.fetch_add(1, Ordering::Relaxed);
360            self.recover();
361        }
362    }
363}
364impl<P: CopperListTuple, S> Drop for TwinWorker<P, S> {
365    fn drop(&mut self) {
366        self.sender.take();
367        if let Some(worker) = self.worker.take() {
368            let _ = worker.join();
369        }
370    }
