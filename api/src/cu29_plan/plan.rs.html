67fn run(args: Args) -> CuResult<()> {
68    let feature_refs = args.features.iter().map(String::as_str).collect::<Vec<_>>();
69    let config = load_single_config(&args.config, &feature_refs)?;
70    if args.list_missions {
71        for (mission, _) in mission_graphs(&config) {
72            println!("{mission}");
73        }
74        return Ok(());
75    }
76
77    let sections = selected_graphs(&config, args.mission.as_deref())?;
78    let svg = render_document(&config, &sections)?;
79    fs::write(&args.output, svg).map_err(|error| {
80        CuError::new_with_cause(
81            &format!("Failed to write plan SVG '{}'.", args.output.display()),
82            error,
83        )
84    })?;
85    if args.open {
86        open_svg(&args.output).map_err(|error| {
87            CuError::new_with_cause(
88                &format!("Failed to open plan SVG '{}'.", args.output.display()),
89                error,
90            )
91        })?;
92    }
93    Ok(())
94}

96fn load_single_config(path: &Path, features: &[&str]) -> CuResult<CuConfig> {
97    let filename = path.to_str().ok_or_else(|| {
98        CuError::from(format!(
99            "Config path '{}' is not valid UTF-8.",
100            path.display()
101        ))
102    })?;
103    if read_multi_configuration_with_features(filename, features).is_ok() {
104        return Err(CuError::from(format!(
105            "Multi-Copper scheduling plans are not supported yet: '{}'. Render each subsystem's copperconfig.ron separately.",
106            path.display()
107        )));
108    }
109    read_configuration_with_features(filename, features).map_err(|error| {
110        CuError::from(format!(
111            "Failed to read Copper config '{}': {error}",
112            path.display()
113        ))
114    })
115}

117fn selected_graphs<'a>(
118    config: &'a CuConfig,
119    requested: Option<&str>,
120) -> CuResult<Vec<(String, &'a CuGraph)>> {
121    let all = mission_graphs(config);
122    let Some(requested) = requested else {
123        return Ok(all);
124    };
125    all.into_iter()
126        .find(|(mission, _)| mission == requested)
127        .map(|section| vec![section])
128        .ok_or_else(|| {
129            let available = mission_graphs(config)
130                .into_iter()
131                .map(|(mission, _)| mission)
132                .collect::<Vec<_>>()
133                .join(", ");
134            CuError::from(format!(
135                "Mission '{requested}' not found. Available missions: {available}"
136            ))
137        })
138}

206fn render_mission(
207    config: &CuConfig,
208    mission: &str,
209    plan: &AssembledPlan,
210) -> CuResult<RenderedSection> {
211    let steps = plan
212        .execution
213        .steps
214        .iter()
215        .map(|unit| match unit {
216            CuExecutionUnit::Step(step) => Ok(step.as_ref()),
217            CuExecutionUnit::Loop(_) => Err(CuError::from(
218                "Nested execution loops are not supported by the plan visualizer.",
219            )),
220        })
221        .collect::<CuResult<Vec<_>>>()?;
222    let stages = steps
223        .iter()
224        .copied()
225        .filter(|step| step.phase != CuStepPhase::AnytimeRefine)
226        .collect::<Vec<_>>();
227    let message_slots = steps
228        .iter()
229        .filter_map(|step| step.output_msg_pack.as_ref())
230        .map(|pack| pack.culist_index as usize + 1)
231        .max()
232        .unwrap_or(0);
233    let in_flight_limit = config
234        .logging
235        .as_ref()
236        .and_then(|logging| logging.copperlist_count)
237        .unwrap_or(DEFAULT_COPPERLIST_COUNT);
238    let refine_totals = refine_totals(&steps);
239
240    let runtime = config.runtime.as_ref();
241    let rate = runtime
242        .and_then(|runtime| runtime.rate_target_hz)
243        .map(|rate| format!("{rate} Hz"))
244        .unwrap_or_else(|| "best effort".to_string());
245    let pools = runtime
246        .map(|runtime| runtime.thread_pools.as_slice())
247        .unwrap_or_default();
248
249    let mut svg = String::new();
250    writeln!(
251        svg,
252        r#"<text class="title" x="0" y="22">Mission: {}</text>"#,
253        xml(mission)
254    )
255    .unwrap();
256    writeln!(
257        svg,
258        r#"<text class="meta" x="0" y="43">{} serial steps · {} parallel stages · {} message slots per CopperList · {} CopperLists max in flight · rate target: {}</text>"#,
259        steps.len(),
260        stages.len(),
261        message_slots,
262        in_flight_limit,
263        xml(&rate)
264    )
265    .unwrap();
266
267    let mut y = 61.0;
268    if pools.is_empty() {
269        svg.push_str(r#"<text class="meta" x="0" y="61">Pools: none configured</text>"#);
270        y += 18.0;
271    } else {
272        for pool in pools {
273            writeln!(
274                svg,
275                r#"<text class="meta mono" x="0" y="{y}">pool {}</text>"#,
276                xml(&format_pool(pool))
277            )
278            .unwrap();
279            y += 16.0;
280        }
281    }
282
283    y += 14.0;
284    writeln!(
285        svg,
286        r#"<text class="subtitle" x="0" y="{y}">Serial projection · main CopperList executor · 1 worker</text>"#
287    )
288    .unwrap();
289    y += 17.0;
290    let serial = render_serial(config, mission, &steps, &plan.entities, &refine_totals, y);
291    svg.push_str(&serial.svg);
292    y += serial.height + 26.0;
293    writeln!(
294        svg,
295        r#"<text class="subtitle" x="0" y="{y}">Parallel projection · generated stage workers + background pools</text>"#
296    )
297    .unwrap();
298    y += 17.0;
299    let parallel = render_parallel(
300        config,
301        &stages,
302        &plan.entities,
303        &refine_totals,
304        in_flight_limit,
305        y,
306    );
307    svg.push_str(&parallel.svg);
308    y += parallel.height;
309
310    Ok(RenderedSection {
311        svg,
312        height: y + 10.0,
313    })
314}

316fn refine_totals(steps: &[&CuExecutionStep]) -> HashMap<u32, u32> {
317    let mut totals = HashMap::new();
318    for step in steps {
319        if step.phase == CuStepPhase::AnytimeRefine {
320            *totals.entry(step.node_id).or_insert(0) += 1;
321        }
322    }
323    totals
324}

326fn render_serial(
327    config: &CuConfig,
328    mission: &str,
329    steps: &[&CuExecutionStep],
330    entities: &[PlanEntity],
331    refine_totals: &HashMap<u32, u32>,
332    top: f64,
333) -> RenderedSection {
334    let mut svg = String::new();
335    let background_pools = background_pool_views(config, steps);
336    let cycles = if background_pools.is_empty() { 1 } else { 2 };
337    let column_count = (steps.len() * cycles).max(1);
338    let mut y = top;
339
340    for block_start in (0..column_count).step_by(WAVE_COLUMNS) {
341        let block_end = (block_start + WAVE_COLUMNS).min(column_count);
342        let row_y = y + 19.0;
343        writeln!(
344            svg,
345            r#"<text class="lane mono" x="0" y="{}">MAIN w1</text>"#,
346            row_y + WAVE_CELL_HEIGHT / 2.0 + 4.0
347        )
348        .unwrap();
349
350        for column in block_start..block_end {
351            let local_column = column - block_start;
352            let x = LABEL_WIDTH + local_column as f64 * (WAVE_CELL_WIDTH + WAVE_CELL_GAP);
353            let cycle = column / steps.len().max(1);
354            let step_index = column % steps.len().max(1);
355            let Some(step) = steps.get(step_index).copied() else {
356                continue;
357            };
358            let entity = &entities[step.node_id as usize];
359            let is_background = step.node.is_background();
360            let (_, entity_fill) = entity_style(entity, step);
361            let fill = if is_background {
362                BACKGROUND_GATEWAY_COLOR
363            } else {
364                entity_fill
365            };
366            let cl_label = if cycle == 0 {
367                "CL n".to_string()
368            } else {
369                format!("CL n+{cycle}")
370            };
371            let refine_ordinal = if step.phase == CuStepPhase::AnytimeRefine {
372                Some(
373                    steps[..=step_index]
374                        .iter()
375                        .filter(|candidate| {
376                            candidate.node_id == step.node_id
377                                && candidate.phase == CuStepPhase::AnytimeRefine
378                        })
379                        .count() as u32,
380                )
381            } else {
382                None
383            };
384            let phase = match step.phase {
385                CuStepPhase::Whole if is_background => "POLL / DISPATCH?".to_string(),
386                CuStepPhase::Whole => "whole".to_string(),
387                CuStepPhase::AnytimeBase => "base".to_string(),
388                CuStepPhase::AnytimeRefine => format!(
389                    "refine {}/{}",
390                    refine_ordinal.unwrap_or_default(),
391                    refine_totals.get(&step.node_id).copied().unwrap_or(0)
392                ),
393            };
394            let resources = entity_resources(config, entity, step);
395            let footer = if is_background {
396                format!("→ pool {}", step.node.background_pool())
397            } else if resources.is_empty() {
398                "res —".to_string()
399            } else {
400                format!("res {}", resources.join(", "))
401            };
402            let key = step_key(mission, entity, step.phase, refine_ordinal);
403            let tooltip = format!(
404                "{} · main executor worker 1\nStep {}: {}\nphase: {}\nresources: {}\nstable key: {}",
405                cl_label,
406                step_index + 1,
407                entity.label,
408                phase,
409                if resources.is_empty() {
410                    "—".to_string()
411                } else {
412                    resources.join(", ")
413                },
414                key
415            );
416            writeln!(
417                svg,
418                r##"<text class="ordinal mono" x="{x}" y="{}">{} · S{:02}</text><g data-step-key="{}" data-serial-column="{}"{}><title>{}</title><rect x="{x}" y="{row_y}" width="{WAVE_CELL_WIDTH}" height="{WAVE_CELL_HEIGHT}" rx="6" fill="{fill}" stroke="{}"{}/><text class="ordinal mono" x="{}" y="{}">MAIN w1 · {}</text><text class="card-title" x="{}" y="{}">{}</text><text class="card-line" x="{}" y="{}">{}</text><text class="card-line mono" x="{}" y="{}">{}</text></g>"##,
419                row_y - 7.0,
420                xml(&cl_label),
421                step_index + 1,
422                xml_attr(&key),
423                column + 1,
424                if is_background { r#" data-background-gateway="true""# } else { "" },
425                xml(&tooltip),
426                if is_background { "#7f56d9" } else { "#98a2b3" },
427                if is_background { r#" stroke-width="2" stroke-dasharray="4 2""# } else { "" },
428                x + 7.0,
429                row_y + 13.0,
430                if is_background { "GATEWAY" } else { "SERIAL" },
431                x + 7.0,
432                row_y + 29.0,
433                xml(&truncate(&entity.label, 16)),
434                x + 7.0,
435                row_y + 45.0,
436                xml(&truncate(&phase, 18)),
437                x + 7.0,
438                row_y + 62.0,
439                xml(&truncate(&footer, 18)),
440            )
441            .unwrap();
442
443            if column + 1 < block_end {
444                writeln!(
445                    svg,
446                    r#"<line class="flow" x1="{}" y1="{}" x2="{}" y2="{}"/>"#,
447                    x + WAVE_CELL_WIDTH,
448                    row_y + WAVE_CELL_HEIGHT / 2.0,
449                    x + WAVE_CELL_WIDTH + WAVE_CELL_GAP,
450                    row_y + WAVE_CELL_HEIGHT / 2.0,
451                )
452                .unwrap();
453            }
454        }
455
456        let pools_top = row_y + WAVE_CELL_HEIGHT + WAVE_CELL_GAP;
457        let mut triggers = Vec::new();
458        for column in block_start..block_end {
459            let step_index = column % steps.len().max(1);
460            let Some(step) = steps.get(step_index).copied() else {
461                continue;
462            };
463            if !step.node.is_background() {
464                continue;
465            }
466            let cycle = column / steps.len().max(1);
467            triggers.push(BackgroundTrigger {
468                column,
469                stage_index: step_index,
470                step,
471                entity: &entities[step.node_id as usize],
472                source_y: row_y + WAVE_CELL_HEIGHT,
473                cl_label: if cycle == 0 {
474                    "CL n".to_string()
475                } else {
476                    format!("CL n+{cycle}")
477                },
478            });
479        }
480        let background = render_background_lanes(
481            steps,
482            entities,
483            &background_pools,
484            &triggers,
485            block_start,
486            block_end,
487            pools_top,
488        );
489        svg.push_str(&background.svg);
490        y = pools_top + background.height + 26.0;
491    }
492
493    RenderedSection {
494        svg,
495        height: y - top,
496    }
497}

522fn render_staggered_wavefront(
523    config: &CuConfig,
524    stages: &[&CuExecutionStep],
525    entities: &[PlanEntity],
526    refine_totals: &HashMap<u32, u32>,
527    in_flight_limit: usize,
528    rt_pool: Option<&ThreadPoolConfig>,
529    top: f64,
530) -> RenderedSection {
531    let mut svg = String::new();
532    let background_pools = background_pool_views(config, stages);
533    let active_lanes = in_flight_limit.min(stages.len()).max(1);
534    let visible_lanes = active_lanes.min(MAX_VISIBLE_COPPERLISTS);
535    let formation_count = stages
536        .len()
537        .saturating_add(visible_lanes)
538        .saturating_sub(1)
539        .max(1);
540
541    writeln!(
542        svg,
543        r#"<text class="subtitle" x="0" y="{}">Staggered concurrency formations · cards in one column can run together</text>"#,
544        top + 14.0
545    )
546    .unwrap();
547    writeln!(
548        svg,
549        r#"<text class="meta" x="0" y="{}">Columns can run together. Purple cells are RT poll/dispatch gateways; arrows feed the aligned background-pool worker lanes.</text>"#,
550        top + 33.0
551    )
552    .unwrap();
553    let depth_note = if active_lanes > visible_lanes {
554        format!(
555            "Configured in-flight depth: {in_flight_limit}. Showing CL n through CL n+{} ({} of {active_lanes} active pipeline positions); the overlap audit still covers every stage.",
556            visible_lanes - 1,
557            visible_lanes,
558        )
559    } else if in_flight_limit < MAX_VISIBLE_COPPERLISTS {
560        format!(
561            "Configured in-flight depth: {in_flight_limit}. CL n+{in_flight_limit} cannot be admitted until an older CopperList is committed and recycled."
562        )
563    } else {
564        format!(
565            "Configured in-flight depth: {in_flight_limit}. The matrix shows every concurrently active worker lane allowed by that bound."
566        )
567    };
568    writeln!(
569        svg,
570        r#"<text class="meta mono" x="0" y="{}">{}</text>"#,
571        top + 51.0,
572        xml(&depth_note)
573    )
574    .unwrap();
575
576    let mut y = top + 76.0;
577    for block_start in (0..formation_count).step_by(WAVE_COLUMNS) {
578        let block_end = (block_start + WAVE_COLUMNS).min(formation_count);
579        for formation in block_start..block_end {
580            let column = formation - block_start;
581            let x = LABEL_WIDTH + column as f64 * (WAVE_CELL_WIDTH + WAVE_CELL_GAP);
582            let collisions =
583                formation_collisions(config, stages, entities, visible_lanes, formation);
584            writeln!(
585                svg,
586                r#"<text class="ordinal mono" x="{x}" y="{y}"{}>FORMATION {:02}{}</text>"#,
587                if collisions.is_empty() {
588                    ""
589                } else {
590                    r##" fill="#b42318""##
591                },
592                formation + 1,
593                if collisions.is_empty() { "" } else { " ⚠" }
594            )
595            .unwrap();
596        }
597
598        let rows_top = y + 11.0;
599        for lane in 0..visible_lanes {
600            let row_y = rows_top + lane as f64 * (WAVE_CELL_HEIGHT + WAVE_CELL_GAP);
601            writeln!(
602                svg,
603                r#"<text class="lane mono" x="0" y="{}">CL n{}</text>"#,
604                row_y + WAVE_CELL_HEIGHT / 2.0 + 4.0,
605                if lane == 0 {
606                    String::new()
607                } else {
608                    format!("+{lane}")
609                }
610            )
611            .unwrap();
612
613            // Connect consecutive occupied cells before drawing the cards so
614            // each CopperList's diagonal progression remains visually explicit.
615            for formation in (block_start + 1)..block_end {
616                let Some(previous_stage) = (formation - 1).checked_sub(lane) else {
617                    continue;
618                };
619                let Some(current_stage) = formation.checked_sub(lane) else {
620                    continue;
621                };
622                if previous_stage >= stages.len() || current_stage >= stages.len() {
623                    continue;
624                }

876fn background_pool_views<'a>(
877    config: &'a CuConfig,
878    stages: &[&CuExecutionStep],
879) -> Vec<BackgroundPoolView<'a>> {
880    let mut assignments = BTreeMap::<String, Vec<usize>>::new();
881    for (stage_index, step) in stages.iter().enumerate() {
882        if step.node.is_background() {
883            assignments
884                .entry(step.node.background_pool().to_string())
885                .or_default()
886                .push(stage_index);
887        }
888    }
889    let configured = config
890        .runtime
891        .as_ref()
892        .map(|runtime| runtime.thread_pools.as_slice())
893        .unwrap_or_default();
894    assignments
895        .into_iter()
896        .map(|(id, stage_indices)| BackgroundPoolView {
897            config: configured.iter().find(|pool| pool.id == id),
898            id,
899            stage_indices,
900        })
901        .collect()
902}

1063fn formation_collisions(
1064    config: &CuConfig,
1065    stages: &[&CuExecutionStep],
1066    entities: &[PlanEntity],
1067    visible_lanes: usize,
1068    formation: usize,
1069) -> HashMap<usize, Vec<String>> {
1070    let mut target_lanes = BTreeMap::<String, BTreeSet<usize>>::new();
1071    for lane in 0..visible_lanes {
1072        let Some(stage_index) = formation.checked_sub(lane) else {
1073            continue;
1074        };
1075        let Some(step) = stages.get(stage_index) else {
1076            continue;
1077        };
1078        // A background stage is only a state-check/dispatch gateway. Its
1079        // declared resources belong to the asynchronous job, whose lifetime
1080        // is independent of this nominal RT formation and is audited below.
1081        if step.node.is_background() {
1082            continue;
1083        }
1084        let entity = &entities[step.node_id as usize];
1085        for (_, target) in entity_resource_bindings(config, entity, step) {
1086            target_lanes.entry(target).or_default().insert(lane);
1087        }
1088    }
1089
1090    let mut collisions = HashMap::<usize, Vec<String>>::new();
1091    for (target, lanes) in target_lanes {
1092        if lanes.len() < 2 {
1093            continue;
1094        }
1095        for lane in lanes {
1096            collisions.entry(lane).or_default().push(target.clone());
1097        }
1098    }
1099    collisions
1100}

1102fn resource_overlap_groups(
1103    config: &CuConfig,
1104    stages: &[&CuExecutionStep],
1105    entities: &[PlanEntity],
1106    in_flight_limit: usize,
1107) -> Vec<(String, Vec<String>)> {
1108    let mut target_stages = BTreeMap::<String, BTreeSet<usize>>::new();
1109    for (stage_index, step) in stages.iter().enumerate() {

1112            target_stages.entry(target).or_default().insert(stage_index);
1113        }
1114    }
1115
1116    target_stages
1117        .into_iter()
1118        .filter(|(_, stage_indices)| {
1119            stage_indices.len() > 1
1120                && (in_flight_limit >= 2
1121                    || stage_indices
1122                        .iter()
1123                        .any(|stage_index| stages[*stage_index].node.is_background()))
1124        })
1125        .map(|(target, stage_indices)| {
1126            let labels = stage_indices
1127                .into_iter()
1128                .map(|stage_index| {
1129                    let step = stages[stage_index];
1130                    let entity = &entities[step.node_id as usize];
1131                    format!(
1132                        "S{:02} {}{}",
1133                        stage_index + 1,
1134                        entity.label,
1135                        if step.node.is_background() {
1136                            " (background job)"
1137                        } else {
1138                            ""
1139                        }
1140                    )
1141                })
1142                .collect();
1143            (target, labels)
1144        })
1145        .collect()
1146}

1182    entity: &PlanEntity,
1183    step: &CuExecutionStep,

1187        PlanEntityKind::BridgeRx {

1220    format!(
1221        "{}: threads={}, affinity={}, policy={}, on_error={}",
1222        pool.id,
1223        pool.threads,
1224        affinity,
1225        format_policy(pool.policy),
1226        match pool.on_error {
1227            OnError::Warn => "Warn",
1228            OnError::Strict => "Strict",
1229        }
1230    )
1231}

1233fn format_policy(policy: SchedulingPolicy) -> String {
1234    match policy {
1235        SchedulingPolicy::Fair => "Fair".to_string(),
1236        SchedulingPolicy::Nice(value) => format!("Nice({value})"),
1237        SchedulingPolicy::Fifo { priority } => format!("Fifo(priority: {priority})"),
1238        SchedulingPolicy::RoundRobin { priority } => {
1239            format!("RoundRobin(priority: {priority})")
1240        }
1241    }
1242}

1244fn truncate(value: &str, max_chars: usize) -> String {
1245    let mut chars = value.chars();
1246    let prefix = chars.by_ref().take(max_chars).collect::<String>();
1247    if chars.next().is_some() {
1248        format!("{prefix}…")
1249    } else {
1250        prefix
1251    }
1252}
