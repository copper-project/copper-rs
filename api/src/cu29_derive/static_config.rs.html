7fn value(v: &Value) -> TokenStream {
8    let ty = quote!(::cu29::__private::StaticValue);
9    match v {
10        Value::Bool(x) => quote!(#ty::Bool(#x)),
11        Value::Char(x) => quote!(#ty::Char(#x)),
12        Value::String(x) => quote!(#ty::String(#x)),
13        Value::Bytes(xs) => quote!(#ty::Bytes(&[#(#xs),*])),
14        Value::Unit => quote!(#ty::Unit),
15        Value::Option(None) => quote!(#ty::Option(None)),
16        Value::Option(Some(x)) => {
17            let x = value(x);
18            quote!(#ty::Option(Some(&#x)))
19        }
20        Value::Seq(xs) => {
21            let xs = xs.iter().map(value);
22            quote!(#ty::Seq(&[#(#xs),*]))
23        }
24        Value::Map(xs) => {
25            let xs = xs.iter().map(|(k, v)| {
26                let k = value(k);
27                let v = value(v);
28                quote!((#k,#v))
29            });
30            quote!(#ty::Map(&[#(#xs),*]))
31        }
32        Value::Number(n) => match n {
33            Number::I8(x) => quote!(#ty::I8(#x)),
34            Number::I16(x) => quote!(#ty::I16(#x)),
35            Number::I32(x) => quote!(#ty::I32(#x)),
36            Number::I64(x) => quote!(#ty::I64(#x)),
37            Number::U8(x) => quote!(#ty::U8(#x)),
38            Number::U16(x) => quote!(#ty::U16(#x)),
39            Number::U32(x) => quote!(#ty::U32(#x)),
40            Number::U64(x) => quote!(#ty::U64(#x)),
41            Number::F32(x) => {
42                let bits = x.0.to_bits();
43                quote!(#ty::F32(f32::from_bits(#bits)))
44            }
45            Number::F64(x) => {
46                let bits = x.0.to_bits();
47                quote!(#ty::F64(f64::from_bits(#bits)))
48            }
49            _ => quote!(compile_error!("unsupported static-config number")),
50        },
51    }
52}

54pub(super) fn component(config: Option<&Config<OwnedStorage>>) -> TokenStream {
55    let Some(config) = config else {
56        return quote!(None);
57    };
58    let mut entries: Vec<_> = config.entries().collect();
59    entries.sort_by_key(|(key, _)| *key);
60    let entries = entries.iter().map(|(key, v)| {
61        let v = value(v.as_ron());
62        quote!((#key,#v))
63    });
64    quote!(Some(::cu29::config::ComponentConfig::from_static(&[#(#entries),*])))
65}

67pub(super) fn topology(
68    config: &cu29_runtime::config::CuConfig,
69    mission: &str,
70) -> cu29_traits::CuResult<TokenStream> {
71    use cu29_runtime::monitoring::build_monitor_topology;
72    let mut topology = build_monitor_topology(config, mission)?;
73    topology.nodes.sort_by(|a, b| a.id.cmp(&b.id));
74    let m = quote!(::cu29::monitoring);
75    let optional = |s: Option<&str>| match s {
76        Some(s) => quote!(Some(#s.into())),
77        None => quote!(None),
78    };
79    let nodes = topology.nodes.iter().map(|n| {
80        let id = &n.id;
81        let name = optional(n.type_name.as_deref());
82        let kind = syn::Ident::new(&format!("{:?}", n.kind), proc_macro2::Span::call_site());
83        let inputs = &n.inputs;
84        let outputs = &n.outputs;
85        quote!(#m::MonitorNode { id: #id.into(), type_name: #name, kind: #m::ComponentType::#kind,
86            inputs: vec![#(#inputs.into()),*], outputs: vec![#(#outputs.into()),*] })
87    });
88    let connections = topology.connections.iter().map(|c| {
89        let (src,dst,msg) = (&c.src,&c.dst,&c.msg);
90        let src_port = optional(c.src_port.as_deref());
91        let dst_port = optional(c.dst_port.as_deref());
92        quote!(#m::MonitorConnection { src: #src.into(), dst: #dst.into(), msg: #msg.into(), src_port: #src_port, dst_port: #dst_port })
93    });
94    Ok(
95        quote!(#m::MonitorTopology { nodes: vec![#(#nodes),*], connections: vec![#(#connections),*] }),
96    )
97}

99/// Lower Serde settings to the same flash-backed scalars, lists and maps as task parameters.
100pub(super) fn settings<T: serde::Serialize>(
101    settings: &Option<T>,
102) -> cu29_traits::CuResult<TokenStream> {
103    fn lower(v: serde_value::Value) -> Value {
104        use serde_value::Value as V;
105        match v {
106            V::Bool(v) => v.into(),
107            V::Char(v) => v.into(),
108            V::String(v) => v.into(),
109            V::U8(v) => v.into(),
110            V::U16(v) => v.into(),
111            V::U32(v) => v.into(),
112            V::U64(v) => v.into(),
113            V::I8(v) => v.into(),
114            V::I16(v) => v.into(),
115            V::I32(v) => v.into(),
116            V::I64(v) => v.into(),
117            V::F32(v) => v.into(),
118            V::F64(v) => v.into(),
119            V::Bytes(v) => Value::Bytes(v),
120            V::Unit => Value::Unit,
121            V::Option(v) => Value::Option(v.map(|x| Box::new(lower(*x)))),
122            V::Newtype(v) => lower(*v),
123            V::Seq(v) => Value::Seq(v.into_iter().map(lower).collect()),
124            V::Map(v) => Value::Map(v.into_iter().map(|(k, v)| (lower(k), lower(v))).collect()),
125        }
126    }
127    if settings.is_none() {
128        return Ok(quote!(::core::option::Option::None));
129    }
130    let tree =
131        serde_value::to_value(settings).map_err(|e| cu29_traits::CuError::from(e.to_string()))?;
132    let expr = value(&lower(tree));
133    Ok(quote!(#expr.deserialize_into().map_err(|error| ::cu29::CuError::from(error.to_string()))?))
134}
