15fn value(v: &Value) -> TokenStream {
16    let ty = quote!(::cu29::__private::StaticValue);
17    match v {
18        Value::Bool(x) => quote!(#ty::Bool(#x)),
19        Value::Char(x) => quote!(#ty::Char(#x)),
20        Value::String(x) => quote!(#ty::String(#x)),
21        Value::Bytes(xs) => quote!(#ty::Bytes(&[#(#xs),*])),
22        Value::Unit => quote!(#ty::Unit),
23        Value::Option(None) => quote!(#ty::Option(None)),
24        Value::Option(Some(x)) => {
25            let x = value(x);
26            quote!(#ty::Option(Some(&#x)))
27        }
28        Value::Seq(xs) => {
29            let xs = xs.iter().map(value);
30            quote!(#ty::Seq(&[#(#xs),*]))
31        }
32        Value::Map(xs) => {
33            let xs = xs.iter().map(|(k, v)| {
34                let k = value(k);
35                let v = value(v);
36                quote!((#k,#v))
37            });
38            quote!(#ty::Map(&[#(#xs),*]))
39        }
40        Value::Number(n) => match n {
41            Number::I8(x) => quote!(#ty::I8(#x)),
42            Number::I16(x) => quote!(#ty::I16(#x)),
43            Number::I32(x) => quote!(#ty::I32(#x)),
44            Number::I64(x) => quote!(#ty::I64(#x)),
45            Number::U8(x) => quote!(#ty::U8(#x)),
46            Number::U16(x) => quote!(#ty::U16(#x)),
47            Number::U32(x) => quote!(#ty::U32(#x)),
48            Number::U64(x) => quote!(#ty::U64(#x)),
49            // A const block keeps the value promotable, so the borrowed table stays `'static`
50            // when it is emitted inline (monitor, bridge, codec and bundle parameters).
51            Number::F32(x) => {
52                let bits = x.0.to_bits();
53                quote!(#ty::F32(const { f32::from_bits(#bits) }))
54            }
55            Number::F64(x) => {
56                let bits = x.0.to_bits();
57                quote!(#ty::F64(const { f64::from_bits(#bits) }))
58            }
59            _ => quote!(compile_error!("unsupported static-config number")),
60        },
61    }
62}

64pub(super) fn component(config: Option<&Config<OwnedStorage>>) -> TokenStream {
65    let Some(config) = config else {
66        return quote!(None);
67    };
68    let mut entries: Vec<_> = config.entries().collect();
69    entries.sort_by_key(|(key, _)| *key);
70    let entries = entries.iter().map(|(key, v)| {
71        let v = value(v.as_ron());
72        quote!((#key,#v))
73    });
74    quote!(Some(::cu29::config::ComponentConfig::from_static(&[#(#entries),*])))
75}

77pub(super) fn topology(
78    config: &cu29_runtime::config::CuConfig,
79    mission: &str,
80) -> cu29_traits::CuResult<TokenStream> {
81    use cu29_runtime::monitoring::build_monitor_topology;
82    let mut topology = build_monitor_topology(config, mission)?;
83    topology.nodes.sort_by(|a, b| a.id.cmp(&b.id));
84    let m = quote!(::cu29::monitoring);
85    let optional = |s: Option<&str>| match s {
86        Some(s) => quote!(Some(#s.into())),
87        None => quote!(None),
88    };
89    let nodes = topology.nodes.iter().map(|n| {
90        let id = &n.id;
91        let name = optional(n.type_name.as_deref());
92        let kind = syn::Ident::new(&format!("{:?}", n.kind), proc_macro2::Span::call_site());
93        let inputs = &n.inputs;
94        let outputs = &n.outputs;
95        quote!(#m::MonitorNode { id: #id.into(), type_name: #name, kind: #m::ComponentType::#kind,
96            inputs: vec![#(#inputs.into()),*], outputs: vec![#(#outputs.into()),*] })
97    });
98    let connections = topology.connections.iter().map(|c| {
99        let (src,dst,msg) = (&c.src,&c.dst,&c.msg);
100        let src_port = optional(c.src_port.as_deref());
101        let dst_port = optional(c.dst_port.as_deref());
102        quote!(#m::MonitorConnection { src: #src.into(), dst: #dst.into(), msg: #msg.into(), src_port: #src_port, dst_port: #dst_port })
103    });
104    Ok(
105        quote!(#m::MonitorTopology { nodes: vec![#(#nodes),*], connections: vec![#(#connections),*] }),
106    )
107}

109/// Emit an expression of the field's Rust type. Struct patterns and enum
110/// matches are exhaustive so schema changes are checked when building the macro.
111pub(super) trait StaticExpr {
112    fn expr(&self) -> TokenStream;
113}

115impl<T: StaticExpr> StaticExpr for Option<T> {
116    fn expr(&self) -> TokenStream {
117        match self {
118            Some(value) => {
119                let value = value.expr();
120                quote!(Some(#value))
121            }
122            None => quote!(None),
123        }
124    }
125}
126
127impl<T: StaticExpr> StaticExpr for Vec<T> {
128    fn expr(&self) -> TokenStream {
129        let values = self.iter().map(StaticExpr::expr);
130        quote!(vec![#(#values),*])
131    }
132}

263impl StaticExpr for RuntimeConfig {
264    fn expr(&self) -> TokenStream {
265        let rate_target_hz = self.rate_target_hz.expr();
266        let planner = self.planner.expr();
267        // The macro reads config with std, even when generating bare-metal code.
268        let thread_pools = cfg!(feature = "std").then(|| {
269            let pools = self.thread_pools.expr();
270            quote!(thread_pools: #pools,)
271        });
272        #[cfg(feature = "clock-sync")]
273        let clock = {
274            let clock = self.clock.expr();
275            quote!(clock: #clock,)
276        };
277        #[cfg(not(feature = "clock-sync"))]
278        let clock = quote!();
279        // A downstream dependency can independently enable another runtime feature.
280        quote!({
281            #[allow(clippy::needless_update)]
282            let runtime = ::cu29::config::RuntimeConfig {
283                rate_target_hz: #rate_target_hz,
284                planner: #planner,
285                #thread_pools
286                #clock
287                ..Default::default()
288            };
289            runtime
290        })
291    }
292}

294impl StaticExpr for LoggingConfig {
295    fn expr(&self) -> TokenStream {
296        let Self {
297            enable_task_logging,
298            enable_keyframe_logging,
299            copperlist_count,
300            slab_size_mib,
301            section_size_mib,
302            keyframe_interval,
303            keyframe_capacity,
304            codecs: _,
305        } = self;
306        let copperlist_count = copperlist_count.expr();
307        let slab_size_mib = slab_size_mib.expr();
308        let section_size_mib = section_size_mib.expr();
309        let keyframe_capacity = keyframe_capacity.expr();
310        // Preserve the static loader's default for an explicitly absent interval.
311        let interval = keyframe_interval.unwrap_or(DEFAULT_KEYFRAME_INTERVAL);
312        // The macro emits codec registrations separately.
313        quote!(::cu29::config::LoggingConfig {
314            enable_task_logging: #enable_task_logging, enable_keyframe_logging: #enable_keyframe_logging,
315            copperlist_count: #copperlist_count, slab_size_mib: #slab_size_mib, section_size_mib: #section_size_mib,
316            keyframe_interval: Some(#interval), keyframe_capacity: #keyframe_capacity, codecs: vec![],
317        })
318    }
319}

321#[cfg(test)]
322mod tests {
323    use std::collections::{BTreeMap, BTreeSet};
324    use syn::{ExprStruct, Item};
325
326    #[test]
327    fn config_struct_mirrors_every_declared_field() {
328        let schema = syn::parse_file(include_str!("../../cu29_runtime/src/config.rs")).unwrap();
329        let structs: BTreeMap<_, _> = schema
330            .items
331            .into_iter()
332            .filter_map(|item| {
333                let Item::Struct(definition) = item else {
334                    return None;
335                };
336                Some((definition.ident.to_string(), definition.fields))
337            })
338            .collect();
339
340        // Read the declarations before cfg expansion so optional mirrors are checked too.
341        let emitter = syn::parse_file(include_str!("static_config.rs")).unwrap();
342        let mut checked = 0;
343        for item in emitter.items {
344            let Item::Macro(item) = item else { continue };
345            if !item.mac.path.is_ident("config_struct") {
346                continue;
347            }
348            let mirror: ExprStruct = item.mac.parse_body().unwrap();
349            let name = mirror.path.get_ident().unwrap().to_string();
350            let expected: BTreeSet<_> = structs[&name]
351                .iter()
352                .map(|field| field.ident.as_ref().unwrap().to_string())
353                .collect();
354            let actual: BTreeSet<_> = mirror
355                .fields
356                .iter()
357                .map(|field| match &field.member {
358                    syn::Member::Named(ident) => ident.to_string(),
359                    syn::Member::Unnamed(_) => panic!("expected named fields in {name}"),
360                })
361                .collect();
362            assert_eq!(actual, expected, "static config fields for {name}");
363            checked += 1;
364        }
365        assert!(checked > 0, "no config_struct! mirrors found");
366    }
