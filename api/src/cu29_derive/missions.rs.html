1//! Typed mission selection for live multi-mission applications.
2use proc_macro2::TokenStream;
3use quote::{format_ident, quote};
4use std::collections::BTreeSet;
5use syn::{Ident, Visibility};
6
7pub(crate) fn generate(app: &Ident, vis: &Visibility, modules: &[Ident]) -> TokenStream {
8    let mission = format_ident!("{app}Mission");
9    let active = format_ident!("{app}RunningMission");
10    let names: Vec<_> = modules.iter().map(ToString::to_string).collect();
11    let mut used = BTreeSet::new();
12    let variants: Option<Vec<_>> = names
13        .iter()
14        .map(|n| {
15            let variant = crate::utils::config_id_to_enum(n);
16            if !used.insert(variant.clone()) {
17                return None;
18            }
19            syn::parse_str::<Ident>(&variant).ok()
20        })
21        .collect();
22    let variants = variants.as_deref().unwrap_or(modules);
23    let api = quote!(::cu29::app);
24    let pre = quote!(::cu29::prelude);
25    let result = quote!(::cu29::CuResult);
26    let runner = format_ident!("{app}Runner");
27    quote! {
28        /// Runs the missions declared by this app.
29        #vis type #runner<'a, S = #pre::NoopSectionStorage, L = #pre::NoopLogger> =
30            ::cu29::supervisor::Supervisor<'a, #mission, S, L>;
31        #[allow(non_camel_case_types)]
32        #[derive(Clone, Copy, Debug, Eq, PartialEq)]
33        #vis enum #mission { #(#variants),* }
34
35        #[allow(non_camel_case_types)]
36        #[doc(hidden)]
37        #vis enum #active<S, L> {
38            #(#variants(#api::CuAppLifecycle<S, L, #modules::#app, #api::Running>)),*
39        }
40        impl<S: #pre::SectionStorage + 'static, L: #pre::UnifiedLogWrite<S> + 'static> #api::ActiveMission for #active<S,L> {
41            fn run_until(&mut self, requested: impl FnMut() -> bool) -> #result<#api::CuRunExit> {
42                match self { #(Self::#variants(app) => app.run_until(requested)),* }
43            }
44            fn wait_until_ready(&mut self, timeout: #pre::CuDuration, ready: impl FnMut() -> bool) -> ::core::result::Result<(), #api::CuWaitError> {
45                match self { #(Self::#variants(app) => app.wait_until_ready(timeout, ready)),* }
46            }
47            fn drain_until(&mut self, timeout: #pre::CuDuration, drained: impl FnMut() -> bool) -> ::core::result::Result<(), #api::CuWaitError> {
48                match self { #(Self::#variants(app) => app.drain_until(timeout, drained)),* }
49            }
50            fn stop(self) -> #result<()> {
51                match self { #(Self::#variants(app) => app.stop().map(drop).map_err(|failed| failed.error)),* }
52            }
53        }
54        impl ::cu29::supervisor::ConfiguredMission for #mission {
55            type Active<S: #pre::SectionStorage + 'static, L: #pre::UnifiedLogWrite<S> + 'static> = #active<S,L>;
56            fn activate<S: #pre::SectionStorage + 'static, L: #pre::UnifiedLogWrite<S> + 'static>(
57                self, settings: &mut ::cu29::supervisor::MissionSettings<'_>,
58                logger: ::cu29::__private::sync::Arc<::cu29::__private::sync::Mutex<L>>
59            ) -> #result<Self::Active<S,L>> {
60                match self { #(Self::#variants => Ok(#active::#variants(#modules::start_supervised::<S,L>(settings,logger)?))),* }
61            }
62        }
63    }
64}
