113#[derive(Debug, Default)]
114struct TimingAccumulator {
115    samples: u64,
116    min: Option<u64>,
117    max: Option<u64>,
118    sum: f64,
119    sum_squares: f64,
120    quantiles: WeightedQuantiles,
121}
122
123impl TimingAccumulator {
124    fn record(&mut self, value: u64) {
125        self.samples = self.samples.saturating_add(1);
126        self.min = Some(self.min.map_or(value, |current| current.min(value)));
127        self.max = Some(self.max.map_or(value, |current| current.max(value)));
128        let value_f64 = value as f64;
129        self.sum += value_f64;
130        self.sum_squares += value_f64 * value_f64;
131        self.quantiles.record(value);
132    }
133
134    fn stats(&self) -> DurationStats {
135        if self.samples == 0 {
136            return DurationStats::default();
137        }
138        let mean = self.sum / self.samples as f64;
139        let variance = (self.sum_squares / self.samples as f64 - mean * mean).max(0.0);
140        DurationStats {
141            min_ns: self.min,
142            max_ns: self.max,
143            mean_ns: Some(mean),
144            stddev_ns: Some(variance.sqrt()),
145            p50_ns: self.quantiles.quantile(0.50),
146            p95_ns: self.quantiles.quantile(0.95),
147        }
148    }
149}

151#[derive(Debug, Default)]
152struct WeightedQuantiles {
153    samples: Vec<u64>,
154    seen: u64,
155}
156
157impl WeightedQuantiles {
158    fn record(&mut self, value: u64) {
159        self.seen = self.seen.saturating_add(1);
160        if self.samples.len() < MAX_QUANTILE_SAMPLES {
161            self.samples.push(value);
162            return;
163        }
164        let candidate = splitmix64(self.seen) % self.seen;
165        if candidate < MAX_QUANTILE_SAMPLES as u64 {
166            self.samples[candidate as usize] = value;
167        }
168    }
169
170    fn quantile(&self, quantile: f64) -> Option<u64> {
171        if self.samples.is_empty() {
172            return None;
173        }
174        let mut values = self.samples.clone();
175        values.sort_unstable();
176        let index = ((values.len().saturating_sub(1)) as f64 * quantile).round() as usize;
177        values.get(index).copied()
178    }
179}

195#[derive(Debug, Default, Clone)]
196struct EdgeAccumulator {
197    samples: u64,
198    none_samples: u64,
199    valid_time_samples: u64,
200    total_raw_bytes: u64,
201    min_end_ns: Option<u64>,
202    max_end_ns: Option<u64>,
203}
204
205impl EdgeAccumulator {
206    fn record_sample(&mut self, payload_bytes: Option<u64>, end_time_ns: Option<u64>) {
207        self.samples = self.samples.saturating_add(1);
208        if let Some(bytes) = payload_bytes {
209            self.total_raw_bytes = self.total_raw_bytes.saturating_add(bytes);
210        } else {
211            self.none_samples = self.none_samples.saturating_add(1);
212        }
213
214        if let Some(end_ns) = end_time_ns {
215            self.valid_time_samples = self.valid_time_samples.saturating_add(1);
216            self.min_end_ns = Some(self.min_end_ns.map_or(end_ns, |min| min.min(end_ns)));
217            self.max_end_ns = Some(self.max_end_ns.map_or(end_ns, |max| max.max(end_ns)));
218        }
219    }
220
221    fn finalize(self, key: EdgeKey) -> EdgeLogStats {
222        let payload_samples = self.samples.saturating_sub(self.none_samples);
223        let avg_raw_bytes = if payload_samples > 0 {
224            Some(self.total_raw_bytes as f64 / payload_samples as f64)
225        } else {
226            None
227        };
228
229        let (rate_hz, throughput_bytes_per_sec) = if self.valid_time_samples >= 2 {
230            match (self.min_end_ns, self.max_end_ns) {
231                (Some(min_ns), Some(max_ns)) if max_ns > min_ns => {
232                    let duration_ns = max_ns - min_ns;
233                    let duration_secs = duration_ns as f64 / 1_000_000_000.0;
234                    let intervals = (self.valid_time_samples - 1) as f64;
235                    (
236                        Some(intervals / duration_secs),
237                        Some(self.total_raw_bytes as f64 / duration_secs),
238                    )
239                }
240                _ => (None, None),
241            }
242        } else {
243            (None, None)
244        };
245
246        EdgeLogStats {
247            src: key.src,
248            src_channel: key.src_channel,
249            dst: key.dst,
250            dst_channel: key.dst_channel,
251            msg: key.msg,
252            samples: self.samples,
253            none_samples: self.none_samples,
254            valid_time_samples: self.valid_time_samples,
255            total_raw_bytes: self.total_raw_bytes,
256            avg_raw_bytes,
257            rate_hz,
258            throughput_bytes_per_sec,
259        }
260    }
261}

263#[derive(Debug)]
264struct PerfAccumulator {
265    stats: CuDurationStatistics,
266    samples: u64,
267    valid_time_samples: u64,
268}
269
270impl PerfAccumulator {
271    fn new() -> Self {
272        Self {
273            stats: CuDurationStatistics::new(CuDuration(MAX_LATENCY_NS)),
274            samples: 0,
275            valid_time_samples: 0,
276        }
277    }
278
279    fn record_sample(&mut self, latency: Option<CuDuration>) {
280        self.samples = self.samples.saturating_add(1);
281        if let Some(latency) = latency {
282            self.stats.record(latency);
283            self.valid_time_samples = self.valid_time_samples.saturating_add(1);
284        }
285    }
286
287    fn finalize(&self) -> PerfStats {
288        let end_to_end = duration_stats_from(&self.stats);
289        let jitter = jitter_stats_from(&self.stats);
290
291        PerfStats {
292            samples: self.samples,
293            valid_time_samples: self.valid_time_samples,
294            end_to_end,
295            jitter,
296        }
297    }
298}

300pub fn compute_logstats<P>(
301    mut reader: impl Read,
302    config: &CuConfig,
303    mission: Option<&str>,
304) -> CuResult<LogStats>
305where
306    P: CopperListTuple + CuPayloadRawBytes,
307{
308    let graph = config.get_graph(mission)?;
309    let signature = build_graph_signature(graph, mission);
310    let output_slots = build_output_slots::<P>(graph)?;
311    let mut edge_accumulators = build_edge_accumulators(graph);
312    let mut perf = PerfAccumulator::new();
313    let resource_bindings = build_resource_bindings::<P>(config, graph);
314    let mut stage_accumulators = BTreeMap::<String, TimingAccumulator>::new();
315    let mut trace_duration_accumulator = TimingAccumulator::default();
316    let mut residual_accumulator = TimingAccumulator::default();
317    let mut representative_traces = Vec::<ExecutionTrace>::new();
318    let mut worst_trace: Option<ExecutionTrace> = None;
319    let mut active_intervals = Vec::<ExecutionInterval>::new();
320    let mut overlap_counts = BTreeMap::<(String, String, String), u64>::new();
321    let mut previous_end_ns = None;
322    let mut trace_count = 0u64;
323    let mut warned_lengths = false;
324
325    for culist in copperlists_reader::<P>(&mut reader) {
326        let payload_sizes = culist.msgs.payload_raw_bytes();
327        let cumsgs = culist.msgs.cumsgs();
328
329        let payload_len = payload_sizes.len();
330        let msg_len = cumsgs.len();
331        let slot_len = output_slots.len();
332        if !warned_lengths && (payload_len != msg_len || payload_len != slot_len) {
333            eprintln!(
334                "Warning: output mapping length mismatch (sizes={}, msgs={}, slots={})",
335                payload_len, msg_len, slot_len
336            );
337            warned_lengths = true;
338        }
339
340        let count = payload_len.min(msg_len).min(slot_len);
341
342        for idx in 0..count {
343            let slot = &output_slots[idx];
344            if slot.edges.is_empty() {
345                continue;
346            }
347            let payload_bytes = payload_sizes[idx];
348            let end_time_ns = extract_end_time_ns(cumsgs[idx].metadata());
349            for edge in &slot.edges {
350                if let Some(acc) = edge_accumulators.get_mut(edge) {
351                    acc.record_sample(payload_bytes, end_time_ns);
352                }
353            }
354        }
355
356        perf.record_sample(compute_end_to_end_latency(&cumsgs));
357
358        if let Some(trace) =
359            build_execution_trace(culist.id, &cumsgs, P::get_all_task_ids(), previous_end_ns)
360        {
361            for interval in &trace.intervals {
362                stage_accumulators
363                    .entry(interval.origin.clone())
364                    .or_default()
365                    .record(interval.end_ns - interval.start_ns);
366            }
367            trace_duration_accumulator.record(trace.duration_ns);
368            if let Some(residual) = trace.residual_before_ns {
369                residual_accumulator.record(residual);
370            }
371            record_resource_overlaps(
372                &trace,
373                &resource_bindings,
374                &mut active_intervals,
375                &mut overlap_counts,
376            );
377            previous_end_ns = trace.intervals.iter().map(|interval| interval.end_ns).max();
378            trace_count = trace_count.saturating_add(1);
379            sample_trace(&mut representative_traces, trace.clone(), trace_count);
380            if worst_trace
381                .as_ref()
382                .is_none_or(|worst| trace.duration_ns > worst.duration_ns)
383            {
384                worst_trace = Some(trace);
385            }
386        }
387    }
388
389    let edges = edge_accumulators
390        .into_iter()
391        .map(|(key, acc)| acc.finalize(key))
392        .collect();
393
394    let duration_p50 = trace_duration_accumulator
395        .stats()
396        .p50_ns
397        .unwrap_or_default();
398    let typical_trace = representative_traces
399        .into_iter()
400        .min_by_key(|trace| trace.duration_ns.abs_diff(duration_p50));
401    let mut traces = Vec::new();
402    if let Some(mut typical) = typical_trace {
403        typical.kind = "typical".to_string();
404        traces.push(typical);
405    }
406    if let Some(mut worst) = worst_trace
407        && traces
408            .first()
409            .is_none_or(|typical| typical.culist_id != worst.culist_id)
410    {
411        worst.kind = "slowest".to_string();
412        traces.push(worst);
413    }
414
415    let stages = stage_accumulators
416        .into_iter()
417        .map(|(origin, accumulator)| StageLogStats {
418            origin,
419            samples: accumulator.samples,
420            durations: accumulator.stats(),
421        })
422        .collect();
423    let resource_overlaps = overlap_counts
424        .into_iter()
425        .map(
426            |((resource, left, right), occurrences)| ResourceOverlapStats {
427                resource,
428                left,
429                right,
430                occurrences,
431            },
432        )
433        .collect();
434
435    Ok(LogStats {
436        schema_version: LOGSTATS_SCHEMA_VERSION,
437        config_signature: signature,
438        mission: mission.map(|value| value.to_string()),
439        edges,
440        perf: perf.finalize(),
441        schedule: Some(ScheduleLogStats {
442            stages,
443            traces,
444            residual_before: residual_accumulator.stats(),
445            resource_overlaps,
446        }),
447    })
448}

450fn build_execution_trace(
451    culist_id: u64,
452    msgs: &[&dyn cu29::prelude::ErasedCuStampedData],
453    origins: &[&str],
454    previous_end_ns: Option<u64>,
455) -> Option<ExecutionTrace> {
456    let mut grouped = BTreeMap::<String, (u64, u64)>::new();
457    for (msg, origin) in msgs.iter().zip(origins.iter()) {
458        let (Some(start), Some(end)) = (
459            extract_start_time_ns(msg.metadata()),
460            extract_end_time_ns(msg.metadata()),
461        ) else {
462            continue;
463        };
464        if end < start {
465            continue;
466        }
467        grouped
468            .entry((*origin).to_string())
469            .and_modify(|range| {
470                range.0 = range.0.min(start);
471                range.1 = range.1.max(end);
472            })
473            .or_insert((start, end));
474    }
475    if grouped.is_empty() {
476        return None;
477    }
478    let mut intervals = grouped
479        .into_iter()
480        .map(|(origin, (start_ns, end_ns))| ExecutionInterval {
481            origin,
482            start_ns,
483            end_ns,
484        })
485        .collect::<Vec<_>>();
486    intervals.sort_by_key(|interval| (interval.start_ns, interval.end_ns));
487    let original_count = intervals.len();
488    let intervals = select_current_execution_cluster(intervals);
489    let start_ns = intervals.first()?.start_ns;
490    let end_ns = intervals.iter().map(|interval| interval.end_ns).max()?;
491    let duration_ns = intervals
492        .iter()
493        .map(|interval| interval.end_ns.saturating_sub(interval.start_ns))
494        .sum();
495    Some(ExecutionTrace {
496        kind: String::new(),
497        culist_id,
498        duration_ns,
499        wall_span_ns: end_ns.saturating_sub(start_ns),
500        excluded_intervals: original_count.saturating_sub(intervals.len()) as u32,
501        residual_before_ns: previous_end_ns.and_then(|end| start_ns.checked_sub(end)),
502        intervals,
503    })
504}

506fn select_current_execution_cluster(intervals: Vec<ExecutionInterval>) -> Vec<ExecutionInterval> {
507    if intervals.len() < 2 {
508        return intervals;
509    }
510    let total_process_ns: u64 = intervals
511        .iter()
512        .map(|interval| interval.end_ns.saturating_sub(interval.start_ns))
513        .sum();
514    let split_gap_ns = total_process_ns
515        .saturating_mul(4)
516        .max(MIN_CURRENT_CLUSTER_GAP_NS);
517    let mut ranges = Vec::new();
518    let mut cluster_start = 0;
519    for index in 1..intervals.len() {
520        let gap = intervals[index]
521            .start_ns
522            .saturating_sub(intervals[index - 1].end_ns);
523        if gap > split_gap_ns {
524            ranges.push(cluster_start..index);
525            cluster_start = index;
526        }
527    }
528    ranges.push(cluster_start..intervals.len());
529    let selected = ranges
530        .into_iter()
531        .max_by_key(|range| (range.len(), intervals[range.end - 1].end_ns))
532        .unwrap_or(0..intervals.len());
533    intervals[selected].to_vec()
534}

536fn sample_trace(samples: &mut Vec<ExecutionTrace>, trace: ExecutionTrace, seen: u64) {
537    if samples.len() < MAX_REPRESENTATIVE_TRACES {
538        samples.push(trace);
539        return;
540    }
541    let hash = splitmix64(trace.culist_id);
542    let candidate = hash % seen;
543    if candidate < MAX_REPRESENTATIVE_TRACES as u64 {
544        samples[candidate as usize] = trace;
545    }
546}

548fn splitmix64(mut value: u64) -> u64 {
549    value = value.wrapping_add(0x9e3779b97f4a7c15);
550    value = (value ^ (value >> 30)).wrapping_mul(0xbf58476d1ce4e5b9);
551    value = (value ^ (value >> 27)).wrapping_mul(0x94d049bb133111eb);
552    value ^ (value >> 31)
553}

555fn build_resource_bindings<P: CopperListTuple>(
556    config: &CuConfig,
557    graph: &CuGraph,
558) -> HashMap<String, Vec<String>> {
559    let mut bindings = HashMap::new();
560    for origin in P::get_all_task_ids() {
561        if bindings.contains_key(*origin) {
562            continue;
563        }
564        let mut targets: Vec<String> = if let Some(rest) = origin.strip_prefix("bridge::") {
565            let bridge_id = rest.split("::").next().unwrap_or_default();
566            config
567                .bridges
568                .iter()
569                .find(|bridge| bridge.id == bridge_id)
570                .and_then(|bridge| bridge.resources.as_ref())
571                .map(|resources| resources.values().cloned().collect())
572                .unwrap_or_default()
573        } else {
574            graph
575                .get_node_id_by_name(origin)
576                .and_then(|node_id| graph.get_node(node_id))
577                .and_then(|node| node.get_resources())
578                .map(|resources| resources.values().cloned().collect())
579                .unwrap_or_default()
580        };
581        targets.sort();
582        targets.dedup();
583        bindings.insert((*origin).to_string(), targets);
584    }
585    bindings
586}

588fn record_resource_overlaps(
589    trace: &ExecutionTrace,
590    bindings: &HashMap<String, Vec<String>>,
591    active: &mut Vec<ExecutionInterval>,
592    counts: &mut BTreeMap<(String, String, String), u64>,
593) {
594    let Some(trace_start) = trace
595        .intervals
596        .iter()
597        .map(|interval| interval.start_ns)
598        .min()
599    else {
600        return;
601    };
602    active.retain(|interval| interval.end_ns > trace_start);
603
604    for (index, interval) in trace.intervals.iter().enumerate() {
605        for other in active.iter().chain(trace.intervals[..index].iter()) {
606            if interval.origin == other.origin
607                || interval.start_ns >= other.end_ns
608                || other.start_ns >= interval.end_ns
609            {
610                continue;
611            }
612            let Some(left_resources) = bindings.get(&interval.origin) else {
613                continue;
614            };
615            let Some(right_resources) = bindings.get(&other.origin) else {
616                continue;
617            };
618            for resource in left_resources {
619                if right_resources.contains(resource) {
620                    let (left, right) = if interval.origin <= other.origin {
621                        (interval.origin.clone(), other.origin.clone())
622                    } else {
623                        (other.origin.clone(), interval.origin.clone())
624                    };
625                    *counts.entry((resource.clone(), left, right)).or_default() += 1;
626                }
627            }
628        }
629    }
630    active.extend(trace.intervals.iter().cloned());
631}

641fn build_output_slots<P: CopperListTuple>(graph: &CuGraph) -> CuResult<Vec<OutputSlot>> {
642    let specs = P::get_output_specs();
643    if specs.is_empty() {
644        return build_output_slots_from_plan(graph);
645    }
646    Ok(specs
647        .iter()
648        .map(|spec| OutputSlot {
649            edges: graph
650                .edges()
651                .filter(|edge| edge.msg == spec.msg_type && edge_matches_origin(edge, spec.task_id))
652                .map(edge_key_from_connection)
653                .collect(),
654        })
655        .collect())

658fn edge_matches_origin(edge: &cu29::config::Cnx, origin: &str) -> bool {
659    if edge.src == origin {
660        return true;
661    }
662    let Some(rest) = origin.strip_prefix("bridge::") else {
663        return false;
664    };
665    let mut parts = rest.split("::");
666    let (Some(bridge), Some(direction), Some(channel), None) =
667        (parts.next(), parts.next(), parts.next(), parts.next())
668    else {
669        return false;
670    };
671    direction == "rx" && edge.src == bridge && edge.src_channel.as_deref() == Some(channel)
672}

684fn build_output_slots_from_plan(graph: &CuGraph) -> CuResult<Vec<OutputSlot>> {
685    let mut packs = Vec::new();
686    collect_output_packs_from_loop(&compute_runtime_plan(graph)?, graph, &mut packs)?;
687    packs.sort_by_key(|pack| pack.culist_index);
688    Ok(packs
689        .into_iter()
690        .flat_map(|pack| {
691            pack.msg_types.into_iter().map(move |msg| OutputSlot {
692                edges: graph
693                    .edges()
694                    .filter(|edge| edge.src == pack.src && edge.msg == msg)
695                    .map(edge_key_from_connection)
696                    .collect(),
697            })
698        })
699        .collect())
700}

702#[derive(Debug)]
703struct OutputPackInfo {
704    culist_index: u32,
705    src: String,
706    msg_types: Vec<String>,
707}
708
709fn collect_output_packs_from_loop(
710    loop_unit: &CuExecutionLoop,
711    graph: &CuGraph,
712    packs: &mut Vec<OutputPackInfo>,
713) -> CuResult<()> {
714    for step in &loop_unit.steps {
715        match step {
716            CuExecutionUnit::Step(step) => {
717                if let Some(output_pack) = &step.output_msg_pack {
718                    let node = graph
719                        .get_node(step.node_id)
720                        .ok_or_else(|| CuError::from("Missing node for output pack"))?;
721                    packs.push(OutputPackInfo {
722                        culist_index: output_pack.culist_index,
723                        src: node.get_id(),
724                        msg_types: output_pack.msg_types.clone(),
725                    });
726                }
727            }
728            CuExecutionUnit::Loop(inner) => {
729                collect_output_packs_from_loop(inner, graph, packs)?;
730            }
731        }
732    }
733    Ok(())
734}

751fn compute_end_to_end_latency(
752    msgs: &[&dyn cu29::prelude::ErasedCuStampedData],
753) -> Option<CuDuration> {
754    let start = msgs
755        .first()
756        .and_then(|msg| extract_start_time_ns(msg.metadata()))?;
757    let end = msgs
758        .last()
759        .and_then(|msg| extract_end_time_ns(msg.metadata()))?;
760    end.checked_sub(start).map(CuDuration::from_nanos)
761}

803fn build_graph_signature(graph: &CuGraph, mission: Option<&str>) -> String {
804    let mut parts = Vec::new();
805    parts.push(format!("mission={}", mission.unwrap_or("default")));
806
807    let mut nodes: Vec<_> = graph.get_all_nodes();
808    nodes.sort_by_key(|a| a.1.get_id());
809    for (_, node) in nodes {
810        parts.push(format!(
811            "node|{}|{}|{}",
812            node.get_id(),
813            node.get_type(),
814            flavor_label(node.get_flavor())
815        ));
816    }
817
818    let mut edges: Vec<String> = graph
819        .edges()
820        .map(|cnx| {
821            format!(
822                "edge|{}|{}|{}",
823                format_endpoint(cnx.src.as_str(), cnx.src_channel.as_deref()),
824                format_endpoint(cnx.dst.as_str(), cnx.dst_channel.as_deref()),
825                cnx.msg
826            )
827        })
828        .collect();
829    edges.sort();
830    parts.extend(edges);
831
832    let joined = parts.join("\n");
833    format!("fnv1a64:{:016x}", fnv1a64(joined.as_bytes()))
834}

837    match flavor {
838        Flavor::Task => "task",
839        Flavor::Bridge => "bridge",
840    }

844    match channel {
845        Some(ch) => format!("{node}/{ch}"),
846        None => node.to_string(),
847    }

850fn fnv1a64(data: &[u8]) -> u64 {
851    const OFFSET_BASIS: u64 = 0xcbf29ce484222325;
852    const PRIME: u64 = 0x100000001b3;
853    let mut hash = OFFSET_BASIS;
854    for byte in data {
855        hash ^= u64::from(*byte);
856        hash = hash.wrapping_mul(PRIME);
857    }
858    hash
859}
